  // ---------------- v158: BRING YOUR WEDDING TO LAZO (JC-LAZO-POPUP-1014) ----------------
  // One popup, one place at a time. Pick where you are coming from (the steps
  // name that site's own menus), then a tile per thing to bring. Each import
  // happens inside the popup: June reads it while you watch, you check and fix
  // what she found (vendors are an editable list: name, what they do, amount),
  // and it ends on a "done" card. Tiles remember what came across this visit.
  static const List<List<String>> _kImportFrom = <List<String>>[
    <String>['knot', 'The Knot'],
    <String>['zola', 'Zola'],
    <String>['weddingwire', 'WeddingWire'],
    <String>['joy', 'Joy'],
    <String>['other', 'Somewhere else'],
  ];

  String _importFromGuess(Map<String, dynamic> couple) {
    final dynamic acq = couple['acq'];
    final String s =
        (acq is Map ? (acq['source'] ?? acq['from'] ?? '') : '').toString().toLowerCase();
    if (s.contains('knot')) return 'knot';
    if (s.contains('zola')) return 'zola';
    if (s.contains('wire')) return 'weddingwire';
    if (s.contains('joy')) return 'joy';
    return '';
  }

  /// The site's own display date when June found one, else the ISO date as "March 27, 2027".
  String _importNiceDate(Map<String, dynamic> site) {
    final String shown = (site['dateDisplay'] ?? '').toString().trim();
    if (shown.isNotEmpty) return shown;
    final String iso = (site['dateIso'] ?? '').toString().trim();
    final DateTime? d = DateTime.tryParse(iso);
    if (d == null) return iso;
    const List<String> mo = <String>['January', 'February', 'March', 'April', 'May', 'June', 'July',
      'August', 'September', 'October', 'November', 'December'];
    return '${mo[d.month - 1]} ${d.day}, ${d.year}';
  }

  String _importFromName(String src) {
    for (final List<String> e in _kImportFrom) {
      if (e[0] == src) return e[1];
    }
    return 'the other site';
  }

  List<String> _guestExportSteps(String src) {
    switch (src) {
      case 'knot':
        return <String>['On The Knot, open Guest List.', 'Choose Manage (or the three-dot menu), then Export. A CSV downloads.'];
      case 'zola':
        return <String>['On Zola, open Guest List.', 'Choose Manage, then Export guest list. A CSV downloads.'];
      case 'weddingwire':
        return <String>['On WeddingWire, open Guests.', 'Choose Export, then CSV.'];
      case 'joy':
        return <String>['On Joy, open your guest list.', 'Export it as a CSV file.'];
      default:
        return <String>['Export your guest list as a spreadsheet (CSV).', 'Any columns work. A name column is enough.'];
    }
  }

  String _siteHint(String src) {
    switch (src) {
      case 'knot':
        return 'theknot.com/us/your-names';
      case 'zola':
        return 'zola.com/wedding/your-names';
      case 'weddingwire':
        return 'weddingwire.com/web/your-names';
      case 'joy':
        return 'withjoy.com/your-names';
      default:
        return 'Your website link, or paste the page text';
    }
  }

        Future<void> readVendors() async {
          if (vText.text.trim().isEmpty) return;
          up(() {
            busy = true;
            busyLabel = 'June is reading your list and looking for each vendor on Lazo…';
            err = '';
          });
          String why = '';
          final Map<String, dynamic>? r = await _juneImport(<String, dynamic>{
            'kind': 'vendors',
            'input': vText.text.trim(),
            'metroId': (couple['metroId'] ?? '').toString()
          }, onError: (String m) => why = m);
          final List<dynamic> vs = r != null && r['vendors'] is List ? r['vendors'] as List : <dynamic>[];
          up(() {
            busy = false;
            if (r == null) {
              err = why.isEmpty ? 'June couldn’t read that. Try again in a moment.' : why;
              return;
            }
            if (vs.isEmpty) {
              err = 'June couldn’t find vendors in that. Put each one on its own line.';
              return;
            }
            vRows.clear();
            for (final dynamic e in vs) {
              if (e is! Map) continue;
              final String cat = (e['category'] ?? '').toString();
              vRows.add(vRow(
                  (e['name'] ?? '').toString(),
                  kCategories.any((Map<String, String> c) => c['slug'] == cat) ? cat : '',
                  e['amount'] is num && (e['amount'] as num) > 0
                      ? (e['amount'] as num).round().toString()
                      : '',
                  (e['lazoVendorId'] ?? '').toString()));
            }
          });
        }

                (r['category'] as String).isNotEmpty)

          final List<Map<String, dynamic>> ready = readyVendors();

          up(() {
            busy = true;
            err = '';
          });
          try {
            final WriteBatch b = FirebaseFirestore.instance.batch();
            for (final Map<String, dynamic> r in ready) {
              final num? amt = num.tryParse(
                  (r['amount'] as TextEditingController).text.replaceAll(RegExp(r'[^0-9.]'), ''));
              b.set(
                  _planRef.doc(r['category'] as String),
                  <String, dynamic>{
                    'status': 'booked',
                    'vendorName': (r['name'] as TextEditingController).text.trim(),
                    if ((r['lazoVendorId'] as String).isNotEmpty) 'vendorId': r['lazoVendorId'],
                    if (amt != null && amt > 0) 'budgetActual': amt,
                    'bookedAt': FieldValue.serverTimestamp(),
                    'source': 'import',
                  },
                  SetOptions(merge: true));
            }
            await b.commit();
            await _coupleRef.set(<String, dynamic>{'importedAt': FieldValue.serverTimestamp()},
                SetOptions(merge: true));
            _logActivity(
                'added ${ready.length} booked vendor${ready.length == 1 ? '' : 's'} from ${_importFromName(src)}.',

                'vendors',

                'They’re marked booked, with the amounts you gave. Message the ones on Lazo straight from Vendor team.');

        Future<void> readSite() async {
          if (wText.text.trim().isEmpty) return;
          up(() {
            busy = true;
            busyLabel = 'June is reading your website…';
            err = '';
          });
          String why = '';
          final Map<String, dynamic>? r = await _juneImport(
              <String, dynamic>{'kind': 'website', 'input': wText.text.trim()},
              onError: (String m) => why = m);
          up(() {
            busy = false;
            if (r == null) {
              err = why.isEmpty ? 'June couldn’t read that. Try again in a moment.' : why;
              return;
            }
            wSite = r['site'] is Map ? Map<String, dynamic>.from(r['site'] as Map) : <String, dynamic>{};
            wRegs = r['registryLinks'] is List ? r['registryLinks'] as List : <dynamic>[];
            wKeep.clear();
            for (final List<String> f in siteFields) {
              if ((wSite[f[0]] ?? '').toString().trim().isNotEmpty) wKeep.add(f[0]);
            }
            if (wRegs.isNotEmpty) wKeep.add('registries');
            if (wKeep.isEmpty) {
              err = 'June couldn’t find wedding details there. If the site has a password, paste the page text instead of the link.';
            }
          });
        }

        Future<void> saveSite() async {
          up(() {
            busy = true;
            err = '';
          });
          try {
            final Map<String, dynamic> keep = <String, dynamic>{
              for (final String k in wKeep)
                if (k != 'registries') k: wSite[k],
            };
            if (wKeep.contains('dateIso') && (wSite['dateDisplay'] ?? '').toString().isNotEmpty) {
              keep['dateDisplay'] = wSite['dateDisplay'];
            }
            final bool live = await _saveImportedSite(
                keep, wKeep.contains('registries') ? wRegs : const <dynamic>[], couple);
            final int n = wKeep.length;
            finish(
                'website',
                '$n detail${n == 1 ? '' : 's'}',
                live ? 'Your Lazo website is updated' : 'Your website is ready to design',
                live
                    ? 'The details you kept are on your site now. Photos you’ll add yourself - they’re yours, and we can’t copy them from another site.'
                    : 'Open Website and pick a design - it’s filled in with what you kept.');
          } catch (e) {
            up(() {
              busy = false;
              err = 'Couldn’t save. Check your connection and try again.';
            });
          }
        }

        Future<void> readBudget(Map<String, dynamic> args) async {
          up(() {
            busy = true;
            busyLabel = 'June is reading your budget…';
            err = '';
          });
          String why = '';
          final Map<String, dynamic>? r = await _juneImport(args, onError: (String m) => why = m);
          up(() {
            busy = false;
            if (r == null) {
              err = why.isEmpty ? 'June couldn’t read that. Try again in a moment.' : why;
              return;
            }
            final List<dynamic> lines = r['lines'] is List ? r['lines'] as List : <dynamic>[];
            bCats = <String, Map<String, num>>{};
            bTotal = 0;
            for (final dynamic e in lines) {
              if (e is! Map) continue;
              final String slug = (e['category'] ?? '').toString();
              if (!kCategories.any((Map<String, String> c) => c['slug'] == slug)) continue;
              final num planned = e['planned'] is num ? e['planned'] as num : 0;
              final num paid = e['paid'] is num ? e['paid'] as num : 0;
              final Map<String, num> cur =
                  bCats.putIfAbsent(slug, () => <String, num>{'planned': 0, 'paid': 0});
              cur['planned'] = (cur['planned'] ?? 0) + planned;
              cur['paid'] = (cur['paid'] ?? 0) + paid;
              bTotal += planned;
            }
            if (bCats.isEmpty) {
              err = lines.isEmpty
                  ? 'June couldn’t find budget lines there.'
                  : 'June found lines but couldn’t match them to vendor categories. Try naming each line by vendor type (Photographer, Florist…).';
            }
          });
        }

        Future<void> pickBudgetFile() async {
          final FilePickerResult? res = await FilePicker.platform.pickFiles(
              type: FileType.custom,
              allowedExtensions: <String>['csv', 'txt', 'png', 'jpg', 'jpeg', 'webp'],
              withData: true);
          if (res == null || res.files.isEmpty || res.files.first.bytes == null) return;
          final String ext = (res.files.first.extension ?? '').toLowerCase();
          final Map<String, dynamic> args = <String, dynamic>{'kind': 'budget'};
          if (ext == 'csv' || ext == 'txt') {
            args['input'] = utf8.decode(res.files.first.bytes!, allowMalformed: true);
          } else {
            if (res.files.first.bytes!.lengthInBytes > 4500 * 1024) {
              up(() => err = 'That image is too large. A screenshot of the budget page works best.');
              return;
            }
            args['imageBase64'] = base64Encode(res.files.first.bytes!);
            args['imageType'] =
                ext == 'png' ? 'image/png' : (ext == 'webp' ? 'image/webp' : 'image/jpeg');
          }
          await readBudget(args);
        }

            });
          }

  /// Reads a guest-list export (Knot, Zola, WeddingWire, Joy or any CSV).
  List<Map<String, dynamic>> _parseGuestFile(Uint8List bytes, void Function(String) onErr) {
    String text = String.fromCharCodes(bytes);
    try {
      text = utf8.decode(bytes, allowMalformed: true);
    } catch (_) {}
    if (text.contains('\t') && !text.contains(',')) text = text.replaceAll('\t', ',');
    final List<List<String>> rows = _parseCsv(text);
    if (rows.length < 2) {
      onErr('That file has no guest rows.');
      return <Map<String, dynamic>>[];
    }
    final List<String> h = rows.first;
    final int iFirst = _colIndex(h, <String>['firstname', 'first']);
    final int iLast = _colIndex(h, <String>['lastname', 'last', 'surname']);
    final int iName = _colIndex(h, <String>['guestname', 'fullname', 'name', 'guest']);
    final int iEmail = _colIndex(h, <String>['email', 'emailaddress']);
    final int iPhone = _colIndex(h, <String>['phone', 'mobile', 'cell']);
    final int iParty = _colIndex(h, <String>['partyname', 'party', 'household', 'group', 'family']);
    final int iRsvp = _colIndex(h, <String>['rsvpstatus', 'rsvp', 'attending', 'response', 'status']);
    final int iMeal = _colIndex(h, <String>['meal', 'mealchoice', 'entree', 'dinner']);
    final int iPlus = _colIndex(h, <String>['plusones', 'plusone', 'additionalguests', 'guests', 'partysize']);
    final int iAddr = _colIndex(h, <String>['address', 'street', 'mailingaddress']);
    if (iName < 0 && iFirst < 0) {
      onErr('Couldn’t find a name column. Check the file has headers like First Name and Last Name.');
      return <Map<String, dynamic>>[];
    }
    String cell(List<String> r, int i) => (i >= 0 && i < r.length) ? r[i].trim() : '';
    final List<Map<String, dynamic>> guests = <Map<String, dynamic>>[];
    for (final List<String> r in rows.skip(1)) {
      // v139: Knot and Zola exports carry First / Last; a column that only
      // CONTAINS "name" (First Name) used to win and drop every surname
      String name = (iFirst >= 0 || iLast >= 0) ? ('${cell(r, iFirst)} ${cell(r, iLast)}').trim() : '';
      if (name.isEmpty && iName >= 0) name = cell(r, iName);
      if (name.isEmpty) continue;
      final String rsvpRaw = cell(r, iRsvp).toLowerCase();
      final String rsvp = rsvpRaw.contains('accept') ||
              rsvpRaw == 'yes' ||
              rsvpRaw.contains('attending') && !rsvpRaw.contains('not')
          ? 'yes'
          : (rsvpRaw.contains('decline') || rsvpRaw == 'no' || rsvpRaw.contains('not attending')
              ? 'no'
              : 'pending');
      int plus = 0;
      final String pRaw = cell(r, iPlus).toLowerCase();
      if (pRaw.isNotEmpty) {
        final int? n = int.tryParse(pRaw.replaceAll(RegExp(r'[^0-9]'), ''));
        if (n != null)
          plus = iPlus >= 0 && h[iPlus].toLowerCase().contains('size') ? (n - 1).clamp(0, 20) : n.clamp(0, 20);
        else if (pRaw == 'yes' || pRaw == 'true') plus = 1;
      }
      guests.add(<String, dynamic>{
        'name': name,
        'email': cell(r, iEmail),
        'phone': cell(r, iPhone),
        'party': cell(r, iParty),
        'rsvp': rsvp,
        'meal': cell(r, iMeal),
        'plusOnes': plus,
        'address': cell(r, iAddr),
        'source': 'import',
      });
    }
    if (guests.isEmpty) onErr('No guests found in that file.');
    return guests;
  }

  /// Adds guests not already on the list (by name). Returns how many were added.
  Future<int> _saveImportedGuests(List<Map<String, dynamic>> guests) async {
    final QuerySnapshot<Map<String, dynamic>> existing = await _coupleRef.collection('guests').get();
    final Set<String> have = existing.docs
        .map((QueryDocumentSnapshot<Map<String, dynamic>> d) =>
            (d.data()['name'] ?? '').toString().trim().toLowerCase())
        .toSet();
    int added = 0;
    WriteBatch b = FirebaseFirestore.instance.batch();
    int inBatch = 0;
    for (final Map<String, dynamic> g in guests) {
      final String key = g['name'].toString().toLowerCase();
      if (have.contains(key)) continue;
      have.add(key); // v139: no dupes within the file
      b.set(_coupleRef.collection('guests').doc(),
          <String, dynamic>{...g, 'createdAt': FieldValue.serverTimestamp()});
      added++;
      inBatch++;
      if (inBatch >= 400) {
        await b.commit();
        b = FirebaseFirestore.instance.batch();
        inBatch = 0;
      }
    }
    if (inBatch > 0) await b.commit();
    await _coupleRef.set(<String, dynamic>{'importedAt': FieldValue.serverTimestamp()}, SetOptions(merge: true));
    if (added > 0) _logActivity('imported $added guest${added == 1 ? '' : 's'} to the list.', open: 'guests');
    return added;
  }

  /// Saves the website details the couple kept. True when a live Lazo site was updated.
  Future<bool> _saveImportedSite(
      Map<String, dynamic> site, List<dynamic> regs, Map<String, dynamic> couple) async {
    final Map<String, dynamic> clean = <String, dynamic>{
      for (final MapEntry<String, dynamic> e in site.entries)
        if ((e.value ?? '').toString().trim().isNotEmpty) e.key: e.value.toString().trim(),
    };
    final QuerySnapshot<Map<String, dynamic>> q = await FirebaseFirestore.instance
        .collection('weddingSites')
        .where('coupleUid', isEqualTo: _cid)
        .limit(1)
        .get();
    if (clean.isNotEmpty) {
      if (q.docs.isNotEmpty) {
        await q.docs.first.reference
            .set(<String, dynamic>{...clean, 'updatedAt': FieldValue.serverTimestamp()}, SetOptions(merge: true));
      } else {
        await _coupleRef.set(<String, dynamic>{'siteDraft': clean}, SetOptions(merge: true));
      }
    }
    final Map<String, dynamic> cp = <String, dynamic>{'importedAt': FieldValue.serverTimestamp()};
    if ((clean['names'] ?? '').toString().isNotEmpty && (couple['names'] ?? '').toString().trim().isEmpty)
      cp['names'] = clean['names'];
    if ((clean['dateIso'] ?? '').toString().isNotEmpty && couple['weddingDate'] == null) {
      final DateTime? d = DateTime.tryParse(clean['dateIso'].toString());
      if (d != null) cp['weddingDate'] = Timestamp.fromDate(d);
    }
    await _coupleRef.set(cp, SetOptions(merge: true));
    if (regs.isNotEmpty) {
      final DocumentSnapshot<Map<String, dynamic>> c = await _coupleRef.get();
      final List<Map<String, dynamic>> cur = <Map<String, dynamic>>[
        for (final dynamic e in (c.data()?['registryLinks'] as List?) ?? const <dynamic>[])
          if (e is Map) Map<String, dynamic>.from(e)
      ];
      for (final dynamic e in regs) {
        if (e is! Map) continue;
        final String url = (e['url'] ?? '').toString();
        if (!_urlOk(url) || cur.any((Map<String, dynamic> l) => l['url'] == url)) continue;
        final List<String>? st = _storeFor(url);
        cur.add(<String, dynamic>{
          'label': (e['label'] ?? (st == null ? 'Registry' : st[1])).toString(),
          'url': url,
          'store': st == null ? '' : st[1]
        });
      }
      await _saveRegistry(cur);
    }
    return q.docs.isNotEmpty;
  }

    final WriteBatch b = FirebaseFirestore.instance.batch();
    for (final MapEntry<String, Map<String, num>> en in byCat.entries) {

          _planRef.doc(en.key),
          <String, dynamic>{
            'budgetPlanned': en.value['planned'],
            if ((en.value['paid'] ?? 0) > 0) 'budgetActual': en.value['paid']
          },
          SetOptions(merge: true));

    await b.commit();
    final DocumentSnapshot<Map<String, dynamic>> c = await _coupleRef.get();
    final num curTotal = c.data()?['budgetTotal'] is num ? c.data()!['budgetTotal'] as num : 0;
    await _coupleRef.set(<String, dynamic>{
      if (curTotal <= 0 && total > 0) 'budgetTotal': total,
      'importedAt': FieldValue.serverTimestamp()
    }, SetOptions(merge: true));
  }
