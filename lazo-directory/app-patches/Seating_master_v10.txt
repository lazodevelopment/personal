  void _animateMatrix(Matrix4 to) {
    if (_reducedMotion(context)) {
      _tc.value = to;
      return;
    }
    _vFrom = _tc.value.clone();
    _vTo = to;
    _vc
      ..stop()
      ..forward(from: 0);
  }

  void _onViewTick() {
    final Matrix4? a = _vFrom;
    final Matrix4? b = _vTo;
    if (a == null || b == null) return;
    final double t = Curves.easeOutCubic.transform(_vc.value);
    final Matrix4 m = Matrix4.zero();
    for (int i = 0; i < 16; i++) {
      m.storage[i] = a.storage[i] + (b.storage[i] - a.storage[i]) * t;
    }
    _tc.value = m;
  }

  Future<void> _resolveCouple() async {
    String cid = _uid!;
    try {
      final DocumentSnapshot<Map<String, dynamic>> u =
          await FirebaseFirestore.instance.collection('users').doc(_uid).get();
      final String shared = (u.data()?['coupleUid'] ?? '').toString();
      if (shared.isNotEmpty) {
        cid = shared;
      }
    } catch (e) {
      debugPrint('LAZO seating cid: ' + e.toString());
    }
    try {
      final DocumentSnapshot<Map<String, dynamic>> c =
          await FirebaseFirestore.instance.collection('couples').doc(cid).get();
      final Map<String, dynamic> d = c.data() ?? <String, dynamic>{};
      final dynamic names = d['names'];
      if (names is String && names.trim().isNotEmpty) {
        _coupleLabel = names.trim();
      } else if (names is List && names.isNotEmpty) {
        _coupleLabel = names.map((dynamic e) => e.toString()).join(' & ');
      }
      final dynamic wd = d['weddingDate'];
      if (wd is Timestamp) {
        _weddingDate = _longDate(wd.toDate());
      } else if (wd is String) {
        final DateTime? p = DateTime.tryParse(wd);
        _weddingDate = p == null ? wd : _longDate(p);
      }
    } catch (_) {}
    if (!mounted) {
      return;
    }
    _cid = cid;
    _subscribe();

  String _fld(String base) =>

  // v9: the two sides of the aisle are named after the couple. The guest
  // card's side field is 'partner1' / 'partner2' / 'both'; partner 1 is the
  // first name in couples.names, partner 2 the second.
  String _sideFirstName(String s) {
    if (s != 'partner1' && s != 'partner2') return '';
    final List<String> parts =
        _coupleLabel.split(RegExp(r'\s*(?:&|\band\b|\+)\s*'));
    final int i = s == 'partner1' ? 0 : 1;
    final String n = parts.length > i ? parts[i].trim() : '';
    if (n.isEmpty) return s == 'partner1' ? 'Partner 1' : 'Partner 2';
    return n.split(' ').first;
  }

  String _sideName(String s) {
    final String n = _sideFirstName(s);
    return n.isEmpty ? '' : '$n\u2019s side';
  }

  String _sideTag(_Guest g) => _plan == 'ceremony' ? _sideName(g.side) : '';

  int _sideCount(String side) => _guests.values

      .length;

  // v10: meal counts at a table, from guests.meal ("Chicken", "fish"...).
  String _mealOf(String gid) {
    final _Guest? g = _guests[gid];
    if (g == null) return '';
    final Map<String, dynamic> raw = _raw[g.parentId] ?? const <String, dynamic>{};
    if (g.isPlus) {
      final Map<String, dynamic> pa = raw[_fld('plusAssign')] is Map
          ? Map<String, dynamic>.from(raw[_fld('plusAssign')] as Map)
          : const <String, dynamic>{};
      final Map<String, dynamic> a = pa['${g.k}'] is Map ? Map<String, dynamic>.from(pa['${g.k}'] as Map) : const <String, dynamic>{};
      final String pm = (a['meal'] ?? '').toString().trim();
      if (pm.isNotEmpty) return pm;
    }
    return (raw['meal'] ?? '').toString().trim();
  }

  Map<String, int> _mealCounts(String tableId) {
    final Map<String, int> out = <String, int>{};
    _occ(tableId).forEach((int seat, String gid) {
      final String m = _mealOf(gid);
      if (m.isEmpty) return;
      final String k = m[0].toUpperCase() + m.substring(1).toLowerCase();
      out[k] = (out[k] ?? 0) + 1;
    });
    return out;
  }

  String _mealLine(String tableId) {
    final Map<String, int> c = _mealCounts(tableId);
    if (c.isEmpty) return '';
    final List<MapEntry<String, int>> e = c.entries.toList()
      ..sort((MapEntry<String, int> a, MapEntry<String, int> b) => b.value.compareTo(a.value));
    return e.map((MapEntry<String, int> x) => '${x.value} ${x.key.toLowerCase()}').join(' \u00b7 ');
  }

  List<String> _mealChoices() {
    final Set<String> out = <String>{};
    for (final _Guest g in _guests.values) {
      if (!_coming(g)) continue;
      final String m = _mealOf(g.id);
      if (m.isNotEmpty) out.add(m[0].toUpperCase() + m.substring(1).toLowerCase());
    }
    final List<String> l = out.toList()..sort();
    return l.take(6).toList();
  }

  int _mealCount(String meal) => _guests.values
      .where((_Guest g) => _coming(g) && _mealOf(g.id).toLowerCase() == meal.toLowerCase())
      .length;

  void _subscribePlan() {
    _seatSub?.cancel();
    _seatSub =
        _seatRef.snapshots().listen((DocumentSnapshot<Map<String, dynamic>> s) {
      // Our own writes echo back; while a drag is live, or the echo is still
      // pending, the local model is the truth.
      if (_dragActive || s.metadata.hasPendingWrites) {
        if (!_ready && mounted) {
          setState(() => _ready = true);
        }
        return;
      }
      final Map<String, dynamic> d = s.data() ?? <String, dynamic>{};
      final Map<String, dynamic> cv = d['canvas'] is Map
          ? Map<String, dynamic>.from(d['canvas'] as Map)
          : <String, dynamic>{};
      final List<_SeatTable> ts = <_SeatTable>[];
      if (d['tables'] is List) {
        for (final dynamic t in d['tables'] as List) {
          if (t is Map) {
            ts.add(_SeatTable.fromMap(Map<String, dynamic>.from(t)));
          }
        }
      }
      final List<_Area> ar = <_Area>[];
      if (d['areas'] is List) {
        for (final dynamic a in d['areas'] as List) {
          if (a is Map) {
            ar.add(_Area.fromMap(Map<String, dynamic>.from(a)));
          }
        }
      }
      if (!mounted) {
        return;
      }
      setState(() {
        _canvasW = cv['w'] is num ? (cv['w'] as num).toDouble() : 1600;
        _canvasH = cv['h'] is num ? (cv['h'] as num).toDouble() : 1000;
        _venueName = (d['venueName'] ?? '').toString();
        _tables = ts;
        _areas = ar;
        if (_plan == 'ceremony') {
          _procCache = <Map<String, dynamic>>[
            if (d['processional'] is List)
              for (final dynamic x in d['processional'] as List)
                if (x is Map) Map<String, dynamic>.from(x)
          ];
        }
        _bg = d['background'] is Map
            ? Map<String, dynamic>.from(d['background'] as Map)
            : <String, dynamic>{};
        _ready = true;
      });
    }, onError: (Object e) {
      debugPrint('LAZO seating doc: ' + e.toString());
      if (mounted) {
        setState(() => _ready = true);
      }
    });
  }

  void _switchPlan(String plan) {
    if (plan == _plan) {
      return;
    }
    _saveT?.cancel();
    _saveLayout();
    setState(() {
      _plan = plan;
      _sel = '';
      _picked = '';
      _pickedParty = '';
      _fitted = false;
      _undo.clear();
      if (_filter.startsWith('side') || _filter.startsWith('meal:')) {
        _filter = 'unseated';
      }
      _tables = <_SeatTable>[];
      _areas = <_Area>[];
      _bg = <String, dynamic>{};
      _rebuildGuests();
    });
    _subscribePlan();
  }

  // Every guest doc becomes one _Guest plus one virtual _Guest per plus-one
  // ("Sarah's +1"), so seats, counts and drags treat them alike.
  void _rebuildGuests() {
    final Map<String, _Guest> next = <String, _Guest>{};
    int fresh = 0;
    _raw.forEach((String id, Map<String, dynamic> d) {
      final _Guest g = _Guest.fromDoc(id, d, _fld('table'), _fld('tableName'),
          _fld('seat'), _fld('plusAssign'));
      next[id] = g;
      for (int k = 1; k <= g.plusOnes; k++) {
        next['$id#$k'] = g.plusOne(k);
      }
      final Timestamp? lo = _lastOpenedAt;
      if (lo != null &&
          g.rsvp == 'yes' &&
          g.updatedAt != null &&
          g.updatedAt!.compareTo(lo) > 0) {
        fresh++;
      }
    });
    _guests = next;
    _newYes = fresh;
  }

  void _scheduleSave() {
    _saveT?.cancel();
    _saveT = Timer(const Duration(milliseconds: 600), _saveLayout);
  }

  Future<void> _saveLayout() async {
    if (_cid.isEmpty) {
      return;
    }
    _saving = true;
    try {
      await _seatRef.set(<String, dynamic>{
        'canvas': <String, dynamic>{'w': _canvasW, 'h': _canvasH},
        'venueName': _venueName,
        'tables': _tables.map((_SeatTable t) => t.toMap()).toList(),
        'areas': _areas.map((_Area a) => a.toMap()).toList(),
        'background': _bg,
        'updatedAt': FieldValue.serverTimestamp(),
      }, SetOptions(merge: true));
      _scheduleShare();
    } catch (e) {
      _toast('Could not save the layout: ' + _errText(e));
    } finally {
      _saving = false;
    }
  }

  Future<void> _saveRules() async {
    try {
      await _mainRef
          .set(<String, dynamic>{'rules': _rules}, SetOptions(merge: true));
    } catch (e) {
      _toast('Could not save the rules: ' + _errText(e));
    }
  }

  // v6: undo covers the layout - tables, areas, background. Guest moves are
  // one drag to reverse and live on the guest docs.
  void _pushUndo() {
    _undo.add(jsonEncode(<String, dynamic>{
      't': _tables.map((_SeatTable t) => t.toMap()).toList(),
      'a': _areas.map((_Area a) => a.toMap()).toList(),
      'bg': _bg,
    }));
    if (_undo.length > 30) {
      _undo.removeAt(0);
    }
  }

  void _undoLayout() {
    if (_undo.isEmpty) {
      _toast('Nothing to undo.');
      return;
    }
    _tick(_Tick.light);
    final Map<String, dynamic> snap =
        Map<String, dynamic>.from(jsonDecode(_undo.removeLast()) as Map);
    setState(() {
      _tables = <_SeatTable>[
        for (final dynamic t in (snap['t'] as List? ?? const <dynamic>[]))
          if (t is Map) _SeatTable.fromMap(Map<String, dynamic>.from(t))
      ];
      _areas = <_Area>[
        for (final dynamic a in (snap['a'] as List? ?? const <dynamic>[]))
          if (a is Map) _Area.fromMap(Map<String, dynamic>.from(a))
      ];
      _bg = snap['bg'] is Map
          ? Map<String, dynamic>.from(snap['bg'] as Map)
          : <String, dynamic>{};
      _sel = '';
    });
    _scheduleSave();
  }

  void _writeSeat(WriteBatch b, String gid, String tableId, int seat) {
    final _SeatTable? t = _tableById(tableId);
    final String tn = t == null ? '' : t.name;
    if (gid.contains('#')) {
      final List<String> p = gid.split('#');
      b.set(
          _guestCol.doc(p[0]),
          <String, dynamic>{
            _fld('plusAssign'): <String, dynamic>{
              p[1]: <String, dynamic>{
                'table': tableId,
                'tableName': tn,
                'seat': seat,
              }
            }
          },
          SetOptions(merge: true));
      return;
    }
    b.set(
        _guestCol.doc(gid),
        <String, dynamic>{
          _fld('table'): tableId,
          _fld('tableName'): tn,
          _fld('seat'): seat,
          'seatedAt': FieldValue.serverTimestamp(),
        },
        SetOptions(merge: true));
  }

  void _writeUnseat(WriteBatch b, String gid) {
    if (gid.contains('#')) {
      final List<String> p = gid.split('#');
      b.set(
          _guestCol.doc(p[0]),
          <String, dynamic>{
            _fld('plusAssign'): <String, dynamic>{p[1]: FieldValue.delete()}
          },
          SetOptions(merge: true));
      return;
    }
    b.set(
        _guestCol.doc(gid),
        <String, dynamic>{
          _fld('table'): '',
          _fld('tableName'): '',
          _fld('seat'): -1,
        },
        SetOptions(merge: true));
  }

  Future<bool> _commit(WriteBatch b) async {
    try {
      await b.commit();
      _tick(_Tick.medium); // v8: the one pulse for a seat that changed
      return true;
    } catch (e) {
      _toast('Could not update: ' + _errText(e));
      return false;
    }
  }

  _SeatTable? _tableById(String id) {
    for (final _SeatTable t in _tables) {
      if (t.id == id) {
        return t;
      }
    }
    return null;
  }

  // v2 stored a table name in guests.table on some rows - honor both.
  String _guestTableId(_Guest g) {
    if (g.table.isEmpty) {
      return '';
    }
    if (_tableById(g.table) != null) {
      return g.table;
    }
    for (final _SeatTable t in _tables) {
      if (t.name.toLowerCase() == g.table.toLowerCase()) {
        return t.id;
      }
    }
    return '';
  }

  // seatIndex -> guestId for one table (a guest with a bad seat index gets
  // the first open one so nothing disappears).
  Map<int, String> _occ(String tableId) {
    final _SeatTable? t = _tableById(tableId);
    if (t == null) {
      return <int, String>{};
    }
    final Map<int, String> m = <int, String>{};
    final List<_Guest> loose = <_Guest>[];
    for (final _Guest g in _guests.values) {
      if (_guestTableId(g) != tableId) {
        continue;
      }
      if (g.seat >= 0 && g.seat < t.seats && !m.containsKey(g.seat)) {
        m[g.seat] = g.id;
      } else {
        loose.add(g);
      }
    }
    for (final _Guest g in loose) {
      for (int i = 0; i < t.seats; i++) {
        if (!m.containsKey(i)) {
          m[i] = g.id;
          break;
        }
      }
    }
    return m;
  }

  int _freeSeats(String tableId) {
    final _SeatTable? t = _tableById(tableId);
    if (t == null) {
      return 0;
    }
    return t.seats - _occ(tableId).length;
  }

  bool _coming(_Guest g) => g.rsvp != 'no';

  int get _seatedCount => _guests.values
      .where((_Guest g) => _coming(g) && _guestTableId(g).isNotEmpty)
      .length;

  int get _comingCount => _guests.values.where(_coming).length;

  int get _pendingCount => _guests.values
      .where((_Guest g) => g.rsvp != 'yes' && g.rsvp != 'no')
      .length;

  List<_Guest> get _unseated => _guests.values
      .where((_Guest g) => _coming(g) && _guestTableId(g).isEmpty)
      .toList()
    ..sort(_guestSort);

  int _guestSort(_Guest a, _Guest b) {
    final int c =
        a.partyLabel.toLowerCase().compareTo(b.partyLabel.toLowerCase());
    if (c != 0) {
      return c;
    }
    if (a.parentId != b.parentId) {
      return a.name.toLowerCase().compareTo(b.name.toLowerCase());
    }
    return a.k.compareTo(b.k);
  }

  // ---------------- v6: parties ----------------

  List<_Guest> _partyMembers(String key) => _guests.values
      .where((_Guest g) => g.partyKey == key && _coming(g))
      .toList()
    ..sort(_guestSort);

  String _partyLabel(String key) {
    final List<_Guest> m = _partyMembers(key);
    if (m.isEmpty) {
      return 'A party';
    }
    if (m.first.party.isNotEmpty) {
      return m.first.party;
    }
    return m.length == 1 ? m.first.name : '${m.first.name} + ${m.length - 1}';
  }

  List<String> _partyKeys() {
    final Set<String> keys = <String>{};
    for (final _Guest g in _guests.values) {
      if (_coming(g)) {
        keys.add(g.partyKey);
      }
    }
    final List<String> out = keys.toList()
      ..sort((String a, String b) =>
          _partyLabel(a).toLowerCase().compareTo(_partyLabel(b).toLowerCase()));
    return out;
  }

  // Warnings per table from the keep-together / keep-apart rules.
  Map<String, List<String>> _computeWarnings() {
    final Map<String, List<String>> out = <String, List<String>>{};
    void add(String tid, String msg) {
      out.putIfAbsent(tid, () => <String>[]);
      if (!out[tid]!.contains(msg)) {
        out[tid]!.add(msg);
      }
    }

    Set<String> tablesOf(String key) => _partyMembers(key)
        .map(_guestTableId)
        .where((String t) => t.isNotEmpty)
        .toSet();
    for (final Map<String, dynamic> r in _rules) {
      final String a = (r['a'] ?? '').toString();
      final String b = (r['b'] ?? '').toString();
      final String kind = (r['kind'] ?? 'together').toString();
      if (a.isEmpty || b.isEmpty) {
        continue;
      }
      final Set<String> ta = tablesOf(a);
      final Set<String> tb = tablesOf(b);
      if (kind == 'together') {
        if (ta.isEmpty || tb.isEmpty) {
          continue;
        }
        final Set<String> all = <String>{...ta, ...tb};
        if (all.length > 1) {
          for (final String t in all) {
            add(t,
                '${_partyLabel(a)} and ${_partyLabel(b)} should sit together');
          }
        }
      } else {
        for (final String t in ta.intersection(tb)) {
          add(t, '${_partyLabel(a)} and ${_partyLabel(b)} should sit apart');
        }
      }
    }
    // v9: a guest seated across the aisle from the side on their guest card.
    if (_plan == 'ceremony') {
      for (final _SeatTable t in _tables) {
        if (t.side.isEmpty) {
          continue;
        }
        _occ(t.id).forEach((int seat, String gid) {
          final _Guest? g = _guests[gid];
          if (g == null) {
            return;
          }
          if ((g.side == 'partner1' || g.side == 'partner2') &&
              g.side != t.side) {
            add(t.id,
                '${g.name} is on ${_sideName(g.side)} on their guest card');
          }
        });
      }
    }
    // A party split across tables is worth a nudge even without a rule.
    for (final String key in _partyKeys()) {
      final List<_Guest> m = _partyMembers(key);
      if (m.length < 2 || m.first.party.isEmpty) {
        continue;
      }
      final Set<String> ts = tablesOf(key);
      if (ts.length > 1) {
        for (final String t in ts) {
          add(t, '${_partyLabel(key)} is split across tables');
        }
      }
    }
    return out;
  }

  String _newId(String prefix) {
    const String al = 'abcdefghijklmnopqrstuvwxyz0123456789';
    final StringBuffer b = StringBuffer(prefix);
    for (int i = 0; i < 8; i++) {
      b.write(al[_rnd.nextInt(al.length)]);
    }
    return b.toString();
  }

  Future<void> _pickGuestPhoto(_Guest g) async {
    final _Guest p = g.isPlus ? (_guests[g.parentId] ?? g) : g;
    FilePickerResult? res;
    try {
      res = await FilePicker.platform
          .pickFiles(type: FileType.image, withData: true);
    } catch (_) {
      _toast('Could not open files.');
      return;
    }
    if (res == null || res.files.isEmpty || res.files.first.bytes == null) {
      return;
    }
    final PlatformFile f = res.files.first;
    if (f.bytes!.length > 8 * 1024 * 1024) {
      _toast(
          'That photo is over 8MB - a smaller one works better here anyway.');
      return;
    }
    final String ext = f.name.toLowerCase().endsWith('.png') ? 'png' : 'jpg';
    final String path =
        'couples/$_cid/guest-${p.id}-${DateTime.now().millisecondsSinceEpoch}.$ext';
    try {
      final Reference r = FirebaseStorage.instance.ref(path);
      await r.putData(
          f.bytes!,
          SettableMetadata(
              contentType: ext == 'png' ? 'image/png' : 'image/jpeg'));
      final String url = await r.getDownloadURL();
      await _guestCol.doc(p.id).set(<String, dynamic>{
        'photoUrl': url,
        'photoPath': path,
        'updatedAt': FieldValue.serverTimestamp(),
      }, SetOptions(merge: true));
      _toast('${p.name}\u2019s photo is on their seat \u2713');
    } catch (e) {
      _toast('Could not save the photo: ' + _errText(e));
    }
  }

  String _initials(String name) {
    final List<String> parts = name
        .replaceAll('&', ' ')
        .split(RegExp(r'\s+'))
        .where((String w) => w.isNotEmpty && w.toLowerCase() != 'and')
        .toList();
    if (parts.isEmpty) {
      return '?';
    }
    if (parts.length == 1) {
      return parts.first[0].toUpperCase();
    }
    return (parts[0][0] + parts[1][0]).toUpperCase();
  }

  TextStyle _serif(
      {double size = 24,
      FontWeight weight = FontWeight.w600,
      Color color = plum,
      double height = 1.05}) {
    // v8: display sizes tighten a touch; body sizes stay as drawn.
    final double track = size >= 26 ? -0.015 * size : 0;
    try {
      return GoogleFonts.cormorantGaramond(
          fontSize: size,
          fontWeight: weight,
          color: color,
          height: height,
          letterSpacing: track,
          fontFeatures: const <FontFeature>[FontFeature.liningFigures()]);
    } catch (_) {
      return TextStyle(
          fontFamily: 'serif',
          fontSize: size,
          fontWeight: weight,
          color: color,
          height: height,
          letterSpacing: track);
    }
  }

  // ---------------- layout edits ----------------

  Offset _freeSpot(double w, double h) {
    // First open spot on a coarse grid; a fresh table never lands on another.
    for (double y = 80; y < _canvasH - h; y += 60) {
      for (double x = 80; x < _canvasW - w; x += 60) {
        final Rect r =
            Rect.fromLTWH(x - kPad, y - kPad, w + kPad * 2, h + kPad * 2);
        bool clash = false;
        for (final _SeatTable t in _tables) {
          final Size s = t.size;
          if (Rect.fromLTWH(t.x - kPad, t.y - kPad, s.width + kPad * 2,
                  s.height + kPad * 2)
              .overlaps(r)) {
            clash = true;
            break;
          }
        }
        if (!clash) {
          for (final _Area a in _areas) {
            if (Rect.fromLTWH(a.x, a.y, a.w, a.h).overlaps(r)) {
              clash = true;
              break;
            }
          }
        }
        if (!clash) {
          return Offset(x, y);
        }
      }
    }
    return Offset(80 + _rnd.nextDouble() * 200, 80 + _rnd.nextDouble() * 200);
  }

  int _nextTableNumber() {
    int n = 0;
    for (final _SeatTable t in _tables) {
      final RegExpMatch? m =
          RegExp(r'^(?:Table|Row) (\d+)$').firstMatch(t.name);
      if (m != null) {
        n = math.max(n, int.parse(m.group(1)!));
      }
    }
    return n + 1;
  }

  void _addTable(String shape) {
    _pushUndo();
    final int n = _nextTableNumber();

          ? 'Sweetheart'
          : (shape == 'row' ? 'Row $n' : 'Table $n'),

  }

  // Tables named "Table N" get renumbered in room order - top to bottom,
  // left to right. Named tables keep their names.
  void _renumber() {
    _pushUndo();
    final List<_SeatTable> order = _tables
        .where((_SeatTable t) =>
            RegExp(r'^(?:Table|Row) \d+$').hasMatch(t.name) || t.shape == 'row')
        .toList()
      ..sort((_SeatTable a, _SeatTable b) {
        final int rowA = (a.y / 120).floor();
        final int rowB = (b.y / 120).floor();
        if (rowA != rowB) {
          return rowA.compareTo(rowB);
        }
        return a.x.compareTo(b.x);
      });
    int n = 1;
    final WriteBatch b = FirebaseFirestore.instance.batch();
    bool any = false;
    setState(() {
      for (final _SeatTable t in order) {
        final String nm = (t.shape == 'row' ? 'Row ' : 'Table ') + '$n';
        n++;
        if (t.name == nm) {
          continue;
        }
        t.name = nm;
        // tableName on every guest at that table follows.
        _occ(t.id).forEach((int seat, String gid) {
          _writeSeat(b, gid, t.id, seat);
          any = true;
        });
      }
    });
    if (any) {
      _commit(b);
    }
    _scheduleSave();
    _toast('Renumbered in room order.');
  }

  int _seatCap(_SeatTable t) => t.shape == 'sweetheart'
      ? 2
      : (t.shape == 'row' || t.shape == 'king' || t.shape == 'serp' ? kMaxRow : kMaxSeats);

  Future<void> _setSeats(_SeatTable t, int seats) async {
    seats = seats.clamp(1, _seatCap(t)).toInt();
    if (seats == t.seats) {
      return;
    }
    _pushUndo();
    if (seats < t.seats) {
      final Map<int, String> occ = _occ(t.id);
      final List<String> bumped = <String>[];
      for (int i = seats; i < t.seats; i++) {
        final String? g = occ[i];
        if (g != null) {
          bumped.add(g);
        }
      }
      if (bumped.isNotEmpty) {
        // Slide them into any seat that stays, unseat only what won't fit.
        final Set<int> taken = occ.keys.where((int k) => k < seats).toSet();
        final WriteBatch b = FirebaseFirestore.instance.batch();
        int unseated = 0;
        for (final String g in bumped) {
          int spot = -1;
          for (int i = 0; i < seats; i++) {
            if (!taken.contains(i)) {
              spot = i;
              taken.add(i);
              break;
            }
          }
          if (spot >= 0) {
            _writeSeat(b, g, t.id, spot);
          } else {
            _writeUnseat(b, g);
            unseated++;
          }
        }
        await _commit(b);
        if (unseated > 0) {
          _toast(
              '$unseated guest${unseated == 1 ? '' : 's'} back to the unseated list.');
        }
      }
    }
    setState(() => t.seats = seats);
    _scheduleSave();
  }

  // ---------------- seating moves ----------------

  // Companions that travel with a guest when they move: their plus-ones that
  // sit with them now (or nowhere yet).
  List<String> _companions(String gid, String fromTable) {
    if (gid.contains('#')) {
      return <String>[];
    }
    final _Guest? g = _guests[gid];
    if (g == null) {
      return <String>[];
    }
    final List<String> out = <String>[];
    for (int k = 1; k <= g.plusOnes; k++) {
      final _Guest? p = _guests['$gid#$k'];
      if (p == null || !_coming(p)) {
        continue;
      }
      final String pt = _guestTableId(p);
      if (pt.isEmpty || pt == fromTable) {
        out.add(p.id);
      }
    }
    return out;
  }

  Future<void> _dropOnSeat(_DragGuest d, String tableId, int seat) async {
    final _SeatTable? t = _tableById(tableId);
    if (t == null) {
      return;
    }
    final Map<int, String> occ = _occ(tableId);
    final String? o = occ[seat];
    if (o == d.guestId) {
      return;
    }
    final WriteBatch b = FirebaseFirestore.instance.batch();
    final Set<int> taken = occ.keys.toSet();
    taken.add(seat);
    _writeSeat(b, d.guestId, tableId, seat);
    if (o != null) {
      if (d.fromTable.isNotEmpty && d.fromTable != tableId) {
        _writeSeat(b, o, d.fromTable, d.fromSeat);
      } else if (d.fromTable == tableId) {
        _writeSeat(b, o, tableId, d.fromSeat);
      } else {
        _writeUnseat(b, o);
      }
    }
    // plus-ones follow into the next open seats here
    int left = 0;
    for (final String c in _companions(d.guestId, d.fromTable)) {
      if (_guestTableId(_guests[c]!) == tableId) {
        continue;
      }
      int spot = -1;
      for (int i = seat + 1; i < seat + 1 + t.seats; i++) {
        final int s = i % t.seats;
        if (!taken.contains(s)) {
          spot = s;
          break;
        }
      }
      if (spot < 0) {
        left++;
        continue;
      }
      taken.add(spot);
      _writeSeat(b, c, tableId, spot);
    }
    await _commit(b);
    if (left > 0) {
      _toast(
          '$left plus-one${left == 1 ? '' : 's'} still need a seat - ${t.name} is full.');
    }
    if (_picked == d.guestId) {
      setState(() => _picked = '');
    }
  }

  Future<void> _dropOnTable(_DragGuest d, String tableId) async {
    final _SeatTable? t = _tableById(tableId);
    if (t == null) {
      return;
    }
    if (d.fromTable == tableId) {
      return;
    }
    final Map<int, String> occ = _occ(tableId);
    for (int i = 0; i < t.seats; i++) {
      if (!occ.containsKey(i)) {
        await _dropOnSeat(d, tableId, i);
        return;
      }
    }
    _toast('${t.name} is full - add a seat or pick another table.');
  }

  // A whole party onto a table: everyone not already there takes an open
  // seat, or the chart says which tables could take them all.
  Future<void> _placeParty(String key, String tableId,
      {int startSeat = 0}) async {
    final _SeatTable? t = _tableById(tableId);
    if (t == null) {
      return;
    }
    final List<_Guest> members = _partyMembers(key);
    final List<_Guest> moving =
        members.where((_Guest g) => _guestTableId(g) != tableId).toList();
    if (moving.isEmpty) {
      _toast('${_partyLabel(key)} is already at ${t.name}.');
      return;
    }
    final Map<int, String> occ = _occ(tableId);
    final int free = t.seats - occ.length;
    if (free < moving.length) {
      final List<String> fits = _tables
          .where((_SeatTable x) =>
              x.id != tableId && _freeSeats(x.id) >= members.length)
          .map((_SeatTable x) => x.name)
          .toList();
      _toast(
          '${t.name} has $free open seat${free == 1 ? '' : 's'}; ${_partyLabel(key)} needs ${moving.length}.${fits.isEmpty ? ' Add a seat or a table.' : ' Room at ${fits.take(3).join(', ')}.'}');
      return;
    }
    final WriteBatch b = FirebaseFirestore.instance.batch();
    final Set<int> taken = occ.keys.toSet();
    int cursor = startSeat.clamp(0, t.seats - 1).toInt();
    for (final _Guest g in moving) {
      int spot = -1;
      for (int i = 0; i < t.seats; i++) {
        final int s = (cursor + i) % t.seats;
        if (!taken.contains(s)) {
          spot = s;
          break;
        }
      }
      if (spot < 0) {
        break;
      }
      taken.add(spot);
      cursor = spot + 1;
      _writeSeat(b, g.id, tableId, spot);
    }
    if (await _commit(b)) {
      _toast('${_partyLabel(key)} seated at ${t.name}.');
    }
    if (_pickedParty == key) {
      setState(() => _pickedParty = '');
    }
  }

  Future<void> _unseat(String gid) async {
    final WriteBatch b = FirebaseFirestore.instance.batch();
    _writeUnseat(b, gid);
    final _Guest? g = _guests[gid];
    if (g != null && !g.isPlus) {
      for (int k = 1; k <= g.plusOnes; k++) {
        _writeUnseat(b, '$gid#$k');
      }
    }
    await _commit(b);
  }

  Future<void> _unseatParty(String key) async {
    final WriteBatch b = FirebaseFirestore.instance.batch();
    for (final _Guest g in _partyMembers(key)) {
      _writeUnseat(b, g.id);
    }
    await _commit(b);
  }

  void _tapSeat(String tableId, int seat, _Guest? occupant) {
    if (_pickedParty.isNotEmpty) {
      _placeParty(_pickedParty, tableId, startSeat: seat);
      return;
    }
    if (_picked.isNotEmpty) {
      final _Guest? g = _guests[_picked];
      if (g == null) {
        setState(() => _picked = '');
        return;
      }
      final String from = _guestTableId(g);
      _dropOnSeat(
          _DragGuest(g.id, from, from.isEmpty ? -1 : g.seat), tableId, seat);
      return;
    }
    _tick(_Tick.select);
    if (occupant != null) {
      _guestSheet(occupant);
    } else {
      _select('table:$tableId');
    }
  }

  // Parties fill tables in room order; a party never splits when a table
  // with room for all of it exists.
  Future<void> _seatEveryone() async {
    final List<_Guest> queue = _unseated;
    if (queue.isEmpty) {
      _toast('Everyone who said yes has a seat.');
      return;
    }
    final List<_SeatTable> order = List<_SeatTable>.from(_tables)
      ..sort((_SeatTable a, _SeatTable b) {
        final int c = a.y.compareTo(b.y);
        return c != 0 ? c : a.x.compareTo(b.x);
      });
    final Map<String, Set<int>> taken = <String, Set<int>>{
      for (final _SeatTable t in order) t.id: _occ(t.id).keys.toSet()
    };
    int freeAt(_SeatTable t) => t.seats - taken[t.id]!.length;
    int takeSeat(_SeatTable t) {
      for (int i = 0; i < t.seats; i++) {
        if (!taken[t.id]!.contains(i)) {
          taken[t.id]!.add(i);
          return i;
        }
      }
      return -1;
    }

    // group the queue by party, biggest parties first
    final Map<String, List<_Guest>> byParty = <String, List<_Guest>>{};
    for (final _Guest g in queue) {
      byParty.putIfAbsent(g.partyKey, () => <_Guest>[]).add(g);
    }
    final List<String> keys = byParty.keys.toList()
      ..sort((String a, String b) =>
          byParty[b]!.length.compareTo(byParty[a]!.length));
    final WriteBatch b = FirebaseFirestore.instance.batch();
    int placed = 0;
    for (final String key in keys) {
      final List<_Guest> ms = byParty[key]!;
      // prefer the table where the rest of this party already sits
      _SeatTable? home;
      for (final _SeatTable t in order) {
        if (t.shape == 'sweetheart') {
          continue;
        }
        final bool hasKin =
            _partyMembers(key).any((_Guest g) => _guestTableId(g) == t.id);
        if (hasKin && freeAt(t) >= ms.length) {
          home = t;
          break;
        }
      }
      home ??= order.cast<_SeatTable?>().firstWhere(
          (_SeatTable? t) =>
              t != null && t.shape != 'sweetheart' && freeAt(t) >= ms.length,
          orElse: () => null);
      if (home != null) {
        for (final _Guest g in ms) {
          final int s = takeSeat(home);
          if (s >= 0) {
            _writeSeat(b, g.id, home.id, s);
            placed++;
          }
        }
        continue;
      }
      // nowhere fits the whole party - fill what we can, table by table
      for (final _Guest g in ms) {
        for (final _SeatTable t in order) {
          if (t.shape == 'sweetheart') {
            continue;
          }
          final int s = takeSeat(t);
          if (s >= 0) {
            _writeSeat(b, g.id, t.id, s);
            placed++;
            break;
          }
        }
      }
    }
    if (placed == 0) {
      _toast('No open seats - add a table first.');
      return;
    }
    await _commit(b);
    final int left = queue.length - placed;
    _toast(left == 0
        ? 'Everyone seated - $placed placed by party. Drag anyone to fine-tune.'
        : '$placed seated, $left still need a table.');
  }

  Future<void> _applyAssignments(List<Map<String, dynamic>> assigns) async {
    // June's seats are per table in list order; the chart assigns indices.
    final Map<String, int> cursor = <String, int>{};
    final Map<String, Set<int>> taken = <String, Set<int>>{};
    WriteBatch b = FirebaseFirestore.instance.batch();
    int n = 0;
    int placed = 0;
    for (final Map<String, dynamic> a in assigns) {
      final String gid = (a['guest'] ?? '').toString();
      final String tid = (a['table'] ?? '').toString();
      final _SeatTable? t = _tableById(tid);
      if (t == null || !_guests.containsKey(gid)) {
        continue;
      }
      taken.putIfAbsent(tid, () => <int>{});
      int spot = -1;
      final int start = cursor[tid] ?? 0;
      for (int i = 0; i < t.seats; i++) {
        final int s = (start + i) % t.seats;
        if (!taken[tid]!.contains(s)) {
          spot = s;
          break;
        }
      }
      if (spot < 0) {
        continue;
      }
      taken[tid]!.add(spot);
      cursor[tid] = spot + 1;
      _writeSeat(b, gid, tid, spot);
      placed++;
      n++;
      if (n >= 400) {
        await _commit(b);
        b = FirebaseFirestore.instance.batch();
        n = 0;
      }
    }
    if (n > 0) {
      await _commit(b);
    }
    _toast('$placed seated by June. Drag anyone to fine-tune.');
  }

  void _scheduleShare() {
    if (_shareToken.isEmpty) {
      return;
    }
    _shareT?.cancel();
    _shareT = Timer(const Duration(milliseconds: 1500), _publishShare);
  }

  Future<String> _ensureShareToken() async {
    if (_shareToken.isNotEmpty) {
      return _shareToken;
    }
    const String al = 'abcdefghijklmnopqrstuvwxyz0123456789';
    final math.Random r = math.Random.secure();
    final StringBuffer b = StringBuffer();
    for (int i = 0; i < 24; i++) {
      b.write(al[r.nextInt(al.length)]);
    }
    final String tok = b.toString();
    await _mainRef
        .set(<String, dynamic>{'shareToken': tok}, SetOptions(merge: true));
    _shareToken = tok;
    await _publishShare();
    return tok;
  }

            'meal': (d['meal'] ?? '').toString(),
            'flags': <String>[
              if (d['accessible'] == true) 'accessible',
              if (d['highChair'] == true) 'high chair',
              if (d['kid'] == true) 'kid',
            ],
          });
        }
        final Map<String, dynamic> pa = d[fPA] is Map
            ? Map<String, dynamic>.from(d[fPA] as Map)
            : <String, dynamic>{};
        pa.forEach((String k, dynamic v) {
          if (v is Map && (v['table'] ?? '').toString() == t.id) {
            seats.add(<String, dynamic>{
              'seat': v['seat'] is num ? (v['seat'] as num).toInt() : -1,
              'name': '$name\u2019s +$k',
              'meal': '',
              'flags': <String>[],
            });
          }
        });
      });
      seats.sort((Map<String, dynamic> a, Map<String, dynamic> c) =>
          (a['seat'] as int).compareTo(c['seat'] as int));
      final Map<String, dynamic> m = t.toMap();
      m['guests'] = seats;
      tables.add(m);
    }
    return <String, dynamic>{
      'canvas': doc['canvas'] ?? <String, dynamic>{'w': 1600, 'h': 1000},
      'venueName': (doc['venueName'] ?? '').toString(),
      'background': doc['background'] ?? <String, dynamic>{},
      'areas': doc['areas'] ?? <dynamic>[],
      'tables': tables,
    };
  }

  Future<void> _uploadBackground(StateSetter setD) async {
    if (_bgBusy) {
      return;
    }
    try {
      final FilePickerResult? res = await FilePicker.platform.pickFiles(
        type: FileType.custom,
        allowedExtensions: <String>['jpg', 'jpeg', 'png', 'webp'],
        withData: true,
      );
      if (res == null || res.files.isEmpty) {
        return;
      }
      final PlatformFile f = res.files.first;
      final Uint8List? bytes = f.bytes;
      if (bytes == null) {
        return;
      }
      if (bytes.length > 8 * 1024 * 1024) {
        _toast(
            'Keep it under 8MB - a screenshot of the venue\u2019s PDF is plenty.');
        return;
      }
      setD(() => _bgBusy = true);
      final String n = f.name.toLowerCase();
      final String ctype = n.endsWith('.png')
          ? 'image/png'
          : (n.endsWith('.webp') ? 'image/webp' : 'image/jpeg');
      final String ext =
          n.endsWith('.png') ? 'png' : (n.endsWith('.webp') ? 'webp' : 'jpg');
      final Reference ref = FirebaseStorage.instance.ref(
          'couples/$_cid/seating-bg-$_plan-${DateTime.now().millisecondsSinceEpoch}.$ext');
      await ref.putData(bytes, SettableMetadata(contentType: ctype));
      final String url = await ref.getDownloadURL();
      _pushUndo();
      setState(() => _bg = <String, dynamic>{'url': url, 'opacity': 0.55});
      setD(() => _bgBusy = false);
      _scheduleSave();
      _toast('Floor plan added - place your tables over it.');
    } catch (e) {
      setD(() => _bgBusy = false);
      _toast('Upload failed: ' + _errText(e));
    }
  }

  Rect _contentBounds() {
    Rect? r;
    for (final _SeatTable t in _tables) {
      final Size s = t.size;
      final Rect b = Rect.fromLTWH(
          t.x - kPad, t.y - kPad, s.width + kPad * 2, s.height + kPad * 2);
      r = r == null ? b : r.expandToInclude(b);
    }
    for (final _Area a in _areas) {
      final Rect b = Rect.fromLTWH(a.x, a.y, a.w, a.h);
      r = r == null ? b : r.expandToInclude(b);
    }
    if (r == null && (_bg['url'] ?? '').toString().isNotEmpty) {
      return Rect.fromLTWH(0, 0, _canvasW, _canvasH);
    }
    return r ?? Rect.fromLTWH(0, 0, _canvasW, _canvasH);
  }

  void _fit() {
    // A first layout pass in FlutterFlow can hand the canvas a sliver of a
    // viewport; fitting to that pinned the room at minimum zoom in a corner.
    if (_viewport.width < 160 || _viewport.height < 160) {
      _fitted = false;
      return;
    }
    final Rect room = Rect.fromLTWH(0, 0, _canvasW, _canvasH);
    final Rect b = room.expandToInclude(_contentBounds()).inflate(40);
    final double s = math
        .min(_viewport.width / b.width, _viewport.height / b.height)
        .clamp(0.25, 1.6)
        .toDouble();
    final double tx = (_viewport.width - b.width * s) / 2 - b.left * s;
    final double ty = (_viewport.height - b.height * s) / 2 - b.top * s;
    _tc.value = Matrix4.identity()
      ..translate(tx, ty)
      ..scale(s);
    _fitted = true;
    _fitVp = _viewport;
  }

  void _zoom(double factor) {
    _movedView = true;
    final double s = _tc.value.getMaxScaleOnAxis();
    final double ns = (s * factor).clamp(0.25, 3.0).toDouble();
    final double f = ns / s;
    final Offset c = Offset(_viewport.width / 2, _viewport.height / 2);
    final Matrix4 m = Matrix4.identity()
      ..translate(c.dx, c.dy)
      ..scale(f)
      ..translate(-c.dx, -c.dy);
    _animateMatrix(m * _tc.value); // v8: glides instead of cutting
  }

  // The fit button: glide to the fitted view rather than cut to it.
  void _fitAnimated() {
    final Matrix4 before = _tc.value.clone();
    _fit();
    final Matrix4 after = _tc.value.clone();
    _tc.value = before;
    setState(() {});
    _animateMatrix(after);
  }

  String _esc(String s) => s
      .replaceAll('&', '&amp;')
      .replaceAll('<', '&lt;')
      .replaceAll('>', '&gt;')

  List<_SeatTable> get _roomOrder => List<_SeatTable>.from(_tables)

  String get _title =>

  String _summaryText() {
    final StringBuffer b = StringBuffer();
    b.writeln(_title);
    if (_venueName.isNotEmpty) {
      b.writeln('Venue: $_venueName');
    }
    b.writeln(
        '${_tables.length} table${_tables.length == 1 ? '' : 's'} \u00b7 $_seatedCount of $_comingCount seated');
    b.writeln('');
    for (final _SeatTable t in _roomOrder) {
      final Map<int, String> occ = _occ(t.id);
      b.writeln(
          '${t.name} - ${_SeatTable.shapeLabel(t.shape)}, ${occ.length} of ${t.seats} seats');
      for (int i = 0; i < t.seats; i++) {
        final _Guest? gg = occ[i] == null ? null : _guests[occ[i]!];
        b.writeln(
            '  ${i + 1}. ${gg == null ? '(open)' : gg.name}${gg != null && gg.note.isNotEmpty ? ' - ${gg.note}' : ''}');
      }
      b.writeln('');
    }
    final List<_Guest> un = _unseated;
    if (un.isNotEmpty) {
      b.writeln(
          'Not seated yet (${un.length}): ${un.map((_Guest g) => g.name).join(', ')}');
    }
    return b.toString();
  }

  // Headline + one line per table; fits the 3,000-character message rule.
  String _tableSummaryText() {
    final StringBuffer b = StringBuffer();
    if (_venueName.isNotEmpty) {
      b.writeln('Venue: $_venueName');
    }
    b.writeln(
        '${_tables.length} table${_tables.length == 1 ? '' : 's'} \u00b7 $_seatedCount of $_comingCount seated \u00b7 seat-by-seat list attached');
    for (final _SeatTable t in _roomOrder) {
      b.writeln('${t.name}: ${_occ(t.id).length} of ${t.seats}');
    }
    final int un = _unseated.length;
    if (un > 0) {
      b.writeln('Not seated yet: $un');
    }
    return b.toString();
  }

  String _svg() {
    final Rect bb = _contentBounds().inflate(40);
    final StringBuffer s = StringBuffer();
    s.write(
        '<svg xmlns="http://www.w3.org/2000/svg" xmlns:xlink="http://www.w3.org/1999/xlink" viewBox="${bb.left} ${bb.top} ${bb.width} ${bb.height}" width="100%" style="max-height:640px;background:#FAF6F0;border:1px solid #E6D6B8;border-radius:14px">');
    final String bgUrl = (_bg['url'] ?? '').toString();
    if (bgUrl.isNotEmpty) {
      final double op =
          _bg['opacity'] is num ? (_bg['opacity'] as num).toDouble() : .55;
      s.write(
          '<image href="${_esc(bgUrl)}" x="0" y="0" width="$_canvasW" height="$_canvasH" preserveAspectRatio="xMidYMid meet" opacity="$op"/>');
    }
    for (final _Area a in _areas) {
      s.write(
          '<rect x="${a.x}" y="${a.y}" width="${a.w}" height="${a.h}" rx="10" fill="${_Area.presetHex(a.kind)}" stroke="#D9B77C"/>');
      s.write(
          '<text x="${a.x + a.w / 2}" y="${a.y + a.h / 2}" text-anchor="middle" dominant-baseline="middle" font-family="Georgia,serif" font-size="16" fill="#52284F">${_esc(a.label)}</text>');
    }
    for (final _SeatTable t in _tables) {
      final Size sz = t.size;
      final Map<int, String> occ = _occ(t.id);
      final double cx = t.x + sz.width / 2;
      final double cy = t.y + sz.height / 2;
      s.write('<g transform="rotate(${t.rotation} $cx $cy)">');
      if (t.shape == 'round') {
        s.write(
            '<circle cx="$cx" cy="$cy" r="${sz.width / 2}" fill="#fff" stroke="#52284F" stroke-width="1.5"/>');
      } else if (t.shape == 'serp') {
        final StringBuffer d = StringBuffer();
        for (int k = 0; k <= 24; k++) {
          final double px = sz.width * k / 24;
          d.write('${k == 0 ? 'M' : 'L'}${(t.x + px).toStringAsFixed(1)} ${(t.y + _SeatTable.serpY(px, sz)).toStringAsFixed(1)} ');
        }
        s.write(
            '<path d="$d" fill="none" stroke="#52284F" stroke-width="${_SeatTable.kSerpHalf * 2 + 1.5}" stroke-linecap="round"/>'
            '<path d="$d" fill="none" stroke="#fff" stroke-width="${_SeatTable.kSerpHalf * 2 - 1.5}" stroke-linecap="round"/>');
      } else {
        s.write(
            '<rect x="${t.x}" y="${t.y}" width="${sz.width}" height="${sz.height}" rx="12" fill="#fff" stroke="#52284F" stroke-width="1.5"/>');
      }
      for (int i = 0; i < t.seats; i++) {
        final Offset c = t.seatCenter(i);
        final _Guest? gg = occ[i] == null ? null : _guests[occ[i]!];
        final double sx = t.x + c.dx;
        final double sy = t.y + c.dy;
        s.write(
            '<circle cx="$sx" cy="$sy" r="$kSeatR" fill="${gg == null ? '#fff' : '#52284F'}" stroke="#D9B77C"/>');
        s.write(
            '<text x="$sx" y="$sy" text-anchor="middle" dominant-baseline="middle" font-family="Arial,sans-serif" font-size="9" font-weight="700" fill="${gg == null ? '#6B5F72' : '#FAF6F0'}">${gg == null ? i + 1 : _esc(gg.isPlus ? '+${gg.k}' : _initials(gg.name))}</text>');
      }
      s.write('</g>');
      s.write(
          '<text x="$cx" y="${cy - 4}" text-anchor="middle" font-family="Georgia,serif" font-size="16" font-weight="600" fill="#52284F">${_esc(t.name)}</text>');
      s.write(
          '<text x="$cx" y="${cy + 14}" text-anchor="middle" font-family="Arial,sans-serif" font-size="11" fill="#6B5F72">${occ.length} of ${t.seats}</text>');
    }
    s.write('</svg>');
    return s.toString();
  }

  // The caterer's sheet: meals per table, allergies, high chairs, kids.
  String _catererHtml() {
    final Map<String, int> totals = <String, int>{};
    final StringBuffer rows = StringBuffer();
    int kids = 0, chairs = 0, access = 0;
    for (final _SeatTable t in _roomOrder) {
      final Map<int, String> occ = _occ(t.id);
      final Map<String, int> meals = <String, int>{};
      final List<String> notes = <String>[];
      int seatedHere = 0;
      for (int i = 0; i < t.seats; i++) {
        final _Guest? gg = occ[i] == null ? null : _guests[occ[i]!];
        if (gg == null) {
          continue;
        }
        seatedHere++;
        final Map<String, dynamic> d = _raw[gg.parentId] ?? <String, dynamic>{};
        final String meal = gg.isPlus
            ? 'TBD (plus-one)'
            : ((d['meal'] ?? '').toString().trim().isEmpty
                ? 'No choice yet'
                : d['meal'].toString().trim());
        meals[meal] = (meals[meal] ?? 0) + 1;
        totals[meal] = (totals[meal] ?? 0) + 1;
        if (!gg.isPlus) {
          final String al =
              (d['allergies'] ?? d['dietary'] ?? '').toString().trim();
          if (al.isNotEmpty) {
            notes.add('${gg.name}: $al');
          }
          if (d['kid'] == true) {
            kids++;
            notes.add('${gg.name}: kid\u2019s meal');
          }
          if (d['highChair'] == true) {
            chairs++;
            notes.add('${gg.name}: high chair');
          }
          if (d['accessible'] == true) {
            access++;
            notes.add('${gg.name}: accessible seat');
          }
        }
      }
      rows.write(
          '<tr><td class="tn">${_esc(t.name)}<div class="s">$seatedHere seated</div></td><td>');
      rows.write(meals.entries
          .map((MapEntry<String, int> e) => '${e.value} ${_esc(e.key)}')
          .join('<br>'));
      rows.write(
          '</td><td class="notes">${notes.map(_esc).join('<br>')}</td></tr>');
    }
    final StringBuffer b = StringBuffer(_htmlHead(
        'Caterer sheet - ${_coupleLabel.isEmpty ? 'our wedding' : _coupleLabel}',
        'table.c{width:100%;border-collapse:collapse;font-family:Arial,sans-serif;font-size:13px}'
            '.c th{text-align:left;font-size:10px;letter-spacing:1.4px;color:#6B5F72;padding:6px 8px;border-bottom:2px solid #E6D6B8}'
            '.c td{padding:9px 8px;border-bottom:1px solid #F0E8DD;vertical-align:top;line-height:1.5}'
            '.c td.tn{font-family:Georgia,serif;font-size:18px;color:#52284F;white-space:nowrap}.c .s{font-family:Arial,sans-serif;font-size:11px;color:#6B5F72}'
            '.c td.notes{color:#8A6A2F;font-size:12px}.tot{margin:10px 0 18px;font-family:Arial,sans-serif;font-size:13px}'));
    b.write('<h1>Caterer sheet</h1><div class="sub">${_subLine()}</div>');
    b.write(
        '<div class="tot"><b>Totals:</b> ${totals.entries.map((MapEntry<String, int> e) => '${e.value} ${_esc(e.key)}').join(' \u00b7 ')}');
    if (kids + chairs + access > 0) {
      b.write(
          '<br>$kids kid${kids == 1 ? '' : 's'} \u00b7 $chairs high chair${chairs == 1 ? '' : 's'} \u00b7 $access accessible seat${access == 1 ? '' : 's'}');
    }
    b.write(
        '</div><table class="c"><tr><th>TABLE</th><th>MEALS</th><th>NOTES</th></tr>');
    b.write(rows.toString());
    b.write('</table></body></html>');
    return b.toString();
  }

  void _openPrint(String kind) {
    if (_tables.isEmpty) {
      _toast('Lay out a table or two first.');
      return;
    }
    String h;
    switch (kind) {
      case 'escort':
        h = _escortHtml();
        break;
      case 'place':
        h = _placeCardsHtml();
        break;
      case 'tents':
        h = _tentsHtml();
        break;
      case 'caterer':
        h = _catererHtml();
        break;
      case 'usher':
        h = _usherHtml();
        break;
      default:
        h = _printHtml();
    }
    try {
      if (!kIsWeb) {
        Clipboard.setData(ClipboardData(text: _summaryText()));
        _toast('Copied the list - paste it into Notes or an email.');
        return;
      }
      final html.Blob blob = html.Blob(<dynamic>[h], 'text/html;charset=utf-8');
      final String url = html.Url.createObjectUrlFromBlob(blob);
      html.window.open(url, '_blank');
    } catch (e) {
      _toast('Could not open the print view: ' + _errText(e));
    }
  }

  Future<void> _processionalSheet() async {
    if (_plan != 'ceremony') {
      _switchPlan('ceremony');
    }
    List<Map<String, dynamic>> steps = <Map<String, dynamic>>[];
    try {
      final DocumentSnapshot<Map<String, dynamic>> d = await _planRef('ceremony').get();
      if (d.data()?['processional'] is List) {
        for (final dynamic x in d.data()!['processional'] as List) {
          if (x is Map) steps.add(Map<String, dynamic>.from(x));
        }
      }
    } catch (_) {}
    if (!mounted) return;
    Future<void> save() async {
      try {
        await _planRef('ceremony').set(<String, dynamic>{
          'processional': steps,
          'updatedAt': FieldValue.serverTimestamp(),
        }, SetOptions(merge: true));
        _scheduleShare();
      } catch (e) {
        _toast('Could not save the processional: ' + _errText(e));
      }
    }

    Future<void> edit(StateSetter setS, [int? idx]) async {
      final TextEditingController who = TextEditingController(text: idx == null ? '' : (steps[idx]['who'] ?? '').toString());
      final TextEditingController note = TextEditingController(text: idx == null ? '' : (steps[idx]['note'] ?? '').toString());
      final bool? ok = await showDialog<bool>(
        context: context,
        builder: (BuildContext ctx) => AlertDialog(
          backgroundColor: ivory,
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(20)),
          title: Text(idx == null ? 'Add a step' : 'Edit step', style: const TextStyle(color: plum, fontSize: 17, fontWeight: FontWeight.w800)),
          content: Column(mainAxisSize: MainAxisSize.min, children: <Widget>[
            TextField(controller: who, autofocus: true, textCapitalization: TextCapitalization.sentences, style: const TextStyle(color: ink), decoration: _deco('Who', 'e.g. Grandparents')),
            const SizedBox(height: 10),
            TextField(controller: note, style: const TextStyle(color: ink), decoration: _deco('Note (optional)', 'to the front row, aisle side')),
          ]),
          actions: <Widget>[
            TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('Cancel', style: TextStyle(color: muted))),
            FilledButton(style: FilledButton.styleFrom(backgroundColor: gold, foregroundColor: plum), onPressed: () => Navigator.pop(ctx, true), child: const Text('Save')),
          ],
        ),
      );
      if (ok != true || who.text.trim().isEmpty) return;
      setS(() {
        if (idx == null) {
          steps.add(<String, dynamic>{'who': who.text.trim(), 'note': note.text.trim()});
        } else {
          steps[idx] = <String, dynamic>{'who': who.text.trim(), 'note': note.text.trim()};
        }
      });
      await save();
    }

    await showModalBottomSheet<void>(
      context: context,
      backgroundColor: ivory,
      isScrollControlled: true,
      constraints: const BoxConstraints(maxWidth: 560),
      shape: const RoundedRectangleBorder(borderRadius: BorderRadius.vertical(top: Radius.circular(26))),
      builder: (BuildContext ctx) => StatefulBuilder(
        builder: (BuildContext ctx2, StateSetter setS) => SafeArea(
          child: SizedBox(
            height: MediaQuery.of(ctx2).size.height * .82,
            child: Column(children: <Widget>[
              Padding(
                padding: const EdgeInsets.fromLTRB(22, 18, 14, 8),
                child: Row(children: <Widget>[

  Future<void> _buildCeremonyLayout(int rows, int seats, bool aisle) async {
    _pushUndo();
    // Anyone on the old rows goes back to the list.
    final WriteBatch b = FirebaseFirestore.instance.batch();
    bool any = false;
    for (final _SeatTable t in _tables) {
      _occ(t.id).forEach((int seat, String gid) {
        _writeUnseat(b, gid);
        any = true;
      });
    }
    if (any) {
      await _commit(b);
    }
    const double margin = 80;
    const double aisleW = 70;
    const double gap = 28;
    const double step = 100; // row pitch: body, chairs, breathing room
    final double rowW = (seats * 40 + 24).clamp(120, 1300).toDouble();
    final Size altar = _Area.presetSize('altar');
    final double totalW =
        aisle ? margin * 2 + rowW * 2 + gap * 2 + aisleW : margin * 2 + rowW;
    final double y0 = 60 + altar.height + 70;
    final List<_SeatTable> ts = <_SeatTable>[];
    final List<_Area> areas =
        _areas.where((_Area a) => a.kind != 'altar' && a.kind != 'aisle').toList();
    areas.add(_Area(
        id: _newId('a'),
        kind: 'altar',
        label: _Area.presetLabel('altar'),
        x: (totalW - altar.width) / 2,
        y: 60,
        w: altar.width,
        h: altar.height));
    String reservedFor(int r) => r == 1
        ? 'Parents & immediate family'
        : (r == 2 ? 'Grandparents & family' : '');
    if (aisle) {
      final double leftX = margin;
      final double aisleX = leftX + rowW + gap;
      final double rightX = aisleX + aisleW + gap;
      areas.add(_Area(
          id: _newId('a'),
          kind: 'aisle',
          label: _Area.presetLabel('aisle'),
          x: aisleX,
          y: y0 - 20,
          w: aisleW,
          h: rows * step + 20));
      for (int r = 1; r <= rows; r++) {
        for (final String side in const <String>['partner1', 'partner2']) {
          ts.add(_SeatTable(
            id: _newId('t'),
            name: 'Row $r',
            shape: 'row',
            seats: seats,
            x: side == 'partner1' ? leftX : rightX,
            y: y0 + (r - 1) * step,
            side: side,
            row: r,
            reserved: reservedFor(r),
          ));
        }
      }
    } else {
      for (int r = 1; r <= rows; r++) {
        ts.add(_SeatTable(
          id: _newId('t'),
          name: 'Row $r',
          shape: 'row',
          seats: seats,
          x: margin,
          y: y0 + (r - 1) * step,
          row: r,
          reserved: reservedFor(r),
        ));
      }
    }
    setState(() {
      _tables = ts;
      _areas = areas;
      _canvasW = math.max(1600.0, totalW);
      _canvasH = math.max(1000.0, y0 + rows * step + 120);
      _sel = '';
      _fitted = false;
      _movedView = false;
    });
    _scheduleSave();
    _toast(aisle
        ? '$rows rows a side, ${_sideFirstName('partner1')} left and ${_sideFirstName('partner2')} right. Drag family into the front rows.'
        : '$rows rows of $seats. Drag family into the front rows.');
  }

  // The usher's page: rows by side, front to back, reserved rows named, and
  // who has open seating on each side.
  String _usherHtml() {
    final StringBuffer b = StringBuffer(_htmlHead(
        'Usher list - ${_coupleLabel.isEmpty ? 'our wedding' : _coupleLabel}',
        '.sides{display:flex;gap:28px;align-items:flex-start}.side{flex:1;min-width:0}'
            'h2{font-family:Georgia,serif;font-size:20px;color:#52284F;margin:18px 0 6px}'
            'table.u{width:100%;border-collapse:collapse;font-family:Arial,sans-serif;font-size:13px}'
            '.u td{padding:6px 8px;border-bottom:1px solid #F0E8DD;vertical-align:top}.u td.r{width:64px;color:#52284F;font-weight:700;white-space:nowrap}'
            '.u .res{display:inline-block;background:#F3E7CE;color:#8A6A2F;border-radius:999px;padding:1px 8px;font-size:11px;font-weight:700;margin-left:6px}'
            '.u .open{color:#6B5F72}.note{font-family:Arial,sans-serif;font-size:12px;color:#6B5F72;margin-top:10px}'));
    b.write('<h1>Usher list</h1><div class="sub">${_subLine()}</div>');
    if (_procCache.isNotEmpty) {
      b.write('<h2>The processional</h2><ol style="font-family:Arial,sans-serif;font-size:13px;line-height:1.7">');
      for (final Map<String, dynamic> st in _procCache) {
        final String n = (st['note'] ?? '').toString();
        b.write('<li><b>${_esc((st['who'] ?? '').toString())}</b>${n.isEmpty ? '' : ' - ' + _esc(n)}</li>');
      }
      b.write('</ol>');
    }
    final List<String> sides = <String>[
      if (_tables.any((_SeatTable t) => t.side == 'partner1')) 'partner1',
      if (_tables.any((_SeatTable t) => t.side == 'partner2')) 'partner2',
      if (_tables.any((_SeatTable t) => t.side.isEmpty)) '',
    ];
    b.write('<div class="sides">');
    for (final String side in sides) {
      final List<_SeatTable> rows = _tables
          .where((_SeatTable t) => t.side == side)
          .toList()
        ..sort((_SeatTable p, _SeatTable q) => p.y.compareTo(q.y));
      b.write('<div class="side"><h2>${side.isEmpty ? 'Seating' : _esc(_sideName(side))}</h2><table class="u">');
      for (final _SeatTable t in rows) {
        final Map<int, String> occ = _occ(t.id);
        final List<String> names = <String>[];
        for (int i = 0; i < t.seats; i++) {
          final String? gid = occ[i];
          if (gid != null) {
            names.add(_esc(_guests[gid]?.name ?? 'Guest'));
          }
        }
        b.write('<tr><td class="r">${_esc(t.name)}</td><td>');
        if (t.reserved.isNotEmpty) {
          b.write('<span class="res">${_esc(t.reserved)}</span> ');
        }
        b.write(names.isEmpty
            ? '<span class="open">${t.reserved.isEmpty ? 'Open seating' : 'Held - no names yet'}</span>'
            : names.join(', '));
        if (names.isNotEmpty && names.length < t.seats) {
          b.write(' <span class="open">· ${t.seats - names.length} open</span>');
        }
        b.write('</td></tr>');
      }
      if (side.isNotEmpty) {
        final List<String> open = _guests.values
            .where((_Guest g) => _coming(g) && g.side == side && _guestTableId(g).isEmpty)
            .map((_Guest g) => _esc(g.name))
            .toList()
          ..sort();
        b.write('<tr><td class="r">Open</td><td class="open">${open.isEmpty ? 'Everyone else on this side sits anywhere behind the reserved rows.' : open.join(', ')}</td></tr>');
      }
      b.write('</table></div>');
    }
    b.write('</div>');
    b.write('<p class="note">Guests marked “both” on their guest card may sit on either side. Reserved rows are held for the people named; ushers seat everyone else from the front open row back.</p>');
    b.write('</body></html>');
    return b.toString();
  }

          .get();
      for (final QueryDocumentSnapshot<Map<String, dynamic>> d in qs.docs) {
        final Map<String, dynamic> m = d.data();
        final bool isBooked = (m['status'] ?? '').toString() == 'booked' ||
            (m['contractStatus'] ?? '').toString() == 'signed';

  Future<bool> _sendToThread(
      String inquiryId, String vendorName, String note) async {
    final DocumentReference<Map<String, dynamic>> ref =
        FirebaseFirestore.instance.collection('inquiries').doc(inquiryId);
    String link = '';
    try {
      link = '$kShareBase?t=${await _ensureShareToken()}';
    } catch (e) {
      debugPrint('LAZO seating token: ' + e.toString());
    }
    // The printable page rides along as a file when Storage lets us; the
    // list is in the message either way, so the planner never gets nothing.
    Map<String, dynamic>? att;
    try {
      final Uint8List bytes = Uint8List.fromList(utf8.encode(_printHtml()));
      final String path =
          'inquiries/$inquiryId/attachments/${DateTime.now().millisecondsSinceEpoch}_seating-chart.html';
      final Reference r = FirebaseStorage.instance.ref(path);
      await r.putData(bytes, SettableMetadata(contentType: 'text/html'));
      final String url = await r.getDownloadURL();
      att = <String, dynamic>{
        'url': url,
        'name': 'Seating chart.html',
        'type': 'text/html',
        'size': bytes.length,
        'path': path,
      };
    } catch (e) {
      debugPrint('LAZO seating attach: ' + e.toString());
      att = null;
    }
    final String head = note.isEmpty ? 'Here\u2019s our seating chart.' : note;
    final String linkLine =
        link.isEmpty ? '' : '\n\nLive chart (always current): $link';
    // Rules cap a message at 3,000 characters. The message carries the note
    // and a per-table summary; the seat-by-seat list rides in the attachment.
    String text = '$head$linkLine\n\n${_summaryText()}';
    if (text.length >= 2900) {
      text = '$head$linkLine\n\n${_tableSummaryText()}';
    }
    if (text.length >= 2900) {
      text =
          '$head$linkLine\n\n${_tables.length} tables \u00b7 $_seatedCount of $_comingCount guests seated. The full chart is attached.';
    }
    if (text.length >= 2900) {
      text = text.substring(0, 2880) + '\u2026';
    }
    try {
      final Map<String, dynamic> msg = <String, dynamic>{
        'senderRole': 'couple',
        'text': text,
        'at': FieldValue.serverTimestamp(),
      };
      if (att != null) {
        msg['attachment'] = att;
      }
      try {
        await ref.collection('messages').add(msg);
      } catch (e) {
        if (att == null) {
          rethrow;
        }
        msg.remove('attachment');
        await ref.collection('messages').add(msg);
      }
    } catch (e) {
      _toast('Could not send to $vendorName: ' + _errText(e));
      return false;
    }
    try {
      await ref.set(<String, dynamic>{
        'lastMessageAt': FieldValue.serverTimestamp(),
        'lastMessageRole': 'couple',
        'lastMessagePreview': 'Seating chart for review',
      }, SetOptions(merge: true));
    } catch (_) {
      // onMessageWritten keeps the thread header honest if the rules say no.
    }
    try {
      await _mainRef.set(<String, dynamic>{
        'sentToPlanner': <String, dynamic>{
          'inquiryId': inquiryId,
          'vendorName': vendorName,
          'at': FieldValue.serverTimestamp(),
        },
      }, SetOptions(merge: true));
    } catch (_) {}
    return true;
  }

  Future<void> _emailIt() async {
    final String subject = Uri.encodeComponent(
        'Seating chart${_coupleLabel.isEmpty ? '' : ' - $_coupleLabel'}');
    final String body = Uri.encodeComponent(
        '${_shareToken.isEmpty ? '' : 'Live chart: $_shareUrl\n\n'}${_summaryText()}');
    final Uri u = Uri.parse('mailto:?subject=$subject&body=$body');
    try {
      await launchUrl(u);
    } catch (_) {
      await Clipboard.setData(ClipboardData(text: _summaryText()));
      _toast('Copied the list - paste it into an email.');
    }
  }

  Widget _canvas() {
    return LayoutBuilder(builder: (BuildContext c, BoxConstraints cons) {
      final Size vp = Size(cons.maxWidth, cons.maxHeight);
      if (vp != _viewport) {
        _viewport = vp;
      }
      final bool refit = !_fitted ||
          (!_movedView &&
              (vp.width - _fitVp.width).abs() +
                      (vp.height - _fitVp.height).abs() >
                  24);
      if (refit) {
        WidgetsBinding.instance.addPostFrameCallback((_) {
          if (mounted) {
            setState(_fit);
          }
        });
      }
      final _Guest? pickedG = _picked.isEmpty ? null : _guests[_picked];
      final String pickedP = _pickedParty;
      final String bgUrl = (_bg['url'] ?? '').toString();
      final double bgOp =
          (_bg['opacity'] is num ? (_bg['opacity'] as num).toDouble() : 0.55)
              .clamp(0.1, 1.0)
              .toDouble();
      final bool empty = _tables.isEmpty && _areas.isEmpty && bgUrl.isEmpty;
      return Container(
        decoration: BoxDecoration(
          color: card,
          borderRadius: BorderRadius.circular(18),
          border: Border.all(color: goldLine),
        ),
        clipBehavior: Clip.antiAlias,
        child: Stack(children: <Widget>[
          Positioned.fill(
            child: InteractiveViewer(
              transformationController: _tc,
              constrained: false,
              boundaryMargin: const EdgeInsets.all(4000),
              onInteractionEnd: (ScaleEndDetails _) => _movedView = true,
              minScale: 0.25,
              maxScale: 3,
              child: GestureDetector(
                behavior: HitTestBehavior.opaque,
                onTap: () {
                  if (_picked.isNotEmpty || _pickedParty.isNotEmpty) {
                    setState(() {
                      _picked = '';
                      _pickedParty = '';

  // A banner that slides down into place and back up the same way out.
  Widget _bannerSwitch(Widget? child) {
    final bool reduced = _reducedMotion(context);
    return AnimatedSwitcher(
      duration: Duration(milliseconds: reduced ? 140 : 260),
      reverseDuration: Duration(milliseconds: reduced ? 140 : 200),
      switchInCurve: Curves.easeOutCubic,
      switchOutCurve: Curves.easeInCubic,
      transitionBuilder: (Widget c, Animation<double> a) => FadeTransition(
        opacity: a,
        child: reduced
            ? c
            : SlideTransition(
                position: Tween<Offset>(
                        begin: const Offset(0, -.35), end: Offset.zero)
                    .animate(a),
                child: c),
      ),
      child: child ?? const SizedBox.shrink(key: ValueKey<String>('none')),
    );
  }

  // A drop of a guest or a whole party.
  bool _acceptsDrop(Object? d, String tableId) {
    if (d is _DragGuest) {
      return d.fromTable != tableId;
    }
    if (d is _DragParty) {
      return true;
    }
    return false;
  }

  void _onTableDrop(Object d, String tableId) {
    if (d is _DragGuest) {
      _dropOnTable(d, tableId);
    } else if (d is _DragParty) {
      _placeParty(d.key, tableId);
    }
  }

  Widget _tableWidget(_SeatTable t) {
    final Size sz = t.size;
    final double bw = sz.width + kPad * 2;
    final double bh = sz.height + kPad * 2;
    final Map<int, String> occ = _occ(t.id);
    final bool sel = _sel == 'table:${t.id}';
    final int filled = occ.length;
    final bool full = filled >= t.seats;
    final List<String> warns = _warn[t.id] ?? const <String>[];
    final bool isRow = t.shape == 'row';

    final bool round = t.shape == 'round';
    final BorderRadius? rr =
        round ? null : BorderRadius.circular(isRow ? 8 : 16);
    // v7: three plates - shadow, rim, top - drawn as offset shapes (no blur;
    // blurred shadows painted as a grey sheet on web inside the zoomed plan).
    Widget body = Stack(clipBehavior: Clip.none, children: <Widget>[
      Positioned(
        left: 0,
        top: 4,
        child: Container(
          width: sz.width,
          height: sz.height,
          decoration: BoxDecoration(
            color: plumDeep.withOpacity(.14),
            shape: round ? BoxShape.circle : BoxShape.rectangle,
            borderRadius: rr,
          ),
        ),
      ),
      AnimatedContainer(
        // v8: selection and rule warnings ease their border on
        duration: const Duration(milliseconds: 180),
        curve: Curves.easeOut,
        width: sz.width,
        height: sz.height,
        decoration: BoxDecoration(
          shape: round ? BoxShape.circle : BoxShape.rectangle,
          borderRadius: rr,
          gradient: const LinearGradient(
              begin: Alignment.topLeft,
              end: Alignment.bottomRight,
              colors: <Color>[Colors.white, linen, linenEdge]),
          border: Border.all(
              color: warns.isNotEmpty
                  ? rose
                  : (sel ? plum : const Color(0xFFD8C7A6)),
              width: sel ? 2.5 : 1.6),
        ),
        alignment: Alignment.center,
        child: Container(
          margin: const EdgeInsets.all(5),
          decoration: BoxDecoration(
            shape: round ? BoxShape.circle : BoxShape.rectangle,
            borderRadius: round ? null : BorderRadius.circular(isRow ? 5 : 12),
            border: Border.all(color: goldLine.withOpacity(.9), width: 1),
          ),
          alignment: Alignment.center,
          child: Padding(
            padding: const EdgeInsets.symmetric(horizontal: 10),
            child: isRow
                ? Text(_rowLabel(t, filled),
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: TextStyle(
                        fontSize: 11,
                        color: full ? vGreen : muted,
                        fontWeight: FontWeight.w700))
                : Column(mainAxisSize: MainAxisSize.min, children: <Widget>[
                    Text(t.name,
                        textAlign: TextAlign.center,
                        maxLines: 2,
                        overflow: TextOverflow.ellipsis,
                        style: _serif(size: 18, height: 1)),
                    const SizedBox(height: 3),
                    Text('$filled of ${t.seats}',
                        style: TextStyle(
                            fontSize: 11,
                            color: full ? vGreen : muted,
                            fontWeight: FontWeight.w700)),
                    if (_plan == 'main' && _mealLine(t.id).isNotEmpty)
                      Padding(
                        padding: const EdgeInsets.only(top: 2),
                        child: Text(_mealLine(t.id),
                            textAlign: TextAlign.center,
                            maxLines: 2,
                            overflow: TextOverflow.ellipsis,
                            style: const TextStyle(
                                fontSize: 9.5,
                                color: Color(0xFF8A6A2F),
                                fontWeight: FontWeight.w700,
                                height: 1.2)),
                      ),
                    if (warns.isNotEmpty)
                      Padding(
                        padding: const EdgeInsets.only(top: 3),
                        child: Tooltip(
                          message: warns.join('\n'),
                          child: Container(
                            padding: const EdgeInsets.symmetric(
                                horizontal: 7, vertical: 2),
                            decoration: BoxDecoration(
                                color: rose.withOpacity(.12),
                                borderRadius: BorderRadius.circular(999),
                                border:
                                    Border.all(color: rose.withOpacity(.5))),
                            child: Text(
                                warns.length == 1
                                    ? 'Check'
                                    : '${warns.length} to check',
                                style: const TextStyle(
                                    fontSize: 9,
                                    color: rose,
                                    fontWeight: FontWeight.w800)),
                          ),
                        ),
                      ),
                  ]),
          ),
        ),
      ),
    ]);
    // v10: a serpentine is a painted band, not a box.
    if (t.shape == 'serp') {
      body = Stack(clipBehavior: Clip.none, children: <Widget>[
        CustomPaint(
          size: sz,
          painter: _SerpPainter(
              border: warns.isNotEmpty ? rose : (sel ? plum : const Color(0xFFD8C7A6)),
              width: sel ? 2.5 : 1.6),
        ),
        Positioned.fill(
          child: Center(
            child: Column(mainAxisSize: MainAxisSize.min, children: <Widget>[
              Text(t.name, textAlign: TextAlign.center, maxLines: 1, overflow: TextOverflow.ellipsis, style: _serif(size: 17, height: 1)),
              Text('$filled of ${t.seats}',

                    style: const TextStyle(fontSize: 9.5, color: Color(0xFF8A6A2F), fontWeight: FontWeight.w700)),
            ]),
          ),
        ),
      ]);

    // Captured in a final: the builder below must not see the later
    // reassignments of `body` (it would return the wrapper that contains it).
    final Widget plate =
        SizedBox(width: sz.width, height: sz.height, child: body);
    body = DragTarget<Object>(
      onWillAccept: (Object? d) {
        final bool ok = _acceptsDrop(d, t.id);
        if (ok) _tick(_Tick.select); // crossed into a table that can take them
        return ok;
      },
      onAccept: (Object d) => _onTableDrop(d, t.id),
      builder: (BuildContext c, List<Object?> cand, List<dynamic> rej) {
        if (cand.isEmpty) {
          return plate;
        }
        return Container(
          decoration: BoxDecoration(
            shape: t.shape == 'round' ? BoxShape.circle : BoxShape.rectangle,
            borderRadius: t.shape == 'round' ? null : BorderRadius.circular(14),
            border: Border.all(color: full ? rose : vGreen, width: 3),
          ),
          child: plate,
        );
      },
    );
    body = GestureDetector(
      behavior: HitTestBehavior.opaque,
      onTap: () {
        if (_pickedParty.isNotEmpty) {
          _placeParty(_pickedParty, t.id);
          return;
        }
        if (_picked.isNotEmpty) {
          final _Guest? g = _guests[_picked];
          if (g != null) {
            final String from = _guestTableId(g);
            _dropOnTable(
                _DragGuest(g.id, from, from.isEmpty ? -1 : g.seat), t.id);
          }
          return;
        }
        _select('table:${t.id}');
      },
      onPanStart: (DragStartDetails d) {
        _pushUndo();
        _tick(_Tick.medium); // picked up
        _dragActive = true;
        _select('table:${t.id}');
      },
      onPanUpdate: (DragUpdateDetails d) {
        setState(() {
          t.x = (t.x + d.delta.dx / _scale)
              .clamp(0.0, math.max(0.0, _canvasW - sz.width))
              .toDouble();
          t.y = (t.y + d.delta.dy / _scale)
              .clamp(0.0, math.max(0.0, _canvasH - sz.height))
              .toDouble();
        });
      },
      onPanEnd: (DragEndDetails d) {
        // v8: released - settle onto the grid (AnimatedPositioned eases it
        // from where the finger left it) and tick as it lands.
        _dragActive = false;
        _tick(_Tick.select);
        setState(() {
          t.x = (t.x / 10).round() * 10.0;
          t.y = (t.y / 10).round() * 10.0;
        });
        _scheduleSave();
      },
      onPanCancel: () => _dragActive = false,
      child: MouseRegion(cursor: SystemMouseCursors.grab, child: body),
    );

    final List<Widget> seats = <Widget>[];
    for (int i = 0; i < t.seats; i++) {
      final Offset c = t.seatCenter(i);
      seats.add(Positioned(
        left: c.dx + kPad - kSeatR,
        top: c.dy + kPad - kSeatR,
        child: Transform.rotate(
          angle: t.seatAngle(i),
          child: _seatWidget(t, i, occ[i]),
        ),
      ));
    }

    return AnimatedPositioned(
      // instant while a finger has it; a short ease when it snaps home
      duration: Duration(
          milliseconds: _dragActive || _reducedMotion(context) ? 0 : 180),
      curve: Curves.easeOutCubic,
      left: t.x - kPad,
      top: t.y - kPad,
      child: Transform.rotate(
        angle: t.rotation * math.pi / 180,
        child: SizedBox(
          width: bw,
          height: bh,
          child: Stack(clipBehavior: Clip.none, children: <Widget>[
            Positioned(left: kPad, top: kPad, child: body),
            ...seats,
          ]),
        ),
      ),
    );
  }

  Widget _seatWidget(_SeatTable t, int i, String? gid) {
    final _Guest? g = gid == null ? null : _guests[gid];
    final bool picking =
        (_picked.isNotEmpty || _pickedParty.isNotEmpty) && g == null;
    final bool isPicked = g != null && g.id == _picked;
    final bool pending = g != null && g.rsvp != 'yes' && g.rsvp != 'no';
    final Map<String, dynamic> raw = g == null
        ? <String, dynamic>{}
        : (_raw[g.parentId] ?? <String, dynamic>{});
    final bool flagged = g != null &&
        !g.isPlus &&
        (raw['accessible'] == true ||
            raw['highChair'] == true ||
            raw['kid'] == true);
    Widget core(bool hot) {
      final bool lit = hot || picking || isPicked;
      final Color tone = g == null
          ? (hot ? gold : const Color(0xFFD8C7A6))
          : _groupColor(g.partyLabel).withOpacity(pending ? .5 : 1);
      // v7: a chair - back at the top (the tile is rotated so it faces away
      // from the table), seat pad below, the guest in the seat.
      return Stack(clipBehavior: Clip.none, children: <Widget>[
        SizedBox(
          width: kSeatR * 2,
          height: kSeatR * 2,
          child: Stack(children: <Widget>[
            Positioned(
              left: 3,
              right: 3,
              top: 0,
              child: Container(
                height: 9,
                decoration: BoxDecoration(
                    color: tone,
                    borderRadius: const BorderRadius.vertical(
                        top: Radius.circular(8), bottom: Radius.circular(3))),
              ),
            ),
            Positioned(
              left: 0,
              right: 0,
              top: 6,
              bottom: 0,
              child: AnimatedContainer(
                // v8: the seat lights as a chip crosses into it
                duration: const Duration(milliseconds: 140),
                curve: Curves.easeOut,
                decoration: BoxDecoration(
                  color: g == null ? Colors.white : linen,
                  borderRadius: BorderRadius.circular(10),
                  border: Border.all(
                      color: lit
                          ? gold
                          : (g == null ? const Color(0xFFD8C7A6) : tone),
                      width: lit ? 2.2 : 1.4),
                ),
                alignment: Alignment.center,
                child: Transform.rotate(
                  angle: -t.seatAngle(i),
                  child: g == null
                      ? Text('${i + 1}',
                          style: const TextStyle(
                              color: muted,
                              fontSize: 9,
                              fontWeight: FontWeight.w800))
                      : _avatar(g, 22),
                ),
              ),
            ),
          ]),
        ),
        if (flagged)
          Positioned(
            right: -3,
            top: -3,
            child: Container(
              width: 12,
              height: 12,
              decoration: BoxDecoration(
                  color: gold,
                  shape: BoxShape.circle,
                  border: Border.all(color: Colors.white, width: 1)),
              child: Icon(
                  raw['accessible'] == true
                      ? Icons.accessible_rounded
                      : (raw['highChair'] == true
                          ? Icons.child_care_rounded
                          : Icons.cake_outlined),
                  size: 8,
                  color: plumDeep),
            ),
          ),
      ]);
    }

    Widget w = DragTarget<Object>(
      onWillAccept: (Object? d) {
        final bool ok =
            (d is _DragGuest && d.guestId != gid) || d is _DragParty;
        if (ok) _tick(_Tick.select); // crossed into a seat
        return ok;
      },
      onAccept: (Object d) {
        if (d is _DragGuest) {
          _dropOnSeat(d, t.id, i);
        } else if (d is _DragParty) {
          _placeParty(d.key, t.id, startSeat: i);
        }
      },
      builder: (BuildContext c, List<Object?> cand, List<dynamic> rej) =>
          core(cand.isNotEmpty),
    );
    if (g != null) {
      w = Draggable<Object>(
        data: _DragGuest(g.id, t.id, i),
        dragAnchorStrategy: pointerDragAnchorStrategy,
        onDragStarted: () => _tick(_Tick.medium),
        feedback: _dragFeedback(g),
        childWhenDragging: Opacity(opacity: .3, child: core(false)),
        child: w,
      );
    }
    return Tooltip(
      message: g == null
          ? 'Seat ${i + 1} - open'
          : <String>[
              g.name,
              if (g.party.isNotEmpty) g.party,
              if (pending) 'RSVP pending',
              if (raw['accessible'] == true) 'accessible seat',
              if (raw['highChair'] == true) 'high chair',
              if (raw['kid'] == true) 'kid\u2019s meal',
            ].join(' \u00b7 '),
      waitDuration: const Duration(milliseconds: 500),

  Widget _rail(
      {bool inSheet = false, VoidCallback? close, VoidCallback? refresh}) {
    final String q = _q.trim().toLowerCase();
    final List<_Guest> all = _guests.values.toList()..sort(_guestSort);
    final List<_Guest> shown = all.where((_Guest g) {
      final bool seated = _guestTableId(g).isNotEmpty;
      final bool declined = g.rsvp == 'no';
      if (_filter == 'declined') {
        return declined;
      }
      if (declined) {
        return false;
      }
      if (_filter == 'unseated' && seated) {
        return false;
      }
      if (_filter == 'seated' && !seated) {
        return false;
      }
      if (_filter == 'side1' && g.side != 'partner1') {
        return false;
      }
      if (_filter == 'side2' && g.side != 'partner2') {
        return false;
      }
      if (_filter.startsWith('meal:') &&
          _mealOf(g.id).toLowerCase() != _filter.substring(5).toLowerCase()) {
        return false;
      }
      if (q.isNotEmpty &&
          !('${g.name} ${g.party} ${g.group} ${g.note}'
              .toLowerCase()
              .contains(q))) {
        return false;
      }
      return true;
    }).toList();

  // The party line: name, size, and a handle that drags the whole household.
  Widget _partyHeader(
      List<_Guest> sec, bool inSheet, VoidCallback? close, VoidCallback poke) {
    final _Guest first = sec.first;
    final String key = first.partyKey;
    final List<_Guest> all = _partyMembers(key);
    final bool multi = all.length > 1 && first.party.isNotEmpty;
    final Color color = _groupColor(first.partyLabel);
    final Widget label = Padding(
      padding: const EdgeInsets.fromLTRB(4, 8, 4, 4),
      child: Row(children: <Widget>[
        Expanded(
          child: Text(
              (multi
                      ? '${first.party} \u00b7 ${all.length}'
                      : (first.party.isEmpty
                          ? (first.group.isEmpty ? 'No party' : first.group)
                          : first.party))
                  .toUpperCase(),
              maxLines: 1,
              overflow: TextOverflow.ellipsis,
              style: TextStyle(
                  color: color,
                  fontSize: 9.5,
                  letterSpacing: 1.4,
                  fontWeight: FontWeight.w800)),
        ),
        if (multi)
          _Press(
            onTap: () {
              if (inSheet && close != null) {
                close();
              }
              _partyFitSheet(key);
            },
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
              decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(999),
                  border: Border.all(color: goldLine)),
              child: Row(mainAxisSize: MainAxisSize.min, children: <Widget>[
                Icon(
                    inSheet
                        ? Icons.groups_rounded
                        : Icons.drag_indicator_rounded,
                    size: 12,
                    color: plum),
                const SizedBox(width: 3),
                Text(inSheet ? 'Seat together' : 'Drag all',
                    style: const TextStyle(
                        fontSize: 9.5,
                        color: plum,
                        fontWeight: FontWeight.w800)),
              ]),
            ),
          ),
      ]),
    );
    if (!multi || inSheet) {
      return label;
    }
    return Draggable<Object>(
      data: _DragParty(key),
      dragAnchorStrategy: pointerDragAnchorStrategy,
      onDragStarted: () => _tick(_Tick.medium),
      feedback: _feedbackChip('${all.length}', first.party, color),
      childWhenDragging: Opacity(opacity: .4, child: label),
      child: MouseRegion(cursor: SystemMouseCursors.grab, child: label),
    );
  }

  Widget _guestRow(
      _Guest g, bool inSheet, VoidCallback? close, VoidCallback poke) {
    final String tid = _guestTableId(g);
    final _SeatTable? t = tid.isEmpty ? null : _tableById(tid);
    final bool declined = g.rsvp == 'no';
    final bool pending = !declined && g.rsvp != 'yes';
    final bool isPicked = _picked == g.id;
    final Map<String, dynamic> raw = _raw[g.parentId] ?? <String, dynamic>{};
    final Widget chip = Opacity(
      opacity: pending ? .6 : 1,
      child: Container(
        margin: EdgeInsets.only(bottom: 4, left: g.isPlus ? 18 : 0),
        padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 8),
        decoration: BoxDecoration(
          color: isPicked ? plum.withOpacity(.08) : Colors.white,
          borderRadius: BorderRadius.circular(12),
          border: Border.all(color: isPicked ? plum : goldLine),
        ),
        child: Row(children: <Widget>[
          _avatar(g, 26,
              fill: declined ? muted.withOpacity(.4) : null, fontSize: 9.5),
          const SizedBox(width: 9),
          Expanded(
            child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: <Widget>[
                  Text(g.name,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: TextStyle(
                          color: declined ? muted : ink,
                          fontSize: 13,
                          decoration:
                              declined ? TextDecoration.lineThrough : null,
                          fontWeight: FontWeight.w600)),
                  if (t != null ||
                      declined ||
                      pending ||
                      g.note.isNotEmpty ||
                      _sideTag(g).isNotEmpty)
                    Text(
                        <String>[
                          if (t != null) '${t.name} \u00b7 seat ${g.seat + 1}',
                          if (_sideTag(g).isNotEmpty) _sideTag(g),
                          if (declined) 'Declined',
                          if (pending) 'RSVP pending',
                          if (g.note.isNotEmpty) g.note,
                        ].join(' \u00b7 '),
                        maxLines: 1,
                        overflow: TextOverflow.ellipsis,
                        style: TextStyle(
                            fontSize: 10.5,
                            color:
                                t != null ? vGreen : (pending ? amber : muted),
                            fontWeight: FontWeight.w700)),
                ]),
          ),
          if (!g.isPlus && raw['accessible'] == true)
            const Padding(
                padding: EdgeInsets.only(right: 4),
                child: Icon(Icons.accessible_rounded, size: 14, color: muted)),
          if (!g.isPlus && raw['highChair'] == true)
            const Padding(
                padding: EdgeInsets.only(right: 4),
                child: Icon(Icons.child_care_rounded, size: 14, color: muted)),
          _Press(
            onTap: () {
              if (inSheet && close != null) {
                close();
              }
              _guestSheet(g);
            },
            child: const Padding(
              padding: EdgeInsets.all(4),
              child: Icon(Icons.more_horiz_rounded, size: 18, color: muted),
            ),
          ),
        ]),
      ),
    );
    Widget row = _Press(
      scale: .985,
      dim: .92,
      tick: _Tick.select,
      onTap: () {
        if (declined) {
          _guestSheet(g);
          return;
        }
        if (inSheet) {
          if (close != null) {
            close();
          }
          setState(() {
            _picked = g.id;
            _pickedParty = '';
          });
          return;
        }
        setState(() {
          _picked = isPicked ? '' : g.id;
          _pickedParty = '';
        });
        poke();
      },
      child: MouseRegion(cursor: SystemMouseCursors.grab, child: chip),
    );
    if (!inSheet && !declined) {
      row = Draggable<Object>(
        data: _DragGuest(g.id, tid, tid.isEmpty ? -1 : g.seat),
        dragAnchorStrategy: pointerDragAnchorStrategy,
        onDragStarted: () => _tick(_Tick.medium),
        feedback: _dragFeedback(g),
        childWhenDragging: Opacity(opacity: .35, child: chip),
        child: row,
      );
    }
    return row;
  }

// ---------------- v8: MOTION ----------------
// Three rules, from the WWDC "Designing Fluid Interfaces" talk:
//  1. Answer the finger on pointer-DOWN, never only on release.
//  2. Critically damped by default. Overshoot only when something arrives
//     with momentum - here, the chip you just picked up.
//  3. Reduced-motion users get a cross-fade - never a slide, never a bounce.

bool _reducedMotion(BuildContext c) =>
    MediaQuery.maybeOf(c)?.disableAnimations ?? false;

// Haptics, in tiers, fired on the commit so the pulse and the visual change
// hit on the same frame. No-ops on web; no permission needed on phones.
//   select - crossing into a seat or table; a chip; the plan switch; a snap.
//   light  - an ordinary button; add, delete, undo.
//   medium - a pickup; a seat that actually changes; Send to your planner.
// v10: the serpentine band. Three plates like the boxes: shadow, rim, top.
class _SerpPainter extends CustomPainter {
  const _SerpPainter({required this.border, required this.width});
  final Color border;
  final double width;

  Path _path(Size s) {
    final Path p = Path();
    for (int k = 0; k <= 40; k++) {
      final double x = s.width * k / 40;
      final double y = _SeatTable.serpY(x, s);
      if (k == 0) {
        p.moveTo(x, y);
      } else {
        p.lineTo(x, y);
      }
    }
    return p;
  }

  @override
  void paint(Canvas canvas, Size size) {
    final Path p = _path(size);
    const double band = _SeatTable.kSerpHalf * 2;
    canvas.save();
    canvas.translate(0, 4);
    canvas.drawPath(p, Paint()..color = const Color(0xFF3D1C3B).withOpacity(.14)..style = PaintingStyle.stroke..strokeWidth = band..strokeCap = StrokeCap.round);
    canvas.restore();
    canvas.drawPath(p, Paint()..color = border..style = PaintingStyle.stroke..strokeWidth = band + width * 2..strokeCap = StrokeCap.round);
    canvas.drawPath(
        p,
        Paint()
          ..shader = const LinearGradient(colors: <Color>[Colors.white, Color(0xFFFBF7F0), Color(0xFFEFE4D2)]).createShader(Offset.zero & size)
          ..style = PaintingStyle.stroke
          ..strokeWidth = band
          ..strokeCap = StrokeCap.round);
    // the inner rim the boxes have, as two thin lines along the band's edges
    final Paint rim = Paint()
      ..color = const Color(0xFFE6D6B8).withOpacity(.9)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1;
    for (final double off in <double>[-(_SeatTable.kSerpHalf - 5), _SeatTable.kSerpHalf - 5]) {
      final Path e = Path();
      for (int k = 0; k <= 40; k++) {
        final double x = size.width * k / 40;
        final double y = _SeatTable.serpY(x, size) + off;
        if (k == 0) {
          e.moveTo(x, y);
        } else {
          e.lineTo(x, y);
        }
      }
      canvas.drawPath(e, rim);
    }
  }

  @override
  bool shouldRepaint(_SerpPainter old) => old.border != border || old.width != width;
}

enum _Tick { none, select, light, medium }

void _tick(_Tick t) {
  switch (t) {
    case _Tick.none:
      return;
    case _Tick.select:
      HapticFeedback.selectionClick();
      return;
    case _Tick.light:
      HapticFeedback.lightImpact();
      return;
    case _Tick.medium:
      HapticFeedback.mediumImpact();
      return;
  }
}

/// Wrap a button's onPressed. Null stays null, so disabled stays disabled.
VoidCallback? _h(VoidCallback? f, [_Tick t = _Tick.light]) => f == null
    ? null
    : () {
        _tick(t);
        f();
      };

/// Same for a typed value: a PopupMenuButton's onSelected.
ValueChanged<T>? _hv<T>(ValueChanged<T>? f, [_Tick t = _Tick.select]) =>
    f == null
        ? null
        : (T v) {
            _tick(t);
            f(v);
          };

/// Same for bool callbacks: a chip's onSelected, a Switch's onChanged.
ValueChanged<bool>? _hb(ValueChanged<bool>? f, [_Tick t = _Tick.select]) =>
    f == null
        ? null
        : (bool v) {
            _tick(t);
            f(v);
          };

/// A tappable that answers the instant the finger lands - 97% in 90ms on the
/// way down, a slower settle on the way up. Use it where the child paints its
/// own background; InkWell's ripple would land underneath and never show.
class _Press extends StatefulWidget {
  const _Press({
    super.key,
    required this.child,
    this.onTap,
    this.onLongPress,
    this.scale = .97,
    this.dim = .88,
    this.enabled = true,
    this.tick = _Tick.light,
    this.tickOnDown = false,
  });
  final Widget child;
  final VoidCallback? onTap;
  final VoidCallback? onLongPress;
  final double scale;
  final double dim;
  final bool enabled;
  final _Tick tick;
  // Harmony: the scale already fires on touch-down, so a control can put its
  // haptic there too instead of a frame after release. Off by default, so no
  // existing call site changes.
  final bool tickOnDown;
  @override
  State<_Press> createState() => _PressState();
}

class _PressState extends State<_Press> {
  bool _down = false;
  void _set(bool v) {
    if (_down != v && mounted) setState(() => _down = v);
  }

  @override
  Widget build(BuildContext context) {
    final bool on = _down && widget.enabled && !_reducedMotion(context);
    final VoidCallback? tap = widget.enabled ? widget.onTap : null;
    return MouseRegion(
      cursor: widget.enabled && widget.onTap != null
          ? SystemMouseCursors.click
          : MouseCursor.defer,
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTapDown: widget.enabled
            ? (_) {
                if (widget.tickOnDown) _tick(widget.tick);
                _set(true);
              }
            : null,
        onTapUp: (_) => _set(false),
        onTapCancel: () => _set(false),
        onTap: tap == null
            ? null
            : () {
                if (!widget.tickOnDown) _tick(widget.tick);
                tap();
              },
        onLongPress: widget.enabled ? widget.onLongPress : null,
        child: AnimatedScale(
          scale: on ? widget.scale : 1,
          duration: Duration(milliseconds: on ? 90 : 240),
          curve: on ? Curves.easeOut : Curves.easeOutCubic,
          child: AnimatedOpacity(
            opacity: on ? widget.dim : 1,
            duration: Duration(milliseconds: on ? 60 : 200),
            child: widget.child,
          ),
        ),
      ),
    );
  }
}

class _SeatTable {
  _SeatTable({
    required this.id,
    required this.name,
    required this.shape,
    required this.seats,
    required this.x,
    required this.y,
    this.rotation = 0,
    this.side = '',
    this.row = 0,
    this.reserved = '',
  });

  String id;
  String name;
  String shape; // round | rect | sweetheart | row
  int seats;
  double x;
  double y;
  double rotation; // degrees, long tables and rows
  // v9: ceremony rows - which side of the aisle ('partner1' | 'partner2' |
  // '' for a single block), the row number from the front, and who the row
  // is held for ('Parents & immediate family', '' = open).
  String side;
  int row;
  String reserved;

  static const double kRing = 20; // gap from body edge to seat centre

  static String shapeLabel(String s) {
    if (s == 'rect') {
      return 'Long table';
    }
    if (s == 'sweetheart') {
      return 'Sweetheart table';
    }
    if (s == 'row') {
      return 'Row';
    }
    if (s == 'king') {
      return 'King\u2019s table';
    }
    if (s == 'serp') {
      return 'Serpentine';
    }
    return 'Round table';
  }

  Size get size {
    if (shape == 'sweetheart') {
      return const Size(130, 64);
    }
    if (shape == 'row') {
      return Size((seats * 40 + 24).clamp(120, 1300).toDouble(), 30);
    }
    if (shape == 'king') {
      // two head seats, the rest split down the long sides
      final int perSide = ((seats - 2) / 2).ceil().clamp(1, 30);
      return Size((perSide * 44 + 60).clamp(160, 1400).toDouble(), 84);
    }
    if (shape == 'serp') {
      final int perSide = (seats / 2).ceil().clamp(2, 30);
      return Size((perSide * 46 + 40).clamp(200, 1400).toDouble(), 150);
    }
    if (shape == 'rect') {
      final int perSide = (seats / 2).ceil();
      return Size((perSide * 44 + 32).clamp(120, 600).toDouble(), 84);
    }
    final double d = (74 + seats * 9).clamp(96, 250).toDouble();
    return Size(d, d);
  }

  // Seat centre relative to the body's top-left corner.
  Offset seatCenter(int i) {
    final Size s = size;
    if (shape == 'sweetheart' || shape == 'row') {
      final double step = s.width / (seats + 1);
      return Offset(step * (i + 1), s.height + kRing);
    }
    if (shape == 'king') {
      if (i == 0) return Offset(-kRing, s.height / 2); // the head, left
      if (i == 1) return Offset(s.width + kRing, s.height / 2); // the head, right
      final int rest = seats - 2;
      final int top = rest - (rest ~/ 2);
      final int j = i - 2;
      if (j < top) {
        final double stepT = s.width / (top + 1);
        return Offset(stepT * (j + 1), -kRing);
      }
      final int k = j - top;
      final int bottom = rest - top;
      final double stepB = s.width / ((bottom == 0 ? 1 : bottom) + 1);
      return Offset(stepB * (k + 1), s.height + kRing);
    }
    if (shape == 'serp') {
      // The band's centre line is an S over the width; seats sit on both edges.
      final int top = seats - (seats ~/ 2);
      final int bottom = seats - top;
      final bool onTop = i < top;
      final int n = onTop ? top : bottom;
      final int j = onTop ? i : i - top;
      final double x = s.width * (j + 1) / (n + 1);
      final double cy = serpY(x, s);
      return Offset(x, onTop ? cy - kSerpHalf - kRing : cy + kSerpHalf + kRing);
    }
    if (shape == 'rect') {
      final int top = seats - (seats ~/ 2); // top row gets the extra seat
      if (i < top) {
        final double stepT = s.width / (top + 1);
        return Offset(stepT * (i + 1), -kRing);
      }
      final int j = i - top;
      final int bottom = seats - top;
      final double stepB = s.width / ((bottom == 0 ? 1 : bottom) + 1);
      return Offset(stepB * (j + 1), s.height + kRing);
    }
    final double r = s.width / 2 + kRing;
    final double a = -math.pi / 2 + (2 * math.pi * i / seats);
    return Offset(
        s.width / 2 + math.cos(a) * r, s.height / 2 + math.sin(a) * r);
  }

  // v10: the serpentine band - half its thickness and its centre line.
  static const double kSerpHalf = 30;
  static double serpY(double x, Size s) =>
      s.height / 2 + (s.height / 2 - kSerpHalf - 6) * math.sin(2 * math.pi * x / s.width);

  // v7: which way a chair's back faces - away from the table.
  double seatAngle(int i) {
    if (shape == 'sweetheart') return math.pi; // chairs below, facing the table
    if (shape == 'row') return 0; // a ceremony row faces forward
    if (shape == 'king') {
      if (i == 0) return -math.pi / 2;
      if (i == 1) return math.pi / 2;
      final int rest = seats - 2;
      final int top = rest - (rest ~/ 2);
      return (i - 2) < top ? 0 : math.pi;
    }
    if (shape == 'serp') {
      final int top = seats - (seats ~/ 2);
      return i < top ? 0 : math.pi;
    }
    if (shape == 'rect') {
      final int top = seats - (seats ~/ 2);
      return i < top ? 0 : math.pi;
    }
    return (-math.pi / 2 + (2 * math.pi * i / seats)) + math.pi / 2;
  }

  Map<String, dynamic> toMap() => <String, dynamic>{
        'id': id,
        'name': name,
        'shape': shape,
        'seats': seats,
        'x': x,
        'y': y,
        'rotation': rotation,
        'side': side,
        'row': row,
        'reserved': reserved,
      };

  static _SeatTable fromMap(Map<String, dynamic> m) {
    final int seats = m['seats'] is num ? (m['seats'] as num).round() : 8;
    String shape = (m['shape'] ?? 'round').toString();
    if (shape == 'rectangle' || shape == 'long') {
      shape = 'rect';
    }
    return _SeatTable(
      id: (m['id'] ?? '').toString(),
      name: (m['name'] ?? 'Table').toString(),
      shape: shape,
      seats: seats.clamp(1, shape == 'row' || shape == 'king' || shape == 'serp' ? 30 : 20).toInt(),
      x: m['x'] is num ? (m['x'] as num).toDouble() : 100,
      y: m['y'] is num ? (m['y'] as num).toDouble() : 100,
      rotation: m['rotation'] is num ? (m['rotation'] as num).toDouble() : 0,
      side: (m['side'] ?? '').toString(),
      row: m['row'] is num ? (m['row'] as num).toInt() : 0,
      reserved: (m['reserved'] ?? '').toString(),
    );
  }
}

// One guest doc gives one _Guest plus a virtual one per plus-one. The
// virtual ones carry k > 0 and read their seat from plusAssign[k].
class _Guest {
  _Guest({
    required this.id,
    required this.parentId,
    required this.k,
    required this.name,
    required this.party,
    required this.group,
    required this.side,
    required this.rsvp,
    required this.table,
    required this.seat,
    required this.note,
    required this.plusOnes,
    required this.updatedAt,
    this.plusAssign = const <String, dynamic>{},
  });

  final String id;
  final String parentId;
  final int k;
  final String name;
  final String party;
  final String group;
  final String side;
  final String rsvp;
  final String table;
  final int seat;
  final String note;
  final int plusOnes;
  final Timestamp? updatedAt;
  final Map<String, dynamic> plusAssign;

  bool get isPlus => k > 0;

  // A household is a party. Without one, a guest and their plus-ones are
  // their own party.
  String get partyKey =>
      party.isNotEmpty ? 'p:${party.toLowerCase().trim()}' : 'g:$parentId';

  String get partyLabel =>
      party.isNotEmpty ? party : (group.isNotEmpty ? group : name);

  _Guest plusOne(int n) {
    final Map<String, dynamic> a = plusAssign['$n'] is Map
        ? Map<String, dynamic>.from(plusAssign['$n'] as Map)
        : <String, dynamic>{};
    return _Guest(
      id: '$id#$n',
      parentId: id,
      k: n,
      name: '$name\u2019s +$n',
      party: party,
      group: group,
      side: side,
      rsvp: rsvp,
      table: (a['table'] ?? '').toString(),
      seat: a['seat'] is num ? (a['seat'] as num).toInt() : -1,
      note: '',
      plusOnes: 0,
      updatedAt: null,
    );
  }

  static _Guest fromDoc(String id, Map<String, dynamic> d, String fTable,
      String fTableName, String fSeat, String fPlus) {
    String name = (d['name'] ?? '').toString().trim();
    if (name.isEmpty) {
      name =
          '${(d['firstName'] ?? d['first'] ?? '').toString()} ${(d['lastName'] ?? d['last'] ?? '').toString()}'
              .trim();
    }
    if (name.isEmpty) {
      name = 'Guest';
    }
    final String party = (d['party'] ?? d['household'] ?? '').toString().trim();
    final String group = (d['group'] ?? '').toString().trim();
    final String rsvp = (d['rsvp'] ?? d['rsvpStatus'] ?? d['status'] ?? '')
        .toString()
        .trim()
        .toLowerCase();
    final dynamic seatRaw = d[fSeat];
    final String note = <String>[
      (d['meal'] ?? '').toString().trim(),
      (d['dietary'] ?? d['allergies'] ?? '').toString().trim(),
    ].where((String s) => s.isNotEmpty).join(', ');
    final dynamic up = d['updatedAt'] ?? d['rsvpAt'] ?? d['createdAt'];
    final dynamic tRaw = d[fTable];
    return _Guest(
      id: id,
      parentId: id,
      k: 0,
      name: name,
      party: party,
      group: group,
      side: (d['side'] ?? '').toString(),
      rsvp: rsvp == 'accepted' || rsvp == 'attending'
          ? 'yes'
          : (rsvp == 'declined' ? 'no' : rsvp),
      table: tRaw == null
          ? ''
          : (tRaw is num ? 'Table ${tRaw.round()}' : tRaw.toString().trim()),
      seat: seatRaw is num ? seatRaw.round() : -1,
      note: note,
      plusOnes: d['plusOnes'] is num
          ? (d['plusOnes'] as num).toInt().clamp(0, 10)
          : 0,
      updatedAt: up is Timestamp ? up : null,
      plusAssign: d[fPlus] is Map
          ? Map<String, dynamic>.from(d[fPlus] as Map)
          : const <String, dynamic>{},
    );
  }
}

class _GridPainter extends CustomPainter {
  _GridPainter(this.color);
  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final Paint fine = Paint()
      ..color = color.withOpacity(.45)
      ..strokeWidth = 1;
    final Paint bold = Paint()
      ..color = color
      ..strokeWidth = 1;
    for (double x = 0; x <= size.width; x += 50) {
      canvas.drawLine(
          Offset(x, 0), Offset(x, size.height), (x % 250 == 0) ? bold : fine);
    }
    for (double y = 0; y <= size.height; y += 50) {
      canvas.drawLine(
          Offset(0, y), Offset(size.width, y), (y % 250 == 0) ? bold : fine);
    }
    final Paint edge = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2;
    canvas.drawRect(Rect.fromLTWH(0, 0, size.width, size.height), edge);
  }

  @override
  bool shouldRepaint(_GridPainter old) => old.color != color;
}
