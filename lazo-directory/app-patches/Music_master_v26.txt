class _Song {
  _Song({
    required this.id,
    required this.title,
    required this.artist,
    this.album = '',
    this.art = '',
    this.preview = '',
    this.ms = 0,
    this.link = '',
    this.note = '',
    this.explicit = false,
    this.time = '',
    this.eventLabel = '',
    this.requestedBy = '',
  });
  final String id;
  final String title;
  final String artist;
  final String album;
  final String art;
  final String preview;
  final int ms;
  final String link;
  final bool explicit;
  String note;
  String time; // moments only: HH:MM pinned from the Day-Of timeline
  String eventLabel;
  String requestedBy;

  Map<String, dynamic> toMap() => <String, dynamic>{
        'id': id,
        'title': title,
        'artist': artist,
        'album': album,
        'art': art,
        'preview': preview,
        'ms': ms,
        'link': link,
        'note': note,
        'explicit': explicit,
        if (time.isNotEmpty) 'time': time,
        if (eventLabel.isNotEmpty) 'eventLabel': eventLabel,
        if (requestedBy.isNotEmpty) 'requestedBy': requestedBy,
      };

  static _Song? fromMap(dynamic m) {
    if (m is! Map) return null;
    final String t = (m['title'] ?? '').toString();
    if (t.isEmpty) return null;
    return _Song(
      id: (m['id'] ?? '').toString(),
      title: t,
      artist: (m['artist'] ?? '').toString(),
      album: (m['album'] ?? '').toString(),
      art: (m['art'] ?? '').toString(),
      preview: (m['preview'] ?? '').toString(),
      ms: m['ms'] is num ? (m['ms'] as num).toInt() : 0,
      link: (m['link'] ?? '').toString(),
      note: (m['note'] ?? '').toString(),
      explicit: m['explicit'] == true,
      time: (m['time'] ?? '').toString(),
      eventLabel: (m['eventLabel'] ?? '').toString(),
      requestedBy: (m['requestedBy'] ?? '').toString(),
    );
  }

  _Song copy() => _Song.fromMap(toMap())!;

  String get spotify =>
      'https://open.spotify.com/search/${Uri.encodeComponent('$title $artist')}';
  String get length {
    if (ms <= 0) return '';
    final int s = ms ~/ 1000;
    return '${s ~/ 60}:${(s % 60).toString().padLeft(2, '0')}';
  }
}

  void _load(Map<String, dynamic> d) {

    if (d['custom'] is List) {

        if (c is Map && (c['key'] ?? '').toString().isNotEmpty) {
          _custom.add(
              <String>[c['key'].toString(), (c['label'] ?? '').toString()]);

      }
    }
    _must
      ..clear()
      ..addAll(_songList(d['mustPlay']));
    _dont
      ..clear()
      ..addAll(_songList(d['doNotPlay']));
    _avoid
      ..clear()

  void _onSearchChanged(String q) {

  Future<List<_Song>> _fetchSongs(String q) async {
    final String enc = Uri.encodeComponent(q);
    final List<String> urls = <String>[
      '$kSearchProxy?q=$enc',
      'https://itunes.apple.com/search?term=$enc&entity=song&limit=25&country=US',
    ];
    String body = '';
    for (final String u in urls) {
      try {
        final httpc.Response r =
            await httpc.get(Uri.parse(u)).timeout(const Duration(seconds: 8));
        if (r.statusCode == 200 && r.body.contains('trackName')) {
          body = r.body;
          break;
        }
      } catch (e) {
        debugPrint('LAZO music search ($u): ' + e.toString());
      }
    }
    if (body.isEmpty) throw Exception('unavailable');
    final Map<String, dynamic> j = json.decode(body) as Map<String, dynamic>;
    final List<dynamic> rs =
        j['results'] is List ? j['results'] as List : <dynamic>[];
    final List<_Song> out = <_Song>[];
    for (final dynamic r in rs) {
      if (r is! Map) continue;
      final String t = (r['trackName'] ?? '').toString();
      if (t.isEmpty) continue;
      out.add(_Song(
        id: (r['trackId'] ?? t.hashCode).toString(),
        title: t,
        artist: (r['artistName'] ?? '').toString(),
        album: (r['collectionName'] ?? '').toString(),
        art: (r['artworkUrl100'] ?? '')
            .toString()
            .replaceAll('100x100bb', '200x200bb'),
        preview: (r['previewUrl'] ?? '').toString(),
        ms: r['trackTimeMillis'] is num
            ? (r['trackTimeMillis'] as num).toInt()
            : 0,
        link: (r['trackViewUrl'] ?? '').toString(),
        explicit: r['explicit'] == true ||
            (r['trackExplicitness'] ?? '') == 'explicit',
      ));
    }
    return out;
  }

  Future<void> _search(String q) async {
    if (q.trim().length < 2) return;
    final int token = ++_searchToken;
    setState(() {
      _searching = true;
      _searchErr = '';
      _tab = 1;
    });
    try {
      final List<_Song> out = await _fetchSongs(q.trim());
      if (token != _searchToken || !mounted) return;
      setState(() {
        _results = out;
        _searching = false;
        _searchErr = out.isEmpty
            ? 'Nothing for \u201c$q\u201d - try the artist and title.'
            : '';
      });
    } catch (e) {
      if (token != _searchToken || !mounted) return;
      setState(() {
        _searching = false;
        _searchErr = 'Search is unavailable right now - try again in a moment.';
      });
    }
  }

  // Best single match for "title artist" - used by requests and June.
  Future<_Song?> _lookup(String title, String artist) async {
    try {
      final List<_Song> r = await _fetchSongs('$title $artist'.trim());
      if (r.isEmpty) return null;
      final String t = title.toLowerCase();
      final String a = artist.toLowerCase();
      for (final _Song s in r) {
        if (s.title.toLowerCase().contains(t) &&
            (a.isEmpty || s.artist.toLowerCase().contains(a))) {
          return s;
        }
      }
      return r.first;
    } catch (_) {
      return null;
    }
  }

  // ---------------- playback + equalizer ----------------

  bool get _isPlaying => _playingId.isNotEmpty && _playing;

  AudioPlayer _ensurePlayer() {
    if (_player != null) return _player!;
    final AudioPlayer p = AudioPlayer();
    // v2.6: an explicit session so iOS treats this as media, not a UI sound
    try {
      // defaults everywhere else: the constructors differ between
      // audioplayers 5 and 6 and the category is the only thing that matters
      p.setAudioContext(AudioContext(
        iOS: AudioContextIOS(category: AVAudioSessionCategory.playback),
        android: AudioContextAndroid(),
      ));
    } catch (e) {
      debugPrint('LAZO music ctx: ' + e.toString());
    }
    p.setReleaseMode(ReleaseMode.stop);
    p.setVolume(0.9);
    _posSub = p.onPositionChanged.listen((Duration d) {
      if (!mounted) return;
      _pos = d.inMilliseconds / 1000.0;
    });
    _durSub = p.onDurationChanged.listen((Duration d) {
      if (!mounted || d.inMilliseconds <= 0) return;
      _dur = d.inMilliseconds / 1000.0;
    });
    _stateSub = p.onPlayerStateChanged.listen((PlayerState st) {
      if (!mounted) return;
      setState(() {
        _playing = st == PlayerState.playing;
        if (st == PlayerState.completed) {
          _pos = 0;
        }
      });
    });
    _player = p;
    return p;
  }

  Future<void> _togglePlay(_Song s) async {
    if (s.preview.isEmpty) {
      _toast('No preview for this one - open it with Listen.');
      return;
    }
    _tick(_Tick.light); // v2.4: playback is about to change
    if (_playingId == s.id) {
      try {
        if (kIsWeb) {
          if (_el != null) {
            if (_elPaused(_el!)) {
              _el!.play().catchError((Object _) {});
              _resumeCtx();
            } else {
              _el!.pause();
            }
          }
        } else {
          final AudioPlayer p = _ensurePlayer();
          if (_playing) {
            await p.pause();
          } else {
            await p.resume();
          }
        }
      } catch (e) {
        debugPrint('LAZO music toggle: ' + e.toString());
      }
      if (mounted) setState(() {});
      return;
    }
    _stop();
    _playingId = s.id;
    _pos = 0;
    _dur = 30;
    _posTimer?.cancel();
    _posTimer =
        Timer.periodic(const Duration(milliseconds: 40), (_) => _onTick());
    if (mounted) setState(() {});
    if (kIsWeb) {
      _startWeb(s, viaProxy: true);
      return;
    }
    // v2.6: direct first, then the relay; say why when both fail
    final AudioPlayer p = _ensurePlayer();
    String why = '';
    for (final String src in <String>[
      s.preview,
      '$kPreviewProxy?u=${Uri.encodeComponent(s.preview)}',
    ]) {
      try {
        await p.stop();
        await p.play(UrlSource(src));
        why = '';
        break;
      } catch (e) {
        why = e.toString();
        debugPrint('LAZO music play (' + src + '): ' + why);
      }
    }
    if (why.isNotEmpty) {
      final String short =
          why.replaceAll(RegExp(r'\s+'), ' ').trim();
      _toast('That preview would not play. ' +
          (short.length > 90 ? short.substring(0, 90) + '\u2026' : short));
      _stop();
      if (mounted) setState(() => _playingId = '');
    }
  }

  // Web: an <audio> element we own, so an AnalyserNode can read it. The
  // relay adds CORS to the clip; without CORS the analyser reads silence.
  // v2.3: package:universal_html's AudioElement (dart:html on web, inert on
  // phones) + Web Audio through universal_html/js_util - both compile on
  // iOS/Android; package:web / dart:js_interop do not.
  void _startWeb(_Song s, {required bool viaProxy}) {
    try {
      final html.AudioElement a = html.AudioElement();
      a.preload = 'auto';
      a.volume = 0.9;
      if (viaProxy) {
        a.crossOrigin = 'anonymous';
        a.src = '$kPreviewProxy?u=${Uri.encodeComponent(s.preview)}';
      } else {
        a.src = s.preview;
      }
      a.onEnded.listen((html.Event e) {
        if (!mounted || _el != a) return;
        _elSet(a, 'currentTime', 0);
        setState(() {
          _pos = 0;
          _playing = false;
        });
      });
      a.onError.listen((html.Event e) {
        if (!mounted || _el != a) return;
        if (viaProxy) {
          // Relay down or host not allowed: play it straight, bars on the beat.
          _vizReal = false;
          _an = null;
          _startWeb(s, viaProxy: false);
          return;
        }
        _toast('That preview would not play.');
        _stop();
        setState(() => _playingId = '');
      });
      a.onPlay.listen((html.Event e) {
        if (mounted && _el == a) setState(() => _playing = true);
      });
      a.onPause.listen((html.Event e) {
        if (mounted && _el == a) setState(() => _playing = false);
      });
      _el = a;
      _vizReal = false;
      _an = null;
      if (viaProxy) _wireAnalyser(a);
      a.play().catchError((Object _) {});
      _resumeCtx();
    } catch (e) {
      debugPrint('LAZO music web play: ' + e.toString());
      _toast('Could not play the preview.');
    }
  }

  // v2.5: read through `dynamic`. On the web the element is dart:html's, so
  // currentTime / duration / paused resolve to the real getters; universal_
  // html's phone stub lacks them, but a dynamic call still compiles there and
  // these only ever run behind kIsWeb.
  static double _elNum(html.AudioElement el, String prop) {
    try {
      final dynamic d = el;
      final Object? v = prop == 'duration' ? d.duration : d.currentTime;
      return (v is num && v.isFinite) ? v.toDouble() : 0;
    } catch (_) {
      return 0;
    }
  }

  static bool _elPaused(html.AudioElement el) {
    try {
      final dynamic d = el;
      return d.paused == true;
    } catch (_) {
      return true;
    }
  }

  static void _elSet(html.AudioElement el, String prop, Object? v) {
    try {
      if (prop == 'currentTime' && v is num) {
        final dynamic d = el;
        d.currentTime = v;
      }
    } catch (_) {}
  }

  void _resumeCtx() {}

  // v2.5: the real spectrum needed dart:js_util; the bars follow the beat
  void _wireAnalyser(html.AudioElement a) {
    _vizReal = false;
    _an = null;
    _silentTicks = 0;
  }

  void _onTick() {
    if (!mounted || _playingId.isEmpty) return;
    bool playing = _playing;
    if (kIsWeb && _el != null) {
      try {
        _pos = _elNum(_el!, 'currentTime');
        final double d = _elNum(_el!, 'duration');
        if (d.isFinite && d > 0) _dur = d;
        playing = !_elPaused(_el!);
        _playing = playing;
      } catch (_) {}
    }
    if (!_vizReal) {
      // Beat-driven fallback: a 120 bpm pulse with a slow drift per bar.
      final double t = _pos;
      final double pulse = playing
          ? 0.55 +
              0.45 *
                  math.pow(math.max(0, math.sin(t * math.pi * 2)), 3).toDouble()
          : 0.0;
      for (int i = 0; i < _bars.length; i++) {
        final double envelope = 0.35 +
            0.65 *
                math.exp(-math.pow(
                    (i - _bars.length * 0.42) / (_bars.length * 0.36), 2));
        final double wobble =
            0.6 + 0.4 * math.sin(t * (2.3 + i * 0.37) + i * 1.7);
        final double target = playing
            ? (0.08 + 0.85 * envelope * wobble * (0.55 + 0.45 * pulse))
            : 0.05;
        _bars[i] = _bars[i] * 0.7 + target * 0.3;
      }
    }
    setState(() {});
  }

  void _stop() {
    _posTimer?.cancel();
    _posTimer = null;
    try {
      _player?.stop();
    } catch (_) {}
    if (kIsWeb) {
      try {
        _el?.pause();
        _el?.src = '';
      } catch (_) {}
      _el = null;
      _an = null;
      _vizReal = false;
    }
    _playing = false;
    _playingId = '';
    _pos = 0;
    for (int i = 0; i < _bars.length; i++) {
      _bars[i] = 0.05;
    }
  }

  String _fmtClock(String hhmm) {
    final List<String> p = hhmm.split(':');
    final int h = int.tryParse(p[0]) ?? 0;
    final int m = p.length > 1 ? int.tryParse(p[1]) ?? 0 : 0;
    final int h12 = h % 12 == 0 ? 12 : h % 12;
    return '$h12:${m.toString().padLeft(2, '0')} ${h < 12 ? 'AM' : 'PM'}';
  }

  // Reception length from the timeline: first event that reads like the
  // reception to the last event. Falls back to four hours.
  int get _receptionMinutes {
    int toMin(String t) {
      final List<String> p = t.split(':');
      return (int.tryParse(p[0]) ?? 0) * 60 +
          (p.length > 1 ? int.tryParse(p[1]) ?? 0 : 0);
    }

    if (_events.length < 2) return 240;
    int start = -1;
    for (final Map<String, dynamic> ev in _events) {
      final String l = (ev['label'] ?? '').toString().toLowerCase();
      if (l.contains('reception') ||
          l.contains('cocktail') ||
          l.contains('dinner') ||
          l.contains('entrance')) {
        start = toMin((ev['time'] ?? '').toString());
        break;
      }
    }
    if (start < 0) start = toMin((_events.first['time'] ?? '').toString());
    final int end = toMin((_events.last['time'] ?? '').toString());
    final int mins = end - start;
    return mins >= 60 ? mins : 240;
  }

  int get _mustMinutes {
    int ms = 0;
    for (final _Song s in _must) {
      ms += s.ms > 0 ? s.ms : 210000;
    }
    return (ms / 60000).round();
  }

  Future<void> _approve(QueryDocumentSnapshot<Map<String, dynamic>> d) async {
    final Map<String, dynamic> r = d.data();
    final String title = (r['song'] ?? '').toString();
    final String artist = (r['artist'] ?? '').toString();
    _toast('Finding $title\u2026');
    final _Song? found = await _lookup(title, artist);
    if (!mounted) return;
    if (found == null) {
      // No catalog match: file it as typed so the DJ still sees it.
      await _approveWith(
          d.id, _Song(id: 'req:${d.id}', title: title, artist: artist));
      return;
    }
    await _approveWith(d.id, found);
  }

  Future<void> _approveWith(String reqId, _Song s) async {
    String who = '';
    for (final QueryDocumentSnapshot<Map<String, dynamic>> d in _requests) {
      if (d.id == reqId) who = (d.data()['guest'] ?? '').toString();
    }
    final _Song n = s.copy()..requestedBy = who;
    if (n.note.isEmpty && who.isNotEmpty) n.note = 'Requested by $who';
    _addMust(n, quiet: true);
    try {
      await _reqRef.doc(reqId).set(<String, dynamic>{
        'status': 'approved',
        'songId': s.id,
        'decidedAt': FieldValue.serverTimestamp()
      }, SetOptions(merge: true));
    } catch (e) {
      _toast('Could not mark it: ' + _errText(e));
    }
    _toast(
        '${n.title} \u2192 must play${who.isEmpty ? '' : ' (from $who)'} \u2713');
  }

      _toast('Could not mark it: ' + _errText(e));
    }
  }

  TextStyle _serif(
      {double size = 24,
      FontWeight weight = FontWeight.w600,
      Color color = plum,
      double height = 1.05}) {
    // v2.4: display sizes tighten as they grow; body stays at 0.
    final double track = size >= 26 ? -0.015 * size : 0;
    try {
      return GoogleFonts.cormorantGaramond(
          fontSize: size,
          fontWeight: weight,
          color: color,
          height: height,
          letterSpacing: track,
          fontFeatures: const <FontFeature>[FontFeature.liningFigures()]);
    } catch (_) {
      return TextStyle(
          fontFamily: 'serif',
          fontSize: size,
          fontWeight: weight,
          color: color,
          height: height,
          letterSpacing: track);
    }
  }

  // v2.4: on phones the Search tab slides in over the plan and leaves the
  // way it came; the plan - the thing underneath - only fades.
  Widget _viewSwitch(Widget child) {
    final bool reduced = _reducedMotion(context);
    return AnimatedSwitcher(
      duration: Duration(milliseconds: reduced ? 140 : 260),
      reverseDuration: Duration(milliseconds: reduced ? 140 : 200),
      switchInCurve: Curves.easeOutCubic,
      switchOutCurve: Curves.easeInCubic,
      layoutBuilder: (Widget? current, List<Widget> previous) => Stack(
        fit: StackFit.expand,
        children: <Widget>[...previous, if (current != null) current],
      ),
      transitionBuilder: (Widget c, Animation<double> a) {
        final Key? k = c.key;
        final bool root = k is ValueKey<String> && k.value.endsWith(':root');
        final Widget faded = FadeTransition(opacity: a, child: c);
        if (reduced || root) return faded;
        return SlideTransition(
          position: Tween<Offset>(begin: const Offset(.04, 0), end: Offset.zero)
              .animate(a),
          child: faded,
        );
      },
      child: child,
    );
  }

  // The play/pause glyph cross-fades and scales instead of snapping.

        duration: Duration(milliseconds: _reducedMotion(context) ? 0 : 160),
        switchInCurve: Curves.easeOutCubic,
        switchOutCurve: Curves.easeIn,
        transitionBuilder: (Widget w, Animation<double> a) => ScaleTransition(
            scale: Tween<double>(begin: .7, end: 1).animate(a),
            child: FadeTransition(opacity: a, child: w)),
        child: child,
      );

  String _clock(double s) =>
      '${(s ~/ 60)}:${(s.round() % 60).toString().padLeft(2, '0')}';

  void _seekFrac(double frac) {
    if (_playingId.isEmpty || _dur <= 0) return;
    final double f = frac.clamp(0.0, 1.0).toDouble();
    try {
      if (kIsWeb) {
        if (_el != null) _elSet(_el!, 'currentTime', f * _dur);
      } else {
        _player?.seek(Duration(milliseconds: (f * _dur * 1000).round()));
      }
      _pos = f * _dur;
      setState(() {});
    } catch (_) {}
  }

  Widget _momentGrid(int cols) {
    final List<Widget> tiles =
        _allMoments.map((List<String> m) => _momentTile(m)).toList();
    final List<Widget> rows = <Widget>[];
    for (int i = 0; i < tiles.length; i += cols) {
      final List<Widget> r = <Widget>[];
      for (int j = 0; j < cols; j++) {
        r.add(Expanded(
            child:
                i + j < tiles.length ? tiles[i + j] : const SizedBox.shrink()));
        if (j < cols - 1) r.add(const SizedBox(width: 10));
      }
      rows.add(Padding(
          padding: const EdgeInsets.only(bottom: 10),
          child:
              Row(crossAxisAlignment: CrossAxisAlignment.start, children: r)));
    }
    return Column(children: rows);
  }

    return Container(

      clipBehavior: Clip.antiAlias,
      child: must
          ? ReorderableListView.builder(
              shrinkWrap: true,
              physics: const NeverScrollableScrollPhysics(),
              buildDefaultDragHandles: false,
              itemCount: list.length,
              // v2.4: a picked-up row lifts with a shadow; a tick on pickup
              // and another when it lands.
              onReorderStart: (int _) => _tick(_Tick.medium),
              proxyDecorator: (Widget child, int i, Animation<double> a) =>
                  AnimatedBuilder(
                animation: a,
                builder: (BuildContext c, Widget? w) {
                  final double t = Curves.easeOut.transform(a.value);
                  return Transform.scale(
                    scale: 1 + .02 * t,
                    child: Material(
                      color: card,
                      elevation: 10 * t,
                      shadowColor: plumDeep.withOpacity(.4),
                      borderRadius: BorderRadius.circular(12),
                      child: w,
                    ),
                  );
                },
                child: child,
              ),
              onReorder: (int a, int b) {
                _tick(_Tick.select);
                setState(() {
                  if (b > a) b--;
                  final _Song s = list.removeAt(a);
                  list.insert(b, s);
                });
                _scheduleSave();
              },
              itemBuilder: (BuildContext c, int i) => _listRow(list[i], i, must,
                  key: ValueKey<String>('m:${list[i].id}')),
            )
          : Column(children: <Widget>[
              for (int i = 0; i < list.length; i++) _listRow(list[i], i, must)
            ]),
    );
  }

  String _summaryText() {
    final StringBuffer b = StringBuffer();
    b.writeln(
        'MUSIC${_coupleLabel.isEmpty ? '' : ' - $_coupleLabel'}${_weddingDate.isEmpty ? '' : ' - $_weddingDate'}');
    b.writeln('');
    if (_moments.isNotEmpty) {
      b.writeln('THE MOMENTS');
      for (final List<String> m in _allMoments) {
        final _Song? s = _moments[m[0]];
        if (s != null)
          b.writeln(
              '${s.time.isEmpty ? '' : '${_fmtClock(s.time)}  '}${m[1]}: ${s.title} - ${s.artist}');
      }
      b.writeln('');
    }
    if (_must.isNotEmpty) {
      b.writeln('MUST PLAY (${_hm(_mustMinutes)})');
      for (int i = 0; i < _must.length; i++) {
        final _Song s = _must[i];
        b.writeln(
            '${i + 1}. ${s.title} - ${s.artist}${s.note.isEmpty ? '' : ' (${s.note})'}');
      }
      b.writeln('');
    }
    if (_dont.isNotEmpty || _avoid.isNotEmpty) {
      b.writeln('DO NOT PLAY');
      for (final _Song s in _dont) {
        b.writeln('${s.title} - ${s.artist}');
      }
      if (_avoid.isNotEmpty) b.writeln('Also avoid: ${_avoid.join(', ')}');
      b.writeln('');
    }
    if (_vibe.isNotEmpty) {
      b.writeln('THE VIBE');
      b.writeln(_vibe.join(' \u00b7 '));
      b.writeln('');
    }
    if (_notes.trim().isNotEmpty) {
      b.writeln('NOTES');
      b.writeln(_notes.trim());
      b.writeln('');
    }
    b.writeln('Planned on Lazo \u00b7 meetlazo.com');
    return b.toString();
  }

  String _songHtml(_Song s, {String prefix = ''}) {
    final String links = <String>[
      if (s.link.isNotEmpty) '<a href="${_esc(s.link)}">Listen</a>',
      '<a href="${_esc(s.spotify)}">Spotify</a>',
    ].join(' &middot; ');
    return '<tr><td style="width:44px">${s.art.isEmpty ? '' : '<img src="${_esc(s.art)}" width="40" height="40" style="border-radius:8px">'}</td>'
        '<td>${prefix.isEmpty ? '' : '<span style="color:#6B5F72;font-size:11px;letter-spacing:1px;text-transform:uppercase">${_esc(prefix)}</span><br>'}'
        '<b>${_esc(s.title)}</b>${s.explicit ? ' <span style="background:#6B5F72;color:#fff;font-size:9px;padding:1px 4px;border-radius:3px">E</span>' : ''}<br><span style="color:#6B5F72">${_esc(s.artist)}${s.length.isEmpty ? '' : ' &middot; ${s.length}'}</span>'
        '${s.note.isEmpty ? '' : '<br><i style="color:#8A6A2F">${_esc(s.note)}</i>'}</td>'
        '<td style="white-space:nowrap;font-size:12px">$links</td></tr>';
  }

  String _printHtml() {
    final StringBuffer h = StringBuffer();
    if (_moments.isNotEmpty) {
      h.write('<h2>The moments</h2><table>');
      for (final List<String> m in _allMoments) {
        final _Song? s = _moments[m[0]];
        if (s != null)
          h.write(_songHtml(s,
              prefix:
                  '${s.time.isEmpty ? '' : '${_fmtClock(s.time)} \u00b7 '}${m[1]}'));
      }
      h.write('</table>');
    }
    if (_must.isNotEmpty) {
      h.write(
          '<h2>Must play (${_must.length} \u00b7 ${_hm(_mustMinutes)})</h2><table>');
      for (final _Song s in _must) {

  Future<bool> _sendToThread(
      String inquiryId, String vendorName, String note) async {
    final DocumentReference<Map<String, dynamic>> ref =
        FirebaseFirestore.instance.collection('inquiries').doc(inquiryId);
    Map<String, dynamic>? att;
    try {
      final Uint8List bytes = Uint8List.fromList(utf8.encode(_printHtml()));
      final String path =
          'inquiries/$inquiryId/attachments/${DateTime.now().millisecondsSinceEpoch}_music.html';
      final Reference r = FirebaseStorage.instance.ref(path);
      await r.putData(bytes, SettableMetadata(contentType: 'text/html'));
      final String url = await r.getDownloadURL();
      att = <String, dynamic>{
        'url': url,
        'name': 'Music.html',
        'type': 'text/html',
        'size': bytes.length,
        'path': path
      };
    } catch (e) {
      debugPrint('LAZO music attach: ' + e.toString());
      att = null;
    }
    String text =
        '${note.isEmpty ? 'Here\u2019s our music plan.' : note}\n\n${_summaryText()}';
    if (text.length >= 2900)
      text = text.substring(0, 2860) + '\u2026\n\n(The full list is attached.)';
    try {
      final Map<String, dynamic> msg = <String, dynamic>{
        'senderRole': 'couple',
        'text': text,
        'at': FieldValue.serverTimestamp()
      };
      if (att != null) msg['attachment'] = att;
      try {
        await ref.collection('messages').add(msg);
      } catch (e) {
        if (att == null) rethrow;
        msg.remove('attachment');
        await ref.collection('messages').add(msg);
      }
    } catch (e) {
      _toast('Could not send: ' + _errText(e));
      return false;
    }
    try {
      await ref.set(<String, dynamic>{
        'lastMessageAt': FieldValue.serverTimestamp(),
        'lastMessageRole': 'couple',
        'lastMessagePreview': 'Music plan from the couple'
      }, SetOptions(merge: true));
    } catch (_) {}
    return true;
  }

// Rounded equalizer bars, drawn from a 0..1 list. mirror draws them growing
// from the vertical centre, which reads better in a bar than a baseline.
// ---------------- v2.4: MOTION ----------------
// Three rules, from the WWDC "Designing Fluid Interfaces" talk:
//  1. Answer the finger on pointer-DOWN, never only on release.
//  2. Critically damped by default. Overshoot only when something arrives
//     with momentum.
//  3. Reduced-motion users get a cross-fade - never a slide, never a bounce.

bool _reducedMotion(BuildContext c) =>
    MediaQuery.maybeOf(c)?.disableAnimations ?? false;

// Haptics, in tiers, fired on the commit so the pulse and the visual change
// hit on the same frame. No-ops on web; no permission needed on phones.
//   select - a choice among peers: tabs, the scrubber, a checkbox.
//   light  - an ordinary button; a preview starting or pausing.
//   medium - a song landing in the plan; a row picked up; Send to your DJ.
enum _Tick { none, select, light, medium }

void _tick(_Tick t) {
  switch (t) {
    case _Tick.none:
      return;
    case _Tick.select:
      HapticFeedback.selectionClick();
      return;
    case _Tick.light:
      HapticFeedback.lightImpact();
      return;
    case _Tick.medium:
      HapticFeedback.mediumImpact();
      return;
  }
}

/// Wrap a button's onPressed. Null stays null, so disabled stays disabled.
VoidCallback? _h(VoidCallback? f, [_Tick t = _Tick.light]) => f == null
    ? null
    : () {
        _tick(t);
        f();
      };

/// Same for bool callbacks: a chip's onSelected, a Switch's onChanged.
ValueChanged<bool>? _hb(ValueChanged<bool>? f, [_Tick t = _Tick.select]) =>
    f == null
        ? null
        : (bool v) {
            _tick(t);
            f(v);
          };

/// A tappable that answers the instant the finger lands - 97% in 90ms on the
/// way down, a slower settle on the way up. Use it where the child paints its
/// own background; InkWell's ripple would land underneath and never show.
class _Press extends StatefulWidget {
  const _Press({
    super.key,
    required this.child,
    this.onTap,
    this.scale = .97,
    this.dim = .88,
    this.enabled = true,
    this.tick = _Tick.light,
    this.tickOnDown = false,
  });
  final Widget child;
  final VoidCallback? onTap;
  final double scale;
  final double dim;
  final bool enabled;
  final _Tick tick;
  // Harmony: the scale already fires on touch-down, so a control can put its
  // haptic there too instead of a frame after release. Off by default, so no
  // existing call site changes.
  final bool tickOnDown;
  @override
  State<_Press> createState() => _PressState();
}

class _PressState extends State<_Press> {
  bool _down = false;
  void _set(bool v) {
    if (_down != v && mounted) setState(() => _down = v);
  }

  @override
  Widget build(BuildContext context) {
    final bool on = _down && widget.enabled && !_reducedMotion(context);
    final VoidCallback? tap = widget.enabled ? widget.onTap : null;
    return MouseRegion(
      cursor: widget.enabled && widget.onTap != null
          ? SystemMouseCursors.click
          : MouseCursor.defer,
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTapDown: widget.enabled
            ? (_) {
                if (widget.tickOnDown) _tick(widget.tick);
                _set(true);
              }
            : null,
        onTapUp: (_) => _set(false),
        onTapCancel: () => _set(false),
        onTap: tap == null
            ? null
            : () {
                if (!widget.tickOnDown) _tick(widget.tick);
                tap();
              },
        child: AnimatedScale(
          scale: on ? widget.scale : 1,
          duration: Duration(milliseconds: on ? 90 : 240),
          curve: on ? Curves.easeOut : Curves.easeOutCubic,
          child: AnimatedOpacity(
            opacity: on ? widget.dim : 1,
            duration: Duration(milliseconds: on ? 60 : 200),
            child: widget.child,
          ),
        ),
      ),
    );
  }
}

class _EqPainter extends CustomPainter {
  _EqPainter(this.bars, this.color, {this.gap = 2, this.mirror = false});
  final List<double> bars;
  final Color color;
  final double gap;
  final bool mirror;

  @override
  void paint(Canvas canvas, Size size) {
    if (bars.isEmpty) return;
    final double w = (size.width - gap * (bars.length - 1)) / bars.length;
    final Paint p = Paint()..color = color;
    final Paint dim = Paint()..color = color.withOpacity(.28);
    for (int i = 0; i < bars.length; i++) {
      final double v = bars[i].clamp(0.04, 1.0);
      final double h = math.max(2.0, size.height * v);
      final double x = i * (w + gap);
      final Rect track = mirror
          ? Rect.fromLTWH(x, 0, w, size.height)
          : Rect.fromLTWH(x, 0, w, size.height);
      canvas.drawRRect(
          RRect.fromRectAndRadius(track, Radius.circular(w / 2)), dim);
      final Rect bar = mirror
          ? Rect.fromLTWH(x, (size.height - h) / 2, w, h)
          : Rect.fromLTWH(x, size.height - h, w, h);
      canvas.drawRRect(RRect.fromRectAndRadius(bar, Radius.circular(w / 2)), p);
    }
  }

  @override
  bool shouldRepaint(_EqPainter old) => true;
}
