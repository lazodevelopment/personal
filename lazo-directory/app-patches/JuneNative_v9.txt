// ---------------------------------------------------------------- the API: the June worker, signed in as this user
class _Api {
  Future<Map<String, String>> _headers() async {
    final User? u = FirebaseAuth.instance.currentUser;
    final String t = u == null ? '' : (await u.getIdToken() ?? '');
    return <String, String>{'authorization': 'Bearer $t', 'content-type': 'application/json'};
  }

  Future<Map<String, dynamic>> get(String path) async {
    final httpc.Response r = await httpc.get(Uri.parse(_kHub + path), headers: await _headers()).timeout(const Duration(seconds: 30));
    final dynamic j = jsonDecode(r.body.isEmpty ? '{}' : r.body);
    if (r.statusCode >= 400) throw Exception(_s(j is Map ? (j['message'] ?? j['error']) : r.statusCode));
    return j is Map ? Map<String, dynamic>.from(j) : <String, dynamic>{'data': j};
  }

  Future<dynamic> post(String path, Map<String, dynamic> body) async {
    final httpc.Response r = await httpc.post(Uri.parse(_kHub + path), headers: await _headers(), body: jsonEncode(body)).timeout(const Duration(seconds: 60));
    final dynamic j = jsonDecode(r.body.isEmpty ? '{}' : r.body);
    if (r.statusCode >= 400) throw Exception(_s(j is Map ? (j['message'] ?? j['error']) : r.statusCode));
    return j;
  }

  // the chat streams server-sent events: delta {text}, done {reply, actions, messages}, error {message}
  Future<void> chat(String text, List<Map<String, dynamic>> history, {required void Function(String) onDelta, required void Function(Map<String, dynamic>) onDone, required void Function(String) onError}) async {
    final httpc.Client client = httpc.Client();
    try {
      final httpc.Request req = httpc.Request('POST', Uri.parse('$_kHub/api/chat'));
      req.headers.addAll(await _headers());
      req.body = jsonEncode(<String, dynamic>{'text': text, 'messages': history});
      final httpc.StreamedResponse resp = await client.send(req).timeout(const Duration(seconds: 90));
      if (resp.statusCode >= 400) {
        onError('June is unavailable (${resp.statusCode}).');
        return;
      }
      String buf = '';
      bool done = false;
      await for (final String chunk in resp.stream.transform(utf8.decoder)) {
        buf += chunk;
        int idx;
        while ((idx = buf.indexOf('\n\n')) >= 0) {
          final String frame = buf.substring(0, idx);
          buf = buf.substring(idx + 2);
          final RegExpMatch? ev = RegExp(r'^event: (.*)$', multiLine: true).firstMatch(frame);
          final RegExpMatch? da = RegExp(r'^data: (.*)$', multiLine: true).firstMatch(frame);
          if (ev == null || da == null) continue;
          dynamic d;
          try {
            d = jsonDecode(da.group(1)!);
          } catch (_) {
            continue;
          }
          final String kind = ev.group(1)!.trim();
          if (kind == 'delta') {
            onDelta(_s(_map(d)['text']));
          } else if (kind == 'done') {
            done = true;
            onDone(_map(d));
          } else if (kind == 'error') {
            onError(_s(_map(d)['message']));
          }
        }
      }
      if (!done) onDone(<String, dynamic>{});
    } catch (e) {
      onError(e.toString());
    } finally {
      client.close();
    }
  }
}

// ---------------------------------------------------------------- the web analyser (JC-LAZO-JUNE-NATIVE-1011-003)
// On the web every <audio> June plays goes through a Web Audio analyser; the script writes
// 40 bands + an envelope to document.body.dataset.juneLv each frame and the core reads them.
// Inert on phones (nothing here runs unless kIsWeb).
const String _kJuneWebJs = r"""
(function(){
  if (window.__june) return;
  var S = { ctx:null, an:null, nodes:new WeakMap(), buf:null, edges:null, NB:40, peak:0.04 };
  function ctx(){ if(!S.ctx){ try{ S.ctx = new (window.AudioContext||window.webkitAudioContext)(); }catch(e){ return null; } } if(S.ctx.state==='suspended'){ S.ctx.resume().catch(function(){}); } return S.ctx; }
  ['pointerdown','keydown','touchstart'].forEach(function(ev){ document.addEventListener(ev, function(){ ctx(); try{ document.body.dataset.juneActive='1'; }catch(e){} }, {passive:true}); });
  try{ if(navigator.userActivation&&navigator.userActivation.hasBeenActive){ document.body.dataset.juneActive='1'; } }catch(e){}
  function tune(an){
    try{ an.fftSize=2048; an.smoothingTimeConstant=0.05; an.minDecibels=-82; an.maxDecibels=-14; }catch(e){}
    var sr=(an.context&&an.context.sampleRate)||44100, bins=an.frequencyBinCount, hz=sr/2/bins;
    S.buf=new Uint8Array(bins); S.time=new Uint8Array(an.fftSize);
    var NB=S.NB, lo=55, hi=Math.min(11000, sr/2-100), edges=new Int32Array(NB+1);
    for(var b=0;b<=NB;b++) edges[b]=Math.max(1, Math.round(lo*Math.pow(hi/lo,b/NB)/hz));
    for(var b2=1;b2<=NB;b2++) if(edges[b2]<=edges[b2-1]) edges[b2]=edges[b2-1]+1;
    S.edges=edges;
  }
  document.addEventListener('june-attach', function(e){
    var id=e.detail&&e.detail.id, el=id&&document.getElementById(id); if(!el) return;
    var c=ctx(); if(!c) return;
    try{
      var n=S.nodes.get(el);
      if(!n){ var src=c.createMediaElementSource(el); var an=c.createAnalyser(); tune(an); src.connect(an); an.connect(c.destination); n={src:src,an:an}; S.nodes.set(el,n); }
      S.an=n.an;
    }catch(err){ S.an=null; }
  });
  document.addEventListener('june-detach', function(){ S.an=null; document.body.dataset.juneLv=''; });
  var out=new Array(S.NB);
  function loop(){
    if(S.an&&S.edges){
      S.an.getByteFrequencyData(S.buf); S.an.getByteTimeDomainData(S.time);
      for(var b=0;b<S.NB;b++){ var m=0; for(var i=S.edges[b];i<S.edges[b+1];i++){ if(S.buf[i]>m) m=S.buf[i]; } var tilt=0.9+0.35*(b/S.NB); out[b]=Math.round(Math.min(1, Math.pow(m/255,1.1)*tilt*1.25)*100); }
      var sq=0,N=S.time.length; for(var j=0;j<N;j++){ var v=(S.time[j]-128)/128; sq+=v*v; }
      var rms=Math.sqrt(sq/N); S.peak=Math.max(rms, S.peak*0.995, 0.04); var env=Math.round(Math.min(1, rms/S.peak)*100);
      document.body.dataset.juneLv=out.join(',')+'|'+env;
    }
    requestAnimationFrame(loop);
  }
  requestAnimationFrame(loop);
  window.__june=S;
})();
""";
bool _juneWebReady = false;
int _juneWebSeq = 0;

void _juneWebInstall() {
  if (!kIsWeb || _juneWebReady) return;
  _juneWebReady = true;
  try {
    final html.ScriptElement sc = html.ScriptElement();
    sc.text = _kJuneWebJs;
    html.document.head?.append(sc);
  } catch (_) {}
}

/// gives the element an id, puts it in the document and asks the script to analyse it
String _juneWebAttach(html.AudioElement a) {
  final String id = 'june-audio-${++_juneWebSeq}';
  try {
    a.id = id;
    a.style.display = 'none';
    html.document.body?.append(a);
    html.document.dispatchEvent(html.CustomEvent('june-attach', detail: <String, dynamic>{'id': id}));
  } catch (_) {}
  return id;
}

void _juneWebDetach(html.AudioElement? a) {
  try {
    html.document.dispatchEvent(html.CustomEvent('june-detach'));
    a?.remove();
  } catch (_) {}
}

/// the latest frame from the script: [levels 0..1 x nb, envelope 0..1], or null
List<double>? _juneWebFrame(int nb) {
  if (!kIsWeb) return null;
  try {
    final String? raw = html.document.body?.dataset['juneLv'];
    if (raw == null || raw.isEmpty) return null;
    final int bar = raw.indexOf('|');
    final List<String> parts = (bar < 0 ? raw : raw.substring(0, bar)).split(',');
    if (parts.length < 8) return null;
    final List<double> out = List<double>.filled(nb + 1, 0);
    for (int b = 0; b < nb; b++) {
      final double x = b * (parts.length - 1) / math.max(1, nb - 1);
      final int j0 = x.floor();
      final int j1 = math.min(parts.length - 1, j0 + 1);
      final double k = x - j0;
      final double a = (int.tryParse(parts[j0]) ?? 0) / 100.0;
      final double c = (int.tryParse(parts[j1]) ?? 0) / 100.0;
      out[b] = a * (1 - k) + c * k;
    }
    out[nb] = bar < 0 ? 0 : (int.tryParse(raw.substring(bar + 1)) ?? 0) / 100.0;
    return out;
  } catch (_) {
    return null;
  }
}

// ---------------------------------------------------------------- the voice out: ElevenLabs through the hub, one clip per sentence
class _Voice {
  _Voice(this.api, this.onState);
  final _Api api;
  final void Function(bool speaking) onState;
  final List<String> _queue = <String>[];
  bool speaking = false;
  bool enabled = true;
  bool hubVoice = true; // false once the hub says tts is not configured; the device voice takes over
  int _gen = 0;
  AudioPlayer? _player;
  html.AudioElement? _el;
  FlutterTts? _tts;

  // JC-LAZO-JUNE-ENV-1010-001: the level track that came with the clip - frames x bands, bytes 0..255
  Uint8List? _trk;
  int _trkNb = 24;
  double _trkFps = 20;
  int _trkFrames = 0;
  DateTime? _t0;
  double _posKnown = 0;
  DateTime? _posAt;
  bool get hasTrack => _trk != null;

  void _setTrack(dynamic env) {
    _trk = null;
    _t0 = null;
    _posAt = null;
    _posKnown = 0;
    final Map<String, dynamic> e = _map(env);
    final String b64 = _s(e['b64']);
    if (b64.isEmpty) return;
    try {
      _trk = base64Decode(b64);
    } catch (_) {
      _trk = null;
      return;
    }
    _trkNb = _n(e['nb']) > 0 ? _n(e['nb']).toInt() : 24;
    _trkFps = _n(e['fps']) > 0 ? _n(e['fps']).toDouble() : 20;
    _trkFrames = _trk!.length ~/ _trkNb;
  }

  double _pos() {
    if (kIsWeb) {
      try {
        // read dynamically: the phone stub of the element has no currentTime (iOS archive failed on it)
        final dynamic a = _el;
        if (a != null) return ((a.currentTime as num?) ?? 0).toDouble();
      } catch (_) {}
    }
    final DateTime? at = _posAt;
    if (at != null) return _posKnown + DateTime.now().difference(at).inMicroseconds / 1e6;
    final DateTime? t0 = _t0;
    return t0 == null ? 0 : DateTime.now().difference(t0).inMicroseconds / 1e6;
  }

  /// levels 0..1 for [nb] bars at the current playback position; null without a track
  List<double>? frame(int nb) {
    final Uint8List? t = _trk;
    if (t == null || _trkFrames == 0) return null;
    final double f = _pos() * _trkFps;
    final int i0 = f.floor().clamp(0, _trkFrames - 1);
    final int i1 = (i0 + 1).clamp(0, _trkFrames - 1);
    final double k = (f - f.floor()).clamp(0.0, 1.0);
    double at(int i, int j) => t[i * _trkNb + j] / 255.0;
    final List<double> out = List<double>.filled(nb, 0);
    for (int b = 0; b < nb; b++) {
      final double x = b * (_trkNb - 1) / math.max(1, nb - 1);
      final int j0 = x.floor();
      final int j1 = math.min(_trkNb - 1, j0 + 1);
      final double kk = x - j0;
      final double a = at(i0, j0) * (1 - kk) + at(i0, j1) * kk;
      final double c = at(i1, j0) * (1 - kk) + at(i1, j1) * kk;
      out[b] = a * (1 - k) + c * k;
    }
    return out;
  }

  AudioPlayer _ensure() {
    if (_player != null) return _player!;
    final AudioPlayer p = AudioPlayer();
    try {
      p.setAudioContext(AudioContext(
        iOS: AudioContextIOS(category: AVAudioSessionCategory.playback),
        android: AudioContextAndroid(),
      ));
    } catch (_) {}
    p.setReleaseMode(ReleaseMode.stop);
    _player = p;
    return p;
  }

  FlutterTts _ensureTts() {
    if (_tts != null) return _tts!;
    final FlutterTts t = FlutterTts();
    try {
      t.setLanguage('en-GB');
      t.setSpeechRate(0.5);
      t.setPitch(1.05);
      t.awaitSpeakCompletion(true);
      t.setIosAudioCategory(IosTextToSpeechAudioCategory.playback, <IosTextToSpeechAudioCategoryOptions>[IosTextToSpeechAudioCategoryOptions.mixWithOthers, IosTextToSpeechAudioCategoryOptions.duckOthers]);
    } catch (_) {}
    _tts = t;
    return t;
  }

  // JC-LAZO-JUNE-NATIVE-1013-005: the phone speaks the sentence itself when a clip cannot be fetched or played
  Future<void> _speakDevice(String text, int gen) async {
    if (kIsWeb || gen != _gen) return;
    try {
      _t0 = DateTime.now();
      await _ensureTts().speak(text).timeout(const Duration(minutes: 2), onTimeout: () => 1);
    } catch (e) {
      debugPrint('LAZO june device voice: ' + e.toString());
    }
  }

  void say(String text) {
    final String t = text.trim();
    if (!enabled || t.isEmpty) return;
    _queue.add(t);
    if (!speaking) _pump();
  }

  Future<void> _pump() async {
    speaking = true;
    onState(true);
    final int gen = _gen;
    while (_queue.isNotEmpty && gen == _gen) {
      final String t = _queue.removeAt(0);
      String url = '';
      dynamic env;
      if (hubVoice) {
        try {
          final dynamic j = await api.post('/api/clip', <String, dynamic>{'text': t});
          url = _s(_map(j)['url']);
          env = _map(j)['env'];
        } catch (e) {
          final String m = e.toString();
          if (m.contains('not configured')) hubVoice = false;
          debugPrint('LAZO june clip: ' + m);
        }
      }
      if (gen != _gen) break;
      bool ok = false;
      if (url.isNotEmpty) ok = await playUrl(url, gen, env);
      if (!ok && gen == _gen) await _speakDevice(t, gen);
    }
    if (gen == _gen) {
      speaking = false;
      onState(false);
    }
  }

  // web: one element, played to the end; false if it errored before it ended
  Future<bool> _playWeb(String url, bool withFft) async {
    final html.AudioElement a = html.AudioElement();
    _el = a;
    if (withFft) a.crossOrigin = 'anonymous';
    a.src = url;
    a.preload = 'auto';
    if (withFft) _juneWebAttach(a);
    final Completer<bool> c = Completer<bool>();
    a.onEnded.listen((_) {
      if (!c.isCompleted) c.complete(true);
    });
    a.onError.listen((_) {
      if (!c.isCompleted) c.complete(false);
    });
    _t0 = DateTime.now();
    await a.play().catchError((Object _) {});
    final bool ok = await c.future.timeout(const Duration(minutes: 6), onTimeout: () => true);
    if (withFft) _juneWebDetach(a);
    try {
      a.remove();
    } catch (_) {}
    if (identical(_el, a)) _el = null;
    return ok;
  }

  // plays one URL to the end (web: an <audio> element; phones: audioplayers)
  Future<bool> playUrl(String url, [int? gen, dynamic env]) async {
    final int g = gen ?? _gen;
    _setTrack(env);
    bool played = true;
    try {
      if (kIsWeb) {
        // through the analyser first; if the element errors (a clip the browser
        // will not share cross-origin), once more plainly so she is never silent
        final bool ok = await _playWeb(url, true);
        if (!ok && g == _gen) played = await _playWeb(url, false);
      } else {
        final AudioPlayer p = _ensure();
        await p.stop();
        final Completer<bool> c = Completer<bool>();
        bool started = false;
        final StreamSubscription<void> sub = p.onPlayerComplete.listen((_) {
          if (!c.isCompleted) c.complete(true);
        });
        final StreamSubscription<Duration> psub = p.onPositionChanged.listen((Duration d) {
          _posKnown = d.inMicroseconds / 1e6;
          _posAt = DateTime.now();
          if (d.inMilliseconds > 0) started = true;
        });
        // JC-LAZO-JUNE-NATIVE-1013-005: a clip that never starts or never ends is abandoned, never waited on for minutes
        final double knownSecs = _trkFrames > 0 ? _trkFrames / _trkFps : 0;
        final Timer startDog = Timer(const Duration(seconds: 5), () {
          if (!started && !c.isCompleted) c.complete(false);
        });
        _t0 = DateTime.now();
        try {
          await p.play(UrlSource(url));
        } catch (e) {
          if (!c.isCompleted) c.complete(false);
          debugPrint('LAZO june play start: ' + e.toString());
        }
        played = await c.future.timeout(Duration(milliseconds: knownSecs > 0 ? ((knownSecs + 4) * 1000).round() : 90000), onTimeout: () => started);
        startDog.cancel();
        if (!played) {
          try {
            await p.stop();
          } catch (_) {}
        }
        await sub.cancel();
        await psub.cancel();
      }
    } catch (e) {
      played = false;
      debugPrint('LAZO june play: ' + e.toString());
    }
    _trk = null;
    return played;
  }

  // v009: one clip (the brief) played as speech - the floating player shows it and its Stop works
  Future<bool> playAlone(String url, dynamic env) async {
    stop();
    final int g = _gen;
    speaking = true;
    onState(true);
    final bool ok = await playUrl(url, g, env);
    if (g == _gen) {
      speaking = false;
      onState(false);
    }
    return ok;
  }

  void stop() {
    _gen++;
    _queue.clear();
    try {
      _el?.pause();
      _juneWebDetach(_el);
      _el = null;
    } catch (_) {}
    try {
      _player?.stop();
    } catch (_) {}
    try {
      _tts?.stop();
    } catch (_) {}
    _trk = null;
    if (speaking) {
      speaking = false;
      onState(false);
    }
  }

  void dispose() {
    stop();
    try {
      _player?.dispose();
    } catch (_) {}
  }
}

// ---------------------------------------------------------------- the voice in: the device's speech recognition
class _Ears {
  _Ears({required this.onText, required this.onState});
  final void Function(String text, bool isFinal) onText;
  final void Function(bool listening) onState;
  final stt.SpeechToText _stt = stt.SpeechToText();
  bool ready = false;
  bool listening = false;

  Future<bool> init() async {
    try {
      ready = await _stt.initialize(
        onStatus: (String s) {
          if (s == 'done' || s == 'notListening') {
            if (listening) {
              listening = false;
              onState(false);
            }
          }
        },
        onError: (dynamic e) {
          if (listening) {
            listening = false;
            onState(false);
          }
        },
      );
    } catch (e) {
      ready = false;
    }
    return ready;
  }

  Future<void> start() async {
    if (!ready && !(await init())) return;
    listening = true;
    onState(true);
    try {
      await _stt.listen(
        onResult: (dynamic r) {
          try {
            onText(_s(r.recognizedWords), r.finalResult == true);
          } catch (_) {}
        },
        listenFor: const Duration(seconds: 20),
        pauseFor: const Duration(seconds: 3),
        localeId: 'en_US',
      );
    } catch (e) {
      listening = false;
      onState(false);
    }
  }

  Future<void> stop() async {
    try {
      await _stt.stop();
    } catch (_) {}
    if (listening) {
      listening = false;
      onState(false);
    }
  }
}

// ---------------------------------------------------------------- the core: a circular spectrum analyser
class _CorePainter extends CustomPainter {
  _CorePainter({required this.levels, required this.caps, required this.swell, required this.bass, required this.rot, required this.tint, required this.live});
  final List<double> levels;
  final List<double> caps;
  final double swell;
  final double bass;
  final double rot;
  final Color tint;
  final bool live;

  @override
  void paint(Canvas canvas, Size size) {
    final double w = size.width;
    final Offset c = Offset(w / 2, size.height / 2);
    final int nb = levels.length;
    final double r0 = w * 0.21 * (1 + bass * 0.05);
    final double len = w * 0.25;
    // glow
    final double gr = r0 * (1.9 + swell * 0.5);
    canvas.drawCircle(c, gr, Paint()..shader = RadialGradient(colors: <Color>[tint.withOpacity(live ? 0.28 : 0.16), tint.withOpacity(0.06), tint.withOpacity(0)], stops: const <double>[0, .5, 1]).createShader(Rect.fromCircle(center: c, radius: gr)));
    // bars, bass at the bottom climbing both sides
    final double bw = math.max(1.5, (math.pi * r0) / nb * 0.62);
    final Paint bar = Paint()
      ..color = tint.withOpacity(live ? 0.92 : 0.5)
      ..strokeWidth = bw
      ..strokeCap = StrokeCap.butt;
    final Paint cap = Paint()
      ..color = Colors.white.withOpacity(live ? 0.85 : 0.35)
      ..strokeWidth = math.max(1, bw * 0.9);
    for (int s = -1; s <= 1; s += 2) {
      for (int b = 0; b < nb; b++) {
        final double a = math.pi / 2 + s * ((b + 0.5) / nb) * math.pi;
        final double l = r0 + math.max(1, levels[b] * len);
        final double cs = math.cos(a), sn = math.sin(a);
        canvas.drawLine(Offset(c.dx + cs * (r0 + 1), c.dy + sn * (r0 + 1)), Offset(c.dx + cs * l, c.dy + sn * l), bar);
        if (caps[b] > 0.06) {
          final double cl = r0 + caps[b] * len + bw * 1.2;
          canvas.drawLine(Offset(c.dx + cs * cl, c.dy + sn * cl), Offset(c.dx + cs * (cl + bw * 0.9), c.dy + sn * (cl + bw * 0.9)), cap);
        }
      }
    }
    // the ring the bars grow from, and a turning dashed orbit
    canvas.drawCircle(c, r0, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.4
      ..color = tint.withOpacity(live ? 0.75 : 0.45));
    final double orbit = r0 + len + bw * 2.2;
    final Paint op = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1
      ..color = tint.withOpacity(0.25 + swell * 0.5);
    const List<double> dash = <double>[0.5, 0.22, 0.12, 0.3];
    double ang = rot;
    for (int i = 0; i < 4; i++) {
      final double sweep = dash[i] * r0 / orbit;
      if (i % 2 == 0) canvas.drawArc(Rect.fromCircle(center: c, radius: orbit), ang, sweep, false, op);
      ang += sweep;
    }
    for (int i = 0; i < 4; i++) {
      final double sweep = dash[i] * r0 / orbit;
      if (i % 2 == 0) canvas.drawArc(Rect.fromCircle(center: c, radius: orbit), ang, sweep, false, op);
      ang += sweep;
    }
    // nucleus
    final double nr = r0 * (0.34 + bass * 0.3) * (live ? 1 : 0.85);
    canvas.drawCircle(c, nr, Paint()..shader = RadialGradient(colors: <Color>[Colors.white, tint.withOpacity(0.9), tint.withOpacity(0.25), tint.withOpacity(0)], stops: const <double>[0, .25, .55, 1]).createShader(Rect.fromCircle(center: c, radius: nr)));
  }

  @override
  bool shouldRepaint(covariant _CorePainter old) => true;
}

  // ---- the core animation: synthetic spectrum with real meter ballistics
  void _tick(Duration now) {
    final double dt = _last == Duration.zero ? 0.016 : math.min(0.1, (now - _last).inMicroseconds / 1e6);
    _last = now;
    _phase += dt;
    if (widget.layout == 'hero' && ++_visTick % 20 == 0) _checkOnScreen();
    if (!_onScreen && !_voice.speaking && _radioName.isEmpty) return;
    final String m = _mode;
    double target = 0;
    if (m == 'thinking') target = 0.3 + 0.2 * math.sin(_phase * 6);
    else if (m == 'listening') target = 0.12 + 0.05 * math.sin(_phase * 3);
    else if (m == 'speaking') target = 0.25 + 0.55 * math.max(0, math.sin(_phase * 11) * 0.6 + math.sin(_phase * 4.3) * 0.4 + 0.2);
    // JC-LAZO-JUNE-NATIVE-1011-003: the real thing when we have it - the web
    // analyser (clips and radio), else the clip's level track; otherwise motion
    final bool radioOn = JuneRadio.I.playing;
    List<double>? live;
    if (kIsWeb && (m == 'speaking' || (radioOn && _radioFft && m == 'idle'))) {
      final List<double>? f = _juneWebFrame(_nb);
      if (f != null) {
        live = f.sublist(0, _nb);
        target = math.min(1, f[_nb] * 0.9 + 0.05);
      }
    }
    if (live == null && m == 'speaking') live = _voice.frame(_nb);
    if (live != null && !(kIsWeb && m != 'speaking')) {
      double sum = 0;
      for (final double v in live) sum += v;
      target = math.min(1, (sum / _nb) * 1.7 + 0.06);
    }
    if (live == null && radioOn && m == 'idle') {
      // phones (and streams without cross-origin audio): music motion - a beat and a slow swell
      final double beat = math.max(0, math.sin(_phase * 2 * math.pi * 1.9));
      target = 0.22 + 0.18 * math.pow(beat, 3) + 0.08 * math.sin(_phase * 0.6);
    }
    final double atk = 1 - math.exp(-dt * 60), rel = (live != null ? 7.5 : 4.2) * dt, capFall = 2.2 * dt;
    for (int b = 0; b < _nb; b++) {
      double shape;
      if (m == 'speaking') shape = math.exp(-math.pow((b - _nb * 0.3) / (_nb * 0.28), 2)) * (0.6 + 0.4 * math.sin(_phase * 9 + b * 0.7).abs());
      else if (m == 'thinking') shape = 0.5 + 0.5 * math.sin(_phase * 5 - b * 0.35);
      else if (radioOn) shape = (0.55 + 0.45 * math.sin(_phase * 3.1 + b * 0.9)) * (0.7 + 0.3 * math.exp(-math.pow((b - _nb * 0.2) / (_nb * 0.35), 2)));
      else shape = 0.4 + 0.3 * math.sin(_phase * 2 + b);
      final double v = live != null ? live[b] : (m == 'idle' && !radioOn ? 0.05 + 0.04 * math.sin(_phase * 1.1 + b * 0.45) * math.sin(_phase * 0.7 + b * 0.2) : target * shape);
      if (v > _lvl[b]) _lvl[b] += (v - _lvl[b]) * atk; else _lvl[b] = math.max(v, _lvl[b] - rel);
      if (_lvl[b] >= _cap[b]) {
        _cap[b] = _lvl[b];
        _capT[b] = 0.25;
      } else {
        _capT[b] -= dt;
        if (_capT[b] <= 0) _cap[b] = math.max(_lvl[b], _cap[b] - capFall);
      }
    }
    _swell += (target - _swell) * 0.15;
    _bass += ((live != null ? (live[0] + live[1] + live[2] + live[3]) / 4 : target * 0.5) - _bass) * (live != null ? 0.3 : 0.1);
    _rot += dt * (m == 'thinking' ? 1.4 : m == 'idle' ? (radioOn ? 0.3 : 0.12) : 0.25 + _swell * 0.6);
    final Color want = m == 'listening' ? _rose : m == 'thinking' ? const Color(0xFFC496C4) : m == 'speaking' ? const Color(0xFFF5E4B4) : (_radioName.isNotEmpty ? _sage : _gold);
    _tint = Color.lerp(_tint, want, 0.08) ?? want;
    if (mounted) setState(() {});
  }

  // ---- chat
  void _addMsg(String role, String text, {Map<String, dynamic>? item, String? url}) {
    _chat.add(<String, dynamic>{'role': role, 'text': text, if (item != null) 'item': item, if (url != null) 'url': url});

  Future<void> _ask(String raw) async {
    final String text = raw.trim();
    if (text.isEmpty || _thinking) return;
    _voice.stop();
    _q.clear();
    _addMsg('me', text);
    final String quick = _local(text);
    if (quick.isNotEmpty) {
      _addMsg('ai', quick);
      _voice.say(quick);
      return;
    }
    setState(() {
      _thinking = true;
      _mode = 'thinking';
    });
    final Map<String, dynamic> msg = <String, dynamic>{'role': 'ai', 'text': ''};
    _chat.add(msg);
    String full = '';
    int spoken = 0;
    void flush(bool fin) {
      final String rest = full.substring(spoken);
      if (fin) {
        if (rest.trim().isNotEmpty) _voice.say(rest);
        spoken = full.length;
        return;
      }
      final RegExpMatch? m = RegExp(r'^[\s\S]*?[.!?](?=\s|$)').firstMatch(rest);
      if (m == null) return;
      _voice.say(m.group(0)!);
      spoken += m.group(0)!.length;
    }

    await _api.chat(text, _hist, onDelta: (String d) {
      full += d;
      msg['text'] = full;
      if (mounted) setState(() => _mode = _voice.speaking ? 'speaking' : 'thinking');
      flush(false);
      if (_chatScroll.hasClients) _chatScroll.jumpTo(_chatScroll.position.maxScrollExtent);
    }, onDone: (Map<String, dynamic> d) {
      full = full.trim();
      msg['text'] = full.isEmpty ? '…' : full;
      flush(true);
      final List<Map<String, dynamic>> hist = _list(d['messages']);
      if (hist.isNotEmpty) _hist = hist;
      for (final Map<String, dynamic> a in _list(d['actions'])) {
        _handleAction(a);
      }
    }, onError: (String e) {
      msg['text'] = (full + '\n[' + e + ']').trim();
    });
    if (!mounted) return;
    setState(() {
      _thinking = false;
      _mode = _voice.speaking ? 'speaking' : 'idle';
    });
    _load();
  }

  void _handleAction(Map<String, dynamic> a) {
    final String t = _s(a['type']);
    if (t == 'open') {
      final String u = _s(a['url']);
      _addMsg('sys', 'Opening ${_s(a['label'])}.', url: u);
      launchUrl(Uri.parse(u), mode: LaunchMode.externalApplication).catchError((Object _) => false);
    } else if (t == 'memory') {
      setState(() => _snap['memory'] = a['value']);
    } else if (t == 'reminders') {
      _loadReminders();
    } else if (t == 'confirm') {
      _addMsg('confirm', _s(_map(a['item'])['summary']), item: _map(a['item']));
    }
  }

  Future<void> _decide(Map<String, dynamic> msg, String decision) async {
    final Map<String, dynamic> item = _map(msg['item']);
    setState(() => msg['busy'] = true);
    try {
      final dynamic rec = await _api.post('/api/act', <String, dynamic>{'item': item, 'decision': decision});
      final Map<String, dynamic> r = _map(rec);
      msg['status'] = _s(r['status']);
      msg['result'] = _s(r['result']);
      if (msg['status'] == 'done') {
        _voice.say(_s(r['result']).isEmpty ? 'Done.' : _s(r['result']));
        _load();
      } else if (msg['status'] == 'failed') {
        _voice.say('That didn\'t go through. ' + _s(r['result']));
      }
    } catch (e) {
      msg['status'] = 'failed';
      msg['result'] = e.toString();
    }
    if (mounted) setState(() => msg['busy'] = false);
  }

  // a few answers that never need the brain
  String _local(String raw) {
    final String t = raw.toLowerCase().replaceAll(RegExp(r'^june,?\s*'), '').replaceAll(RegExp(r'[.!?]+$'), '').trim();
    if (RegExp(r'^(clear|reset|wipe|start over)( the| our| my)? ?(chat|conversation|page)?$').hasMatch(t) && t.length > 4) {
      _clearChat();
      return 'Fresh page. What next?';
    }
    if (RegExp(r'^(what time is it|time)$').hasMatch(t)) {
      final DateTime d = DateTime.now();
      final int h = d.hour % 12 == 0 ? 12 : d.hour % 12;
      return 'It\'s $h:${d.minute.toString().padLeft(2, '0')} ${d.hour < 12 ? 'AM' : 'PM'}.';
    }
    if (RegExp(r'^(stop|quiet|shush|never ?mind|cancel)$').hasMatch(t)) {
      _voice.stop();
      return ' ';
    }
    if (RegExp(r'^(stop|pause|turn off|kill|mute) (the )?(radio|music)$|^radio off$').hasMatch(t)) {
      _radioStop();
      return 'Radio off.';
    }
    final RegExpMatch? rp = RegExp(r'^(?:play|put on|turn on|start)\s+(?:the\s+|some\s+)?(radio|music|.+?)(?:\s+radio)?$').firstMatch(t);
    if (rp != null) {
      final String want = rp.group(1)!.replaceAll(RegExp(r'^something\s+'), '').replaceAll(' ', '');
      final int byName = _radio.indexWhere((List<String> r) => r[0].toLowerCase().replaceAll(' ', '') == want);
      int i = want == 'radio' || want == 'music' ? 0 : byName >= 0 ? byName : _radio.indexWhere((List<String> r) => r[2].split(',').any((String k) => want.contains(k)));
      if (i >= 0) {
        _radioPlay(i);
        return 'Playing ${_radio[i][0]}. Say "stop the music" or tap Stop under the core whenever you like.';
      }
    }
    return '';
  }

  }

  // ---- the deck (JC-LAZO-JUNE-DECK-1010-001): June across the top of a dashboard. The host
  // draws the surface and the numbers; this is her core, her words, and the way to talk to her.
  String _greeting() {
    final int h = DateTime.now().hour;
    final String tod = h < 4 ? 'Still up' : h < 12 ? 'Good morning' : h < 17 ? 'Good afternoon' : 'Good evening';
    String who = _couple ? _s(_map(_snap['couple'])['names']) : '';
    if (who.isEmpty) {
      // the person's own first name, from their Lazo sign-in
      final String dn = (FirebaseAuth.instance.currentUser?.displayName ?? '').trim();
      if (dn.isNotEmpty) who = dn.split(' ').first;
    }
    return who.isEmpty ? '$tod.' : '$tod, $who.';
  }

  String _headline() {
    if (_loading && _snap.isEmpty) return 'Reading your day…';
    final Map<String, dynamic> ms = _map(_snap['milestone']);
    final String mark = _s(ms['label']).isNotEmpty ? (_couple ? '${ms['label']}. ' : '${ms['label']} for ${_s(ms['who'])}. ') : '';
    final String brief = mark + _s(_map(_snap['brief'])['text']).trim();
    if (brief.isNotEmpty) {
      final List<String> sents = brief.split(RegExp(r'(?<=[.!?])\s+'));
      return sents.take(2).join(' ');
    }
    final int w = _list(_snap['waiting']).length;
    final Map<String, dynamic> m = _map(_snap['money']);
    final double due = _n(m[_couple ? 'open' : 'outstanding']).toDouble();
    final List<String> parts = <String>[
      w == 0 ? 'Nobody is waiting on you.' : '$w ${_couple ? (w == 1 ? 'vendor is' : 'vendors are') : (w == 1 ? 'couple is' : 'couples are')} waiting on you.',
      if (due > 0) '\$${due.round()} ${_couple ? 'due soon' : 'outstanding'}.',
      'Ask me anything.',
    ];
    return parts.join(' ');
  }

  // ---------------------------------------------------------------- the hero (JC-LAZO-JUNE-NATIVE-1013-006)
  String _heroGreeting() {
    final int h = DateTime.now().hour;
    final String tod = h < 4 ? 'Still up' : h < 12 ? 'Good morning' : h < 17 ? 'Good afternoon' : 'Good evening';
    String who = '';
    if (_couple) {
      final String names = _s(_map(_snap['couple'])['names']).trim();
      if (names.isNotEmpty && names.length <= 24) who = names;
    }
    if (who.isEmpty) {
      final String dn = (FirebaseAuth.instance.currentUser?.displayName ?? '').trim();
      if (dn.isNotEmpty) who = dn.split(' ').first;
    }
    return who.isEmpty ? '$tod.' : '$tod, $who.';
  }

  String _heroLine() {
    final String brief = _s(_map(_snap['brief'])['text']).trim();
    if (brief.isNotEmpty) {
      final RegExp sent = RegExp(r'^[\s\S]*?[.!?](?=\s|$)');
      final String first = (sent.firstMatch(brief)?.group(0) ?? brief).trim();
      // the brief opens with its own greeting; the hero already greeted, so take the sentence after it
      if (RegExp(r'^(good (morning|afternoon|evening)|hello|hi|hey|morning|evening)\b', caseSensitive: false).hasMatch(first) && brief.length > first.length + 8) {
        final RegExpMatch? m2 = sent.firstMatch(brief.substring(first.length).trim());
        if (m2 != null) return m2.group(0)!.trim();
      }
      return first;
    }
    if (_loading && _snap.isEmpty) return '';
    final String h = _headline();
    return h == 'Reading your day…' ? '' : h;
  }

// ---------------------------------------------------------------- v008: one radio for the whole app
class JuneRadio {
  JuneRadio._();
  static final JuneRadio I = JuneRadio._();
  final ValueNotifier<int> rev = ValueNotifier<int>(0);
  AudioPlayer? _player;
  html.AudioElement? _el;
  int idx = -1;
  double vol = 0.7;
  bool fft = false;
  bool playing = false;
  bool _on = false; // a station is loaded (playing or paused)
  bool _ducked = false;
  void Function(String message)? onError;
  final Set<Object> _visible = <Object>{};
  OverlayEntry? _entry;

  static List<List<String>> get stations => _JuneNativeState._radio;
  static List<String> get groups => _JuneNativeState._radioGroups;
  String get name => _on && idx >= 0 ? stations[idx][0] : '';
  bool get anyJuneVisible => _visible.isNotEmpty;

  void _bump() => rev.value = rev.value + 1;
  double get _eff => _ducked ? vol * 0.25 : vol;

  Future<void> play(int i, [BuildContext? ctx]) async {
    _halt();
    idx = i;
    _on = true;
    playing = true;
    _bump();
    if (ctx != null) _ensureMini(ctx);
    final List<String> st = stations[i];
    try {
      if (kIsWeb) {
        await _playWeb(st[1], true);
      } else {
        final AudioPlayer p = AudioPlayer();
        try {
          p.setAudioContext(AudioContext(iOS: AudioContextIOS(category: AVAudioSessionCategory.playback), android: AudioContextAndroid()));
        } catch (_) {}
        p.setVolume(_eff);
        _player = p;
        await p.play(UrlSource(st[1]));
      }
    } catch (e) {
      onError?.call('That station wouldn\'t start: ' + e.toString());
      stop();
    }
  }

  // web: through the analyser first (needs the stream to allow cross-origin audio); on error once more plainly
  Future<void> _playWeb(String url, bool withFft) async {
    final html.AudioElement a = html.AudioElement();
    if (withFft) a.crossOrigin = 'anonymous';
    a.src = url;
    a.volume = _eff;
    _el = a;
    fft = withFft;
    if (withFft) _juneWebAttach(a);
    a.onError.listen((_) {
      if (!identical(_el, a)) return;
      if (withFft) {
        _juneWebDetach(a);
        _el = null;
        _playWeb(url, false);
      } else {
        onError?.call('That station wouldn\'t start.');
        stop();
      }
    });
    _bump();
    await a.play().catchError((Object _) {});
  }

  void _halt() {
    try {
      _el?.pause();
      if (fft) _juneWebDetach(_el);
      _el?.remove();
      _el = null;
    } catch (_) {}
    try {
      _player?.stop();
      _player?.dispose();
      _player = null;
    } catch (_) {}
    fft = false;
  }

  void stop() {
    _halt();
    _on = false;
    playing = false;
    if (!JuneSpeech.talking.value) _removeMini();
    _bump();
  }

  void pause() {
    if (!_on || !playing) return;
    try {
      _el?.pause();
    } catch (_) {}
    try {
      _player?.pause();
    } catch (_) {}
    playing = false;
    _bump();
  }

  void resume() {
    if (!_on) {
      play(idx < 0 ? 0 : idx);
      return;
    }
    if (playing) return;
    try {
      _el?.play().catchError((Object _) {});
    } catch (_) {}
    try {
      _player?.resume();
    } catch (_) {}
    playing = true;
    _bump();
  }

  void toggle() => playing ? pause() : resume();

  void step(int d, [BuildContext? ctx]) {
    if (idx < 0) {
      play(0, ctx);
      return;
    }
    play((idx + d + stations.length) % stations.length, ctx);
  }

  void _applyVol() {
    try {
      _el?.volume = _eff;
    } catch (_) {}
    try {
      _player?.setVolume(_eff);
    } catch (_) {}
  }

  void setVol(double v) {
    vol = v.clamp(0.0, 1.0);
    _applyVol();
    _bump();
  }

  /// the music dips while June speaks
  void duck(bool on) {
    if (_ducked == on) return;
    _ducked = on;
    _applyVol();
  }

  void setVisible(Object who, bool on) {
    final bool before = _visible.isNotEmpty;
    if (on) {
      _visible.add(who);
    } else {
      _visible.remove(who);
    }
    if (before != _visible.isNotEmpty) {
      // never during a build: the mini player listens to rev
      WidgetsBinding.instance.addPostFrameCallback((_) => _bump());
    }
  }

  void _ensureMini(BuildContext ctx) {
    if (_entry != null) return;
    try {
      final OverlayState? o = Overlay.maybeOf(ctx, rootOverlay: true);
      if (o == null) return;
      final OverlayEntry e = OverlayEntry(builder: (BuildContext c) => const _JuneMini());
      _entry = e;
      o.insert(e);
    } catch (_) {}
  }

  void _removeMini() {
    try {
      _entry?.remove();
    } catch (_) {}
    _entry = null;
  }
}

// ---------------------------------------------------------------- v008: June's voice for other widgets
/// JuneSpeech.say('...') from any Lazo widget: through the June on screen if there is one (her core
/// moves with it), else a voice of its own. Honors the June voice switch (prefs 'juneSpeak'), waits
/// for the first tap on the web, and a newer line replaces an older one.
class JuneSpeech {
  /// v009: June's one voice for the whole app; every June widget speaks through it, so she
  /// outlives the page she started on (like JuneRadio)
  static final ValueNotifier<bool> talking = ValueNotifier<bool>(false);
  static final _Voice voice = _Voice(_Api(), (bool s) {
    JuneRadio.I.duck(s);
    talking.value = s;
    // never during a build: the floating player listens to rev
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!talking.value && !JuneRadio.I._on) JuneRadio.I._removeMini();
      JuneRadio.I._bump();
    });
  });
  /// true while something else (the first-run tour) is narrating: the dashboard line keeps quiet
  static bool hold = false;
  static int _gen = 0;
  static Timer? _wait;

  static Future<bool> enabled() async {
    try {
      final SharedPreferences p = await SharedPreferences.getInstance();
      return p.getBool('juneSpeak') ?? true;
    } catch (_) {
      return true;
    }
  }

  static Future<void> setEnabled(bool on) async {
    try {
      final SharedPreferences p = await SharedPreferences.getInstance();
      await p.setBool('juneSpeak', on);
    } catch (_) {}
    voice.enabled = on;
    if (!on) stop();
  }

  static Future<void> say(String text) async {
    final int g = ++_gen;
    _wait?.cancel();
    stop(keepGen: true);
    if (text.trim().isEmpty || !await enabled() || g != _gen) return;
    void go() {
      if (g != _gen) return;
      voice.say(text);
    }
    if (juneCanSpeak()) {
      go();
      return;
    }
    int n = 0;
    _wait = Timer.periodic(const Duration(milliseconds: 300), (Timer t) {
      if (g != _gen || ++n > 400) {
        t.cancel();
        return;
      }
      if (juneCanSpeak()) {
        t.cancel();
        go();
      }
    });
  }

  static void stop({bool keepGen = false}) {
    if (!keepGen) {
      _gen++;
      _wait?.cancel();
    }
    try {
      voice.stop();
    } catch (_) {}
  }
}
