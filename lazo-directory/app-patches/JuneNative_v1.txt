String _day(String iso) {
  try {
    final DateTime d = DateTime.parse(iso.length == 10 ? iso + 'T12:00:00' : iso).toLocal();
    const List<String> w = <String>['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    const List<String> m = <String>['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    return '${w[d.weekday - 1]}, ${m[d.month - 1]} ${d.day}';
  } catch (_) {
    return iso;
  }
}
String _when(String iso) {
  try {
    final DateTime d = DateTime.parse(iso).toLocal();
    final int h = d.hour % 12 == 0 ? 12 : d.hour % 12;
    return '${_day(iso)} · $h:${d.minute.toString().padLeft(2, '0')} ${d.hour < 12 ? 'AM' : 'PM'}';
  } catch (_) {
    return iso;
  }
}

// ---------------------------------------------------------------- the API: the June worker, signed in as this user
class _Api {
  Future<Map<String, String>> _headers() async {
    final User? u = FirebaseAuth.instance.currentUser;
    final String t = u == null ? '' : (await u.getIdToken() ?? '');
    return <String, String>{'authorization': 'Bearer $t', 'content-type': 'application/json'};
  }

  Future<Map<String, dynamic>> get(String path) async {
    final httpc.Response r = await httpc.get(Uri.parse(_kHub + path), headers: await _headers()).timeout(const Duration(seconds: 30));
    final dynamic j = jsonDecode(r.body.isEmpty ? '{}' : r.body);
    if (r.statusCode >= 400) throw Exception(_s(j is Map ? (j['message'] ?? j['error']) : r.statusCode));
    return j is Map ? Map<String, dynamic>.from(j) : <String, dynamic>{'data': j};
  }

  Future<dynamic> post(String path, Map<String, dynamic> body) async {
    final httpc.Response r = await httpc.post(Uri.parse(_kHub + path), headers: await _headers(), body: jsonEncode(body)).timeout(const Duration(seconds: 60));
    final dynamic j = jsonDecode(r.body.isEmpty ? '{}' : r.body);
    if (r.statusCode >= 400) throw Exception(_s(j is Map ? (j['message'] ?? j['error']) : r.statusCode));
    return j;
  }

  // the chat streams server-sent events: delta {text}, done {reply, actions, messages}, error {message}
  Future<void> chat(String text, List<Map<String, dynamic>> history, {required void Function(String) onDelta, required void Function(Map<String, dynamic>) onDone, required void Function(String) onError}) async {
    final httpc.Client client = httpc.Client();
    try {
      final httpc.Request req = httpc.Request('POST', Uri.parse('$_kHub/api/chat'));
      req.headers.addAll(await _headers());
      req.body = jsonEncode(<String, dynamic>{'text': text, 'messages': history});
      final httpc.StreamedResponse resp = await client.send(req).timeout(const Duration(seconds: 90));
      if (resp.statusCode >= 400) {
        onError('June is unavailable (${resp.statusCode}).');
        return;
      }
      String buf = '';
      bool done = false;
      await for (final String chunk in resp.stream.transform(utf8.decoder)) {
        buf += chunk;
        int idx;
        while ((idx = buf.indexOf('\n\n')) >= 0) {
          final String frame = buf.substring(0, idx);
          buf = buf.substring(idx + 2);
          final RegExpMatch? ev = RegExp(r'^event: (.*)$', multiLine: true).firstMatch(frame);
          final RegExpMatch? da = RegExp(r'^data: (.*)$', multiLine: true).firstMatch(frame);
          if (ev == null || da == null) continue;
          dynamic d;
          try {
            d = jsonDecode(da.group(1)!);
          } catch (_) {
            continue;
          }
          final String kind = ev.group(1)!.trim();
          if (kind == 'delta') {
            onDelta(_s(_map(d)['text']));
          } else if (kind == 'done') {
            done = true;
            onDone(_map(d));
          } else if (kind == 'error') {
            onError(_s(_map(d)['message']));
          }
        }
      }
      if (!done) onDone(<String, dynamic>{});
    } catch (e) {
      onError(e.toString());
    } finally {
      client.close();
    }
  }
}

// ---------------------------------------------------------------- the voice out: ElevenLabs through the hub, one clip per sentence
class _Voice {
  _Voice(this.api, this.onState);
  final _Api api;
  final void Function(bool speaking) onState;
  final List<String> _queue = <String>[];
  bool speaking = false;
  bool enabled = true;
  int _gen = 0;
  AudioPlayer? _player;
  html.AudioElement? _el;

  AudioPlayer _ensure() {
    if (_player != null) return _player!;
    final AudioPlayer p = AudioPlayer();
    try {
      p.setAudioContext(AudioContext(
        iOS: AudioContextIOS(category: AVAudioSessionCategory.playback),
        android: AudioContextAndroid(),
      ));
    } catch (_) {}
    p.setReleaseMode(ReleaseMode.stop);
    _player = p;
    return p;
  }

  void say(String text) {
    final String t = text.trim();
    if (!enabled || t.isEmpty) return;
    _queue.add(t);
    if (!speaking) _pump();
  }

  Future<void> _pump() async {
    speaking = true;
    onState(true);
    final int gen = _gen;
    while (_queue.isNotEmpty && gen == _gen) {
      final String t = _queue.removeAt(0);
      String url = '';
      try {
        final dynamic j = await api.post('/api/clip', <String, dynamic>{'text': t});
        url = _s(_map(j)['url']);
      } catch (e) {
        debugPrint('LAZO june clip: ' + e.toString());
      }
      if (url.isEmpty || gen != _gen) continue;
      await playUrl(url, gen);
    }
    if (gen == _gen) {
      speaking = false;
      onState(false);
    }
  }

  // plays one URL to the end (web: an <audio> element; phones: audioplayers)
  Future<void> playUrl(String url, [int? gen]) async {
    final int g = gen ?? _gen;
    try {
      if (kIsWeb) {
        final html.AudioElement a = html.AudioElement();
        _el = a;
        a.src = url;
        a.preload = 'auto';
        final Completer<void> c = Completer<void>();
        a.onEnded.listen((_) {
          if (!c.isCompleted) c.complete();
        });
        a.onError.listen((_) {
          if (!c.isCompleted) c.complete();
        });
        await a.play().catchError((Object _) {});
        await c.future.timeout(const Duration(minutes: 6), onTimeout: () {});
        if (identical(_el, a)) _el = null;
      } else {
        final AudioPlayer p = _ensure();
        await p.stop();
        final Completer<void> c = Completer<void>();
        final StreamSubscription<void> sub = p.onPlayerComplete.listen((_) {
          if (!c.isCompleted) c.complete();
        });
        await p.play(UrlSource(url));
        await c.future.timeout(const Duration(minutes: 6), onTimeout: () {});
        await sub.cancel();
      }
    } catch (e) {
      debugPrint('LAZO june play: ' + e.toString());
    }
    if (g != _gen) return;
  }

  void stop() {
    _gen++;
    _queue.clear();
    try {
      _el?.pause();
      _el = null;
    } catch (_) {}
    try {
      _player?.stop();
    } catch (_) {}
    if (speaking) {
      speaking = false;
      onState(false);
    }
  }

  void dispose() {
    stop();
    try {
      _player?.dispose();
    } catch (_) {}
  }
}

// ---------------------------------------------------------------- the voice in: the device's speech recognition
class _Ears {
  _Ears({required this.onText, required this.onState});
  final void Function(String text, bool isFinal) onText;
  final void Function(bool listening) onState;
  final stt.SpeechToText _stt = stt.SpeechToText();
  bool ready = false;
  bool listening = false;

  Future<bool> init() async {
    try {
      ready = await _stt.initialize(
        onStatus: (String s) {
          if (s == 'done' || s == 'notListening') {
            if (listening) {
              listening = false;
              onState(false);
            }
          }
        },
        onError: (dynamic e) {
          if (listening) {
            listening = false;
            onState(false);
          }
        },
      );
    } catch (e) {
      ready = false;
    }
    return ready;
  }

  Future<void> start() async {
    if (!ready && !(await init())) return;
    listening = true;
    onState(true);
    try {
      await _stt.listen(
        onResult: (dynamic r) {
          try {
            onText(_s(r.recognizedWords), r.finalResult == true);
          } catch (_) {}
        },
        listenFor: const Duration(seconds: 20),
        pauseFor: const Duration(seconds: 3),
        localeId: 'en_US',
      );
    } catch (e) {
      listening = false;
      onState(false);
    }
  }

  Future<void> stop() async {
    try {
      await _stt.stop();
    } catch (_) {}
    if (listening) {
      listening = false;
      onState(false);
    }
  }
}

// ---------------------------------------------------------------- the core: a circular spectrum analyser
class _CorePainter extends CustomPainter {
  _CorePainter({required this.levels, required this.caps, required this.swell, required this.bass, required this.rot, required this.tint, required this.live});
  final List<double> levels;
  final List<double> caps;
  final double swell;
  final double bass;
  final double rot;
  final Color tint;
  final bool live;

  @override
  void paint(Canvas canvas, Size size) {
    final double w = size.width;
    final Offset c = Offset(w / 2, size.height / 2);
    final int nb = levels.length;
    final double r0 = w * 0.21 * (1 + bass * 0.05);
    final double len = w * 0.25;
    // glow
    final double gr = r0 * (1.9 + swell * 0.5);
    canvas.drawCircle(c, gr, Paint()..shader = RadialGradient(colors: <Color>[tint.withOpacity(live ? 0.28 : 0.16), tint.withOpacity(0.06), tint.withOpacity(0)], stops: const <double>[0, .5, 1]).createShader(Rect.fromCircle(center: c, radius: gr)));
    // bars, bass at the bottom climbing both sides
    final double bw = math.max(1.5, (math.pi * r0) / nb * 0.62);
    final Paint bar = Paint()
      ..color = tint.withOpacity(live ? 0.92 : 0.5)
      ..strokeWidth = bw
      ..strokeCap = StrokeCap.butt;
    final Paint cap = Paint()
      ..color = Colors.white.withOpacity(live ? 0.85 : 0.35)
      ..strokeWidth = math.max(1, bw * 0.9);
    for (int s = -1; s <= 1; s += 2) {
      for (int b = 0; b < nb; b++) {
        final double a = math.pi / 2 + s * ((b + 0.5) / nb) * math.pi;
        final double l = r0 + math.max(1, levels[b] * len);
        final double cs = math.cos(a), sn = math.sin(a);
        canvas.drawLine(Offset(c.dx + cs * (r0 + 1), c.dy + sn * (r0 + 1)), Offset(c.dx + cs * l, c.dy + sn * l), bar);
        if (caps[b] > 0.06) {
          final double cl = r0 + caps[b] * len + bw * 1.2;
          canvas.drawLine(Offset(c.dx + cs * cl, c.dy + sn * cl), Offset(c.dx + cs * (cl + bw * 0.9), c.dy + sn * (cl + bw * 0.9)), cap);
        }
      }
    }
    // the ring the bars grow from, and a turning dashed orbit
    canvas.drawCircle(c, r0, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.4
      ..color = tint.withOpacity(live ? 0.75 : 0.45));
    final double orbit = r0 + len + bw * 2.2;
    final Paint op = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1
      ..color = tint.withOpacity(0.25 + swell * 0.5);
    const List<double> dash = <double>[0.5, 0.22, 0.12, 0.3];
    double ang = rot;
    for (int i = 0; i < 4; i++) {
      final double sweep = dash[i] * r0 / orbit;
      if (i % 2 == 0) canvas.drawArc(Rect.fromCircle(center: c, radius: orbit), ang, sweep, false, op);
      ang += sweep;
    }
    for (int i = 0; i < 4; i++) {
      final double sweep = dash[i] * r0 / orbit;
      if (i % 2 == 0) canvas.drawArc(Rect.fromCircle(center: c, radius: orbit), ang, sweep, false, op);
      ang += sweep;
    }
    // nucleus
    final double nr = r0 * (0.34 + bass * 0.3) * (live ? 1 : 0.85);
    canvas.drawCircle(c, nr, Paint()..shader = RadialGradient(colors: <Color>[Colors.white, tint.withOpacity(0.9), tint.withOpacity(0.25), tint.withOpacity(0)], stops: const <double>[0, .25, .55, 1]).createShader(Rect.fromCircle(center: c, radius: nr)));
  }

  @override
  bool shouldRepaint(covariant _CorePainter old) => true;
}

  bool _micAvailable = false;

  String _heard = '';

  final List<double> _capT = List<double>.filled(_nb, 0);
  double _swell = 0, _bass = 0, _rot = 0, _phase = 0;
  Duration _last = Duration.zero;
  Color _tint = _gold;

  bool get _couple => widget.role == 'couple';

  @override
  void initState() {

  }

  // ---- the core animation: synthetic spectrum with real meter ballistics
  void _tick(Duration now) {
    final double dt = _last == Duration.zero ? 0.016 : math.min(0.1, (now - _last).inMicroseconds / 1e6);
    _last = now;
    _phase += dt;
    final String m = _mode;
    double target = 0;
    if (m == 'thinking') target = 0.3 + 0.2 * math.sin(_phase * 6);
    else if (m == 'listening') target = 0.12 + 0.05 * math.sin(_phase * 3);
    else if (m == 'speaking') target = 0.25 + 0.55 * math.max(0, math.sin(_phase * 11) * 0.6 + math.sin(_phase * 4.3) * 0.4 + 0.2);
    final double atk = 1 - math.exp(-dt * 60), rel = 4.2 * dt, capFall = 2.2 * dt;
    for (int b = 0; b < _nb; b++) {
      double shape;
      if (m == 'speaking') shape = math.exp(-math.pow((b - _nb * 0.3) / (_nb * 0.28), 2)) * (0.6 + 0.4 * math.sin(_phase * 9 + b * 0.7).abs());
      else if (m == 'thinking') shape = 0.5 + 0.5 * math.sin(_phase * 5 - b * 0.35);
      else shape = 0.4 + 0.3 * math.sin(_phase * 2 + b);
      final double v = m == 'idle' ? 0.03 + 0.025 * math.sin(_phase * 1.3 + b * 0.5) : target * shape;
      if (v > _lvl[b]) _lvl[b] += (v - _lvl[b]) * atk; else _lvl[b] = math.max(v, _lvl[b] - rel);
      if (_lvl[b] >= _cap[b]) {
        _cap[b] = _lvl[b];
        _capT[b] = 0.25;
      } else {
        _capT[b] -= dt;
        if (_capT[b] <= 0) _cap[b] = math.max(_lvl[b], _cap[b] - capFall);
      }
    }
    _swell += (target - _swell) * 0.15;
    _bass += (target * 0.5 - _bass) * 0.1;
    _rot += dt * (m == 'thinking' ? 1.4 : m == 'idle' ? 0.12 : 0.25 + _swell * 0.6);
    final Color want = m == 'listening' ? _rose : m == 'thinking' ? const Color(0xFFC496C4) : m == 'speaking' ? const Color(0xFFF5E4B4) : (_radioName.isNotEmpty ? _sage : _gold);
    _tint = Color.lerp(_tint, want, 0.08) ?? want;
    if (mounted) setState(() {});
  }

  // ---- data
  Future<void> _load({bool first = false}) async {
    try {
      final Map<String, dynamic> s = await _api.get('/api/state');
      if (!mounted) return;
      setState(() {
        _snap = s;
        _loading = false;
        _err = '';
      });
      if (first) {
        if (_map(s['brief'])['text'] == null) _makeBrief(false);
        final List<Map<String, dynamic>> due = _list(s['due']);
        if (due.isNotEmpty) {
          final String line = due.length == 1 ? 'You asked me to remind you: ${_s(due[0]['text'])}.' : 'You have ${due.length} reminders due. ' + due.take(3).map((Map<String, dynamic> r) => _s(r['text'])).join('; ') + '.';
          _addMsg('ai', line);
          _voice.say(line);
        }
      }
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _loading = false;
        _err = e.toString().replaceFirst('Exception: ', '');
      });
    }
  }

  Future<void> _makeBrief(bool force) async {
    setState(() => _briefBusy = true);
    try {
      final dynamic b = await _api.post('/api/brief', <String, dynamic>{'force': force});
      if (!mounted) return;
      setState(() => _snap['brief'] = _map(b));
    } catch (e) {
      debugPrint('LAZO june brief: ' + e.toString());
    }
    if (mounted) setState(() => _briefBusy = false);
  }

  Future<void> _listenBrief() async {
    if (_briefPlaying) {
      _voice.stop();
      setState(() => _briefPlaying = false);
      return;
    }
    final String text = _s(_map(_snap['brief'])['text']);
    if (text.isEmpty) return;
    setState(() => _briefPlaying = true);
    try {
      // the hub keeps the brief's ElevenLabs audio; ask for it as a clip first, else read the text
      final dynamic j = await _api.post('/api/clip', <String, dynamic>{'brief': true}).catchError((Object _) => <String, dynamic>{});
      final String url = _s(_map(j)['url']);
      if (url.isNotEmpty) {
        setState(() => _mode = 'speaking');
        await _voice.playUrl(url);
        if (mounted) setState(() => _mode = 'idle');
      } else {
        for (final String sent in text.split(RegExp(r'(?<=[.!?])\s+'))) {
          _voice.say(sent);
        }
        while (_voice.speaking && mounted) {
          await Future<void>.delayed(const Duration(milliseconds: 300));
        }
      }
    } catch (_) {}
    if (mounted) setState(() => _briefPlaying = false);
  }

  // ---- chat
  void _addMsg(String role, String text, {Map<String, dynamic>? item, String? url}) {
    _chat.add(<String, dynamic>{'role': role, 'text': text, if (item != null) 'item': item, if (url != null) 'url': url});
    if (mounted) setState(() {});
    Future<void>.delayed(const Duration(milliseconds: 60), () {
      if (_chatScroll.hasClients) _chatScroll.jumpTo(_chatScroll.position.maxScrollExtent);
    });
  }

  Future<void> _ask(String raw) async {
    final String text = raw.trim();
    if (text.isEmpty || _thinking) return;
    _voice.stop();
    _q.clear();
    _addMsg('me', text);
    final String quick = _local(text);
    if (quick.isNotEmpty) {
      _addMsg('ai', quick);
      _voice.say(quick);
      return;
    }
    setState(() {
      _thinking = true;
      _mode = 'thinking';
    });
    final Map<String, dynamic> msg = <String, dynamic>{'role': 'ai', 'text': ''};
    _chat.add(msg);
    String full = '';
    int spoken = 0;
    void flush(bool fin) {
      final String rest = full.substring(spoken);
      if (fin) {
        if (rest.trim().isNotEmpty) _voice.say(rest);
        spoken = full.length;
        return;
      }
      final RegExpMatch? m = RegExp(r'^[\s\S]*?[.!?](?=\s|$)').firstMatch(rest);
      if (m == null) return;
      _voice.say(m.group(0)!);
      spoken += m.group(0)!.length;
    }

    await _api.chat(text, _hist, onDelta: (String d) {
      full += d;
      msg['text'] = full;
      if (mounted) setState(() => _mode = _voice.speaking ? 'speaking' : 'thinking');
      flush(false);
      if (_chatScroll.hasClients) _chatScroll.jumpTo(_chatScroll.position.maxScrollExtent);
    }, onDone: (Map<String, dynamic> d) {
      full = full.trim();
      msg['text'] = full.isEmpty ? '…' : full;
      flush(true);
      final List<Map<String, dynamic>> hist = _list(d['messages']);
      if (hist.isNotEmpty) _hist = hist;
      for (final Map<String, dynamic> a in _list(d['actions'])) {
        _handleAction(a);
      }
    }, onError: (String e) {
      msg['text'] = (full + '\n[' + e + ']').trim();
    });
    if (!mounted) return;
    setState(() {
      _thinking = false;
      _mode = _voice.speaking ? 'speaking' : 'idle';
    });
    _load();
  }

  void _handleAction(Map<String, dynamic> a) {
    final String t = _s(a['type']);
    if (t == 'open') {
      final String u = _s(a['url']);
      _addMsg('sys', 'Opening ${_s(a['label'])}.', url: u);
      launchUrl(Uri.parse(u), mode: LaunchMode.externalApplication).catchError((Object _) => false);
    } else if (t == 'memory') {
      setState(() => _snap['memory'] = a['value']);
    } else if (t == 'reminders') {
      _loadReminders();
    } else if (t == 'confirm') {
      _addMsg('confirm', _s(_map(a['item'])['summary']), item: _map(a['item']));
    }
  }

  Future<void> _decide(Map<String, dynamic> msg, String decision) async {
    final Map<String, dynamic> item = _map(msg['item']);
    setState(() => msg['busy'] = true);
    try {
      final dynamic rec = await _api.post('/api/act', <String, dynamic>{'item': item, 'decision': decision});
      final Map<String, dynamic> r = _map(rec);
      msg['status'] = _s(r['status']);
      msg['result'] = _s(r['result']);
      if (msg['status'] == 'done') {
        _voice.say(_s(r['result']).isEmpty ? 'Done.' : _s(r['result']));
        _load();
      } else if (msg['status'] == 'failed') {
        _voice.say('That didn\'t go through. ' + _s(r['result']));
      }
    } catch (e) {
      msg['status'] = 'failed';
      msg['result'] = e.toString();
    }
    if (mounted) setState(() => msg['busy'] = false);
  }

  // a few answers that never need the brain
  String _local(String raw) {
    final String t = raw.toLowerCase().replaceAll(RegExp(r'^june,?\s*'), '').replaceAll(RegExp(r'[.!?]+$'), '').trim();
    if (RegExp(r'^(what time is it|time)$').hasMatch(t)) {
      final DateTime d = DateTime.now();
      final int h = d.hour % 12 == 0 ? 12 : d.hour % 12;
      return 'It\'s $h:${d.minute.toString().padLeft(2, '0')} ${d.hour < 12 ? 'AM' : 'PM'}.';
    }
    if (RegExp(r'^(stop|quiet|shush|never ?mind|cancel)$').hasMatch(t)) {
      _voice.stop();
      return ' ';
    }
    if (RegExp(r'^(stop|pause|turn off|kill|mute) (the )?(radio|music)$|^radio off$').hasMatch(t)) {
      _radioStop();
      return 'Radio off.';
    }
    final RegExpMatch? rp = RegExp(r'^(?:play|put on|turn on|start)\s+(?:the\s+|some\s+)?(radio|music|.+?)(?:\s+radio)?$').firstMatch(t);
    if (rp != null) {
      final String want = rp.group(1)!.replaceAll(' ', '');
      int i = want == 'radio' || want == 'music' ? 0 : _radio.indexWhere((List<String> r) => r[2].split(',').any((String k) => want.contains(k)));
      if (i >= 0) {
        _radioPlay(i);
        return ' ';
      }
    }
    return '';
  }

    <String>['Radio Paradise', 'https://stream.radioparadise.com/aac-128', 'paradise,radioparadise'],
    <String>['Paradise Mellow', 'https://stream.radioparadise.com/mellow-128', 'mellow'],

  AudioPlayer? _radioPlayer;
  html.AudioElement? _radioEl;

  Future<void> _radioPlay(int i) async {
    _radioStop();
    final List<String> st = _radio[i];
    setState(() => _radioName = st[0]);
    try {
      if (kIsWeb) {
        final html.AudioElement a = html.AudioElement();
        a.src = st[1];
        a.volume = 0.7;
        _radioEl = a;
        await a.play().catchError((Object _) {});
      } else {
        final AudioPlayer p = AudioPlayer();
        try {
          p.setAudioContext(AudioContext(iOS: AudioContextIOS(category: AVAudioSessionCategory.playback), android: AudioContextAndroid()));
        } catch (_) {}
        p.setVolume(0.7);
        _radioPlayer = p;
        await p.play(UrlSource(st[1]));
      }
    } catch (e) {
      _addMsg('sys', 'That station wouldn\'t start: ' + e.toString());
      setState(() => _radioName = '');
    }
  }

  void _radioStop() {
    try {
      _radioEl?.pause();
      _radioEl = null;
    } catch (_) {}
    try {
      _radioPlayer?.stop();
      _radioPlayer?.dispose();
      _radioPlayer = null;
    } catch (_) {}
    if (_radioName.isNotEmpty && mounted) setState(() => _radioName = '');
  }

  Widget _upcomingCard() {
    List<Map<String, dynamic>> ups;
    if (_couple) {
      ups = _list(_snap['upcoming']);
    } else {
      ups = <Map<String, dynamic>>[
        for (final Map<String, dynamic> w in _list(_snap['weddings'])) <String, dynamic>{'when': _s(w['weddingDate']), 'kind': 'Wedding', 'who': _s(w['couple']), 'mode': _s(w['venue'])},
        for (final Map<String, dynamic> c in _list(_snap['consults'])) <String, dynamic>{'when': _s(c['startAt']), 'kind': _s(c['type']).isEmpty ? 'Consult' : _s(c['type']), 'who': _s(c['who']), 'mode': _s(c['mode']), 'time': true},
      ]..sort((Map<String, dynamic> a, Map<String, dynamic> b) => _s(a['when']).compareTo(_s(b['when'])));
    }
    return _card(
      title: 'Coming up',
      trailing: _couple && _snap['daysToGo'] != null ? _tag(_n(_snap['daysToGo']) > 0 ? '${_n(_snap['daysToGo']).toInt()} days to go' : 'today') : null,
      child: ups.isEmpty
          ? Text(_couple ? 'Nothing scheduled yet. Consults with vendors and the key dates from the app land here.' : 'Nothing booked ahead yet.', style: const TextStyle(color: _muted, fontSize: 13.5))
          : Column(children: ups.take(10).map((Map<String, dynamic> u) => _row(title: _s(u['who']), d: (u['time'] == true ? _when(_s(u['when'])) : _day(_s(u['when']))) + (_s(u['mode']).isNotEmpty ? ' · ' + _s(u['mode']) : ''), tag: _tag(_s(u['kind']), bg: _s(u['kind']) == 'Task' ? null : _sage, fg: _s(u['kind']) == 'Task' ? null : _plumDeep))).toList()),
    );
  }

  Widget _weatherCard() {
    final Map<String, dynamic> w = _map(_snap['weather']);
    final Map<String, dynamic> cur = _map(w['current']);
    final Map<String, dynamic> d = _map(w['daily']);
    final String place = _s(_map(_snap[_couple ? 'couple' : 'vendor'])['metro']).split(',').first;

    final List<dynamic> days = d['time'] is List ? d['time'] as List<dynamic> : <dynamic>[];
    return _card(
      title: 'Weather',
      trailing: _tag(place),
      child: cur.isEmpty
          ? const Text('Weather unavailable right now.', style: TextStyle(color: _muted))
          : Column(crossAxisAlignment: CrossAxisAlignment.start, children: <Widget>[
              Row(crossAxisAlignment: CrossAxisAlignment.end, children: <Widget>[
                Text('${_n(cur['temperature_2m']).round()}°', style: _serif(44, w: FontWeight.w600)),
                const SizedBox(width: 12),
                Expanded(child: Text('${_wmo(cur['weather_code'])}\nfeels ${_n(cur['apparent_temperature']).round()}° · wind ${_n(cur['wind_speed_10m']).round()} mph', style: const TextStyle(color: _muted, fontSize: 12.5, height: 1.4))),
              ]),
              if (wx.isNotEmpty) Padding(padding: const EdgeInsets.only(top: 6), child: Text('Your day: ${_s(wx['text'])}, high ${_s(wx['hi'])}°, rain ${_s(wx['rain'])}%, sunset ${_s(wx['sunset'])}', style: const TextStyle(color: _goldSoft, fontSize: 13))),
              const SizedBox(height: 10),
              Row(children: <Widget>[
                for (int i = 1; i < math.min(6, days.length); i++)
                  Expanded(
                    child: Column(children: <Widget>[
                      Text(_day(_s(days[i])).split(',').first, style: const TextStyle(color: _ivory, fontSize: 11.5, fontWeight: FontWeight.w700)),
                      Icon(_wmoIcon((d['weather_code'] as List<dynamic>)[i]), size: 18, color: _gold),
                      Text('${_n((d['temperature_2m_max'] as List<dynamic>)[i]).round()}°/${_n((d['temperature_2m_min'] as List<dynamic>)[i]).round()}°', style: const TextStyle(color: _muted, fontSize: 11)),
                    ]),
                  ),
              ]),
              for (final Map<String, dynamic> a in _list(w['nws'])) Padding(padding: const EdgeInsets.only(top: 8), child: Text('⚠ ${_s(a['headline']).isEmpty ? _s(a['event']) : _s(a['headline'])}', style: const TextStyle(color: _rose, fontSize: 12))),
            ]),
    );
  }

  Widget _moneyCard() {
    final Map<String, dynamic> m = _map(_snap['money']);
    if (_couple) {
      final num bt = _n(m['budgetTotal']);
      final num committed = _n(m['committed']);
      final double pct = bt > 0 ? math.min(1, committed / bt) : 0;
      return _card(
        title: 'Money',
        trailing: bt > 0 ? _tag('${(pct * 100).round()}% of ${_money(bt)}') : null,
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: <Widget>[
          Row(children: <Widget>[_stat(bt > 0 ? _money(bt) : '–', 'budget'), _stat(_money(committed), 'committed'), _stat(_money(m['invoicesOpen']), 'invoices open', color: _n(m['overdue']) > 0 ? _red : null)]),
          if (bt > 0) ...<Widget>[
            const SizedBox(height: 10),
            ClipRRect(borderRadius: BorderRadius.circular(999), child: LinearProgressIndicator(value: pct, minHeight: 8, backgroundColor: Colors.black.withOpacity(.25), color: committed > bt ? _rose : _gold)),
            const SizedBox(height: 6),
            Text(committed > bt ? '${_money(committed - bt)} over budget' : '${_money(bt - committed)} left to allocate · paid so far ${_money(m['paid'])}', style: const TextStyle(color: _muted, fontSize: 12.5)),
          ] else
            const Padding(padding: EdgeInsets.only(top: 8), child: Text('Set a budget in the app, or say "our budget is 25,000".', style: TextStyle(color: _muted, fontSize: 12.5))),
          for (final Map<String, dynamic> i in _list(m['open'])) _row(title: '${_s(i['vendor'])} · ${_s(i['title'])}', d: _s(i['dueDate']).isEmpty ? 'No due date' : 'Due ' + _day(_s(i['dueDate'])), side: Text(_money(i['total']), style: const TextStyle(color: _goldSoft, fontWeight: FontWeight.w700))),
        ]),
      );
    }
    return _card(
      title: 'Money',
      child: m.isEmpty
          ? const Text('No invoices yet.', style: TextStyle(color: _muted))
          : Column(children: <Widget>[
              Row(children: <Widget>[_stat(_money(m['outstanding']), 'outstanding'), _stat(_money(m['overdue']), 'overdue', color: _n(m['overdue']) > 0 ? _red : null), _stat(_money(m['collected30']), 'collected, 30 days')]),
              for (final Map<String, dynamic> i in _list(m['open'])) _row(title: '${_s(i['couple'])} · ${_s(i['title'])}', d: _s(i['dueDate']).isEmpty ? 'No due date' : 'Due ' + _day(_s(i['dueDate'])), tag: i['overdue'] == true ? _tag('overdue', bg: _red, fg: Colors.white) : null, side: Text(_money(i['total']), style: const TextStyle(color: _goldSoft, fontWeight: FontWeight.w700))),
            ]),
    );
  }

  Widget _remindersCard() {
    final List<Map<String, dynamic>> list = _list(_snap['reminders']);
    final String now = DateTime.now().toUtc().toIso8601String();
    return _card(
      title: 'Reminders',
      sub: 'Say "remind me Friday to call the florist", or add one here.',
      child: Column(children: <Widget>[
        Row(children: <Widget>[
          Expanded(
            child: TextField(
              controller: _remText,
              style: const TextStyle(color: _ivory, fontSize: 13.5),
              decoration: InputDecoration(hintText: 'What should I remind you of?', hintStyle: const TextStyle(color: _muted, fontSize: 13), isDense: true, filled: true, fillColor: Colors.black.withOpacity(.3), contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10), border: OutlineInputBorder(borderRadius: BorderRadius.circular(12), borderSide: BorderSide(color: _gold.withOpacity(.35)))),
            ),
          ),
          const SizedBox(width: 8),
          _btn(_remWhen == null ? 'When' : _when(_remWhen!.toIso8601String()), () async {
            final DateTime now = DateTime.now();
            final DateTime? d = await showDatePicker(context: context, initialDate: now, firstDate: now, lastDate: now.add(const Duration(days: 730)));
            if (d == null || !mounted) return;
            final TimeOfDay? t = await showTimePicker(context: context, initialTime: const TimeOfDay(hour: 9, minute: 0));
            if (t == null) return;
            setState(() => _remWhen = DateTime(d.year, d.month, d.day, t.hour, t.minute));
          }),
          const SizedBox(width: 6),
          _btn('Add', () {
            if (_remText.text.trim().isEmpty || _remWhen == null) {
              _addMsg('sys', 'Give the reminder a text and a time.');
              return;
            }
            final DateTime w = _remWhen!;
            final String local = '${w.year}-${w.month.toString().padLeft(2, '0')}-${w.day.toString().padLeft(2, '0')}T${w.hour.toString().padLeft(2, '0')}:${w.minute.toString().padLeft(2, '0')}';
            _remPost(<String, dynamic>{'text': _remText.text.trim(), 'when': local});
            _remText.clear();
            setState(() => _remWhen = null);
          }, gold: true),
        ]),
        if (list.isEmpty) const Padding(padding: EdgeInsets.only(top: 10), child: Text('No reminders set.', style: TextStyle(color: _muted, fontSize: 13))),
        for (final Map<String, dynamic> r in list.take(12))
          _row(
            title: _s(r['text']),
            d: _when(_s(r['at'])),
            tag: _s(r['at']).compareTo(now) <= 0 ? _tag('due', bg: _rose, fg: _plumDeep) : null,
            side: Row(mainAxisSize: MainAxisSize.min, children: <Widget>[_btn('Done', () => _remPost(<String, dynamic>{'done': _s(r['id'])})), const SizedBox(width: 6), _btn('✕', () => _remPost(<String, dynamic>{'remove': _s(r['id'])}))]),
          ),
      ]),
    );
  }

  Widget build(BuildContext context) {
    final List<Widget> left = <Widget>[_console(), _briefCard(), _waitingCard(), _upcomingCard()];
    final List<Widget> right = <Widget>[
      _weatherCard(),
      if (_couple) _teamCard(),
      _moneyCard(),
      if (_couple) _guestsCard() else _pageCard(),
      _remindersCard(),
      _memoryCard(),
      _radioCard(),
    ];
    Widget body;
    if (_err.isNotEmpty) {
      body = Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: <Widget>[Text('June couldn\'t load your data.', style: _serif(22)), const SizedBox(height: 8), Text(_err, style: const TextStyle(color: _muted, fontSize: 13)), const SizedBox(height: 14), _btn('Try again', () => _load(first: true), gold: true)]));
    } else {
      body = LayoutBuilder(builder: (BuildContext c, BoxConstraints box) {
        final bool wide = box.maxWidth >= 900;
        return SingleChildScrollView(
          padding: const EdgeInsets.fromLTRB(16, 12, 16, 28),
          child: wide
              ? Row(crossAxisAlignment: CrossAxisAlignment.start, children: <Widget>[
                  Expanded(flex: 8, child: Column(children: left)),
                  const SizedBox(width: 14),
                  Expanded(flex: 5, child: Column(children: right)),
                ])
              : Column(children: <Widget>[...left, ...right]),
        );
      });
    }
    return Container(
      width: widget.width,
      height: widget.height,
      decoration: const BoxDecoration(
        gradient: LinearGradient(begin: Alignment.topLeft, end: Alignment.bottomRight, colors: <Color>[Color(0xFF241022), Color(0xFF2E1530), Color(0xFF3D1C3B)]),
        borderRadius: BorderRadius.all(Radius.circular(22)),
      ),
      clipBehavior: Clip.antiAlias,
      child: _loading && _snap.isEmpty && _err.isEmpty
          ? const Center(child: SizedBox(width: 26, height: 26, child: CircularProgressIndicator(strokeWidth: 2, color: _gold)))
          : body,
    );
  }
