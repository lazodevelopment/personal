  // A couple who tapped "Check your date" on a vendor page arrives with
  // ?vendor=<id>. goNamed drops the query, so it rides in a preference and
  // the couple home opens that vendor's inquiry the moment they land.
  // The role gate stores the visitor's pre-auth choice the same way.
  Future<void> _stashDeepLink() async {
    try {
      final SharedPreferences p = await SharedPreferences.getInstance();
      final String? v = Uri.base.queryParameters['vendor'];
      if (v != null && v.trim().isNotEmpty) {
        await p.setString(kPrefVendor, v.trim());
      }
      final String vendorId = (p.getString(kPrefVendor) ?? '').trim();
      if (vendorId.isNotEmpty) {
        try {
          final DocumentSnapshot<Map<String, dynamic>> vs =
              await FirebaseFirestore.instance
                  .collection('vendors')
                  .doc(vendorId)
                  .get();
          final String nm = (vs.data()?['name'] ?? '').toString();
          if (nm.isNotEmpty && mounted) setState(() => _pendingVendorName = nm);
        } catch (_) {}
      }
      // A couple who tapped Save on a vendor page: ?save=<vendorId>.
      final String? sv = Uri.base.queryParameters['save'];
      if (sv != null && sv.trim().isNotEmpty) {
        await p.setString(kPrefSave, sv.trim());
        if (mounted && FirebaseAuth.instance.currentUser == null) {
          setState(() {
            _rolePick = 'couple';
            _mode = _Mode.signup;
          });
        }
      }
      // A couple who picked a website design: ?template=sage[&palette=..].
      final String? tp = Uri.base.queryParameters['template'];
      if (tp != null && tp.trim().isNotEmpty) {
        await p.setString(kPrefTemplate, tp.trim().toLowerCase());
        final String? pal = Uri.base.queryParameters['palette'];
        if (pal != null && pal.trim().isNotEmpty)
          await p.setString(kPrefPalette, pal.trim().toLowerCase());
        if (mounted && FirebaseAuth.instance.currentUser == null) {
          setState(() {
            _rolePick = 'couple';
            _mode = _Mode.signup;
            _pendingTemplateName = tp.trim();
          });
        }
      }
      // A vendor claiming their listing: the site sends ?claim=<id or name>.
      final String? cl = Uri.base.queryParameters['claim'];
      if (cl != null && cl.trim().isNotEmpty) {
        await p.setString(kPrefClaim, cl.trim());
      }
      final String claim = (p.getString(kPrefClaim) ?? '').trim();
      if (claim.isNotEmpty) {
        String label = claim;
        if (claim.length > 12 && !claim.contains(' ')) {
          try {
            final DocumentSnapshot<Map<String, dynamic>> vs =
                await FirebaseFirestore.instance
                    .collection('vendors')
                    .doc(claim)
                    .get();
            label = (vs.data()?['name'] ?? claim).toString();
          } catch (_) {}
        }
        if (mounted && FirebaseAuth.instance.currentUser == null) {
          setState(() {
            _pendingClaimName = label;
            _rolePick = 'vendor';
            _mode = _Mode.signup;
          });
        }
        return;
      }
      final String pending = p.getString(kPrefRole) ?? '';
      // v20: only while they are still on Welcome - never pull someone off
      // the login form they already opened.
      if ((pending == 'couple' || pending == 'vendor') &&
          FirebaseAuth.instance.currentUser == null &&
          mounted &&
          _mode == _Mode.welcome) {
        setState(() {
          _rolePick = pending;
          _mode = _Mode.signup;
        });
      }
    } catch (_) {}
  }

          (q['intent'] ?? '').isNotEmpty ||
          (q['names'] ?? '').isNotEmpty ||

      const List<String> keep = <String>[

        'intent',
        'names',

        'metro',
        'design',
        'template',
        'source',
        'guests',
        'month',
        'utm_source',
        'utm_medium',
        'utm_campaign',
        'utm_content',
        'utm_term',
        'fbclid'
      ];
      final Map<String, String> out = <String, String>{};
      for (final String k in keep) {
        final String v = (q[k] ?? '').trim();
        if (v.isNotEmpty) out[k] = v.length > 200 ? v.substring(0, 200) : v;

          }
        });
      }
    } catch (_) {}
  }

  // ---------------- v14: PASSWORD STRENGTH ----------------
  // 0 = too short, 1 weak, 2 fair, 3 good, 4 strong. Length carries most of
  // the weight, variety the rest. Runs, keyboard walks, common words, and
  // anything that is just their own name or email pull it down - those are
  // the first things a guesser tries.
  static const List<String> _kWeakWords = <String>[
    'password',
    'passw0rd',
    'letmein',
    'welcome',
    'iloveyou',
    'wedding',
    'married',
    'lazo',
    'qwerty',
    'asdf',
    'zxcv',
    '1234',
    '0000',
    '1111',
    'abcd',
  ];

  static int _passScore(String p, {String name = '', String email = ''}) {
    if (p.isEmpty) return 0;
    final int n = p.length;
    if (n < 6) return 0;
    double s = 0;
    if (n >= 6) s += 1;
    if (n >= 8) s += 1;
    if (n >= 12) s += 1;
    if (n >= 16) s += .5;
    int classes = 0;
    if (RegExp(r'[a-z]').hasMatch(p)) classes++;
    if (RegExp(r'[A-Z]').hasMatch(p)) classes++;
    if (RegExp(r'\d').hasMatch(p)) classes++;
    if (RegExp(r'[^A-Za-z0-9]').hasMatch(p)) classes++;
    s += (classes - 1) * .6;
    final String low = p.toLowerCase();
    if (RegExp(r'(.)\1\1').hasMatch(low)) s -= 1;
    for (final String w in _kWeakWords) {
      if (low.contains(w)) {
        s -= 1.5;
        break;
      }
    }
    final String em = email.split('@').first.toLowerCase().trim();
    if (em.length >= 3 && low.contains(em)) s -= 1.5;
    for (final String w in name.toLowerCase().split(RegExp(r'\s+'))) {
      if (w.length >= 3 && low.contains(w)) {
        s -= 1.5;
        break;
      }
    }
    if (s < 1) return 1;
    if (s >= 4) return 4;
    return s.floor();
  }

  // What would move the needle - one thing, the biggest one.
  static String _passHint(String p, int score) {
    if (score >= 4) return 'That will hold.';
    final int n = p.length;
    if (n < 6) return 'At least 6 characters.';
    if (n < 8) return 'Make it 8 or more - length beats clever.';
    int classes = 0;
    if (RegExp(r'[a-z]').hasMatch(p)) classes++;
    if (RegExp(r'[A-Z]').hasMatch(p)) classes++;
    if (RegExp(r'\d').hasMatch(p)) classes++;
    if (RegExp(r'[^A-Za-z0-9]').hasMatch(p)) classes++;
    final String low = p.toLowerCase();
    if (_kWeakWords.any((String w) => low.contains(w)) ||
        RegExp(r'(.)\1\1').hasMatch(low)) {
      return 'Skip the common words and runs.';
    }
    if (n < 12) return 'A few more characters would do it.';
    if (classes < 3) return 'Mix in a number or a symbol.';
    return 'Try a phrase nobody else would.';
  }

  Future<void> _buildAdvance() async {

  Future<void> _routeIfSignedIn() async {

    try {

  // v21: a couple goes to the dashboard only once couples/{uid} exists.
  // Until then they are owed the Make-it-yours step - whether they got
  // here from the Router after signup, from a cold open, or after login.
  Future<void> _routeCouple(String uid, {Map<String, dynamic>? userDoc}) async {
    bool needsSetup = false;
    try {
      final DocumentSnapshot<Map<String, dynamic>> c =
          await _coupleRef(uid).get();
      needsSetup = !c.exists;
    } catch (e) {
      debugPrint('LAZO couple read: $e');

    if (!mounted) return;
    if (!needsSetup) {
      _goCouple();
      return;

              FirebaseAuth.instance.currentUser?.displayName ??

          .toString()
          .trim();
      if (dn.isNotEmpty) _name.text = dn;
    }
    setState(() => _resolving = false);
    _setMode(_Mode.date);
  }

  Future<void> _signUp() async {
    setState(() => _touched.addAll(<String>['name', 'email', 'pass', 'pass2']));
    if (_name.text.trim().isEmpty) {
      _fail('Tell us your name - vendors reply to a person.',
          focus: _nameFocus);
      return;
    }
    if (!_emailLooksValid(_email.text)) {
      _fail('That email doesn’t look right.', focus: _emailFocus);
      return;
    }
    if (_pass.text.length < 6) {
      _fail('Password needs at least 6 characters.', focus: _passFocus);
      return;
    }
    if (_pass.text != _pass2.text) {
      _fail('Passwords don’t match.', focus: _pass2Focus);
      return;
    }
    if (!_tosAccepted) {
      _fail('Please agree to the Terms of Service to continue.');
      return;
    }
    setState(() {
      _busy = true;
      _error = null;
      _notice = null;
    });
    try {
      final cred = await FirebaseAuth.instance.createUserWithEmailAndPassword(
          email: _email.text.trim(), password: _pass.text);
      final u = cred.user!;
      // v17: the user doc goes first. FlutterFlow's auth listener can route
      // to the Router page the instant the account exists, and the Router
      // reads users/{uid}.role - so the role must already be there.
      // FlutterFlow may have created users/{uid} already; merge keeps it a
      // single doc either way. created_time only lands on a fresh doc.
      final DocumentSnapshot<Map<String, dynamic>> existing =
          await _userRef(u.uid).get();
      await _userRef(u.uid).set(<String, dynamic>{
        'uid': u.uid,
        'email': u.email,
        'display_name': _name.text.trim(),
        if (!existing.exists) 'created_time': FieldValue.serverTimestamp(),
        'role': _rolePick,
        'tosAcceptedAt': FieldValue.serverTimestamp(),
        if (!(existing.data()?.containsKey('signupSource') ?? false))
          'signupSource': _signupSource(),
      }, SetOptions(merge: true));
      if (_name.text.trim().isNotEmpty) {
        try {
          await u.updateDisplayName(_name.text.trim());
        } catch (_) {}
      }
      await _clearPendingRole();
      // v20: let the browser / password manager offer to save these.
      TextInput.finishAutofillContext();
      if (!mounted) return;
      if (_rolePick == 'couple') {
        _setMode(_Mode.date);
      } else {
        _goVendor();
      }
    } on FirebaseAuthException catch (e) {
      _fail(_authMessage(e.code),
          focus: e.code == 'email-already-in-use' ? _emailFocus : null);
    } catch (e) {
      _fail('Something went wrong. Try again.${_why(e)}');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _logIn() async {
    setState(() {
      _busy = true;
      _error = null;
      _notice = null;
    });
    try {
      final cred = await FirebaseAuth.instance.signInWithEmailAndPassword(
          email: _email.text.trim(), password: _pass.text);
      final u = cred.user!;
      final snap = await _userRef(u.uid).get();
      final role = snap.data()?['role'] as String?;
      TextInput.finishAutofillContext();
      if (!mounted) return;
      if (role == 'couple') {
        await _routeCouple(u.uid, userDoc: snap.data());
      } else if (role == 'vendor') {
        _goVendor();
      } else {
        _setMode(_Mode.rolepick);
      }
    } on FirebaseAuthException catch (e) {
      final bool wrong = e.code == 'wrong-password' ||
          e.code == 'invalid-credential' ||
          e.code == 'user-not-found';
      _fail(_authMessage(e.code), focus: wrong ? _passFocus : null);
      if (wrong) _pass.clear();
    } catch (_) {
      _fail('Something went wrong. Try again.');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _pickPhoto() async {
    _dropKeyboard();
    try {
      // v18: FileType.image opens the photo library on iOS and Android and
      // an image-only chooser on the web. FileType.custom opened the
      // documents browser, which is not where anyone keeps a photo of the
      // two of them.
      final FilePickerResult? res = await FilePicker.platform.pickFiles(
        type: FileType.image,
        withData: true,
      );
      if (res == null || res.files.isEmpty) return;
      final Uint8List? b = res.files.first.bytes;
      if (b == null) return;
      final String ext = (res.files.first.extension ?? 'jpg').toLowerCase();
      Uint8List out = b;
      String type = ext == 'png'
          ? 'image/png'
          : (ext == 'webp' ? 'image/webp' : 'image/jpeg');
      // Only these three can be shown everywhere the photo is used. The
      // iPhone library often hands over HEIC, which the web dashboard
      // cannot draw - so anything else is decoded and re-encoded.
      final bool webFormat =
          ext == 'jpg' || ext == 'jpeg' || ext == 'png' || ext == 'webp';
      // A 12 MB phone photo becomes the hero on every open of the dashboard.
      // Anything over 2.5 MB is downscaled to 1600px before it leaves the device.
      if (b.lengthInBytes > 2500 * 1024 || !webFormat) {
        bool ok = false;
        try {
          final ui.Codec codec =
              await ui.instantiateImageCodec(b, targetWidth: 1600);
          final ui.FrameInfo fi = await codec.getNextFrame();
          final ByteData? bd =
              await fi.image.toByteData(format: ui.ImageByteFormat.png);
          if (bd != null) {
            out = bd.buffer.asUint8List();
            type = 'image/png';
            ok = true;
          }
        } catch (_) {}
        // v20: a format this device cannot decode (HEIC in a browser) must
        // not be uploaded mislabelled - the dashboard could never show it.
        if (!ok && !webFormat) {
          if (mounted) {
            setState(() => _error =
                'That photo format isn’t supported here - try a JPG or PNG.');
          }
          return;
        }
      }
      if (!mounted) return;
      setState(() {
        _error = null;
        _photoBytes = out;
        _photoType = type;
      });
    } catch (_) {
      if (mounted) setState(() => _error = 'Couldn’t open your photos.');
    }
  }

  Future<void> _pickDate() async {
    _dropKeyboard();
    // v20: the picker asserts first <= initial <= last; a date picked just
    // before midnight and reopened after it would have tripped that.
    final DateTime now = DateTime.now();
    final DateTime first = DateTime(now.year, now.month, now.day);
    final DateTime last = first.add(const Duration(days: 365 * 4));
    DateTime initial = _pickedDate ?? first.add(const Duration(days: 270));
    if (initial.isBefore(first)) initial = first;
    if (initial.isAfter(last)) initial = last;
    final picked = await showDatePicker(
      context: context,
      initialDate: initial,
      firstDate: first,
      lastDate: last,
      builder: (ctx, child) => Theme(
        data: Theme.of(ctx).copyWith(
          colorScheme: const ColorScheme.light(
              primary: plum, onPrimary: ivory, onSurface: ink),
        ),
        child: child!,
      ),
    );
    if (picked != null) {
      _tick(_Tick.select);
      setState(() => _pickedDate = picked);
    }
  }

  Future<void> _finishOnboarding({required bool skipDate}) async {
    final u = FirebaseAuth.instance.currentUser;
    if (u == null) return;
    _dropKeyboard();
    final bool hasDate = !skipDate && _pickedDate != null;
    final String city = _metroLabel();
    // v19: the steps are the real work, in the order it happens.
    final List<String> steps = <String>[
      if (hasDate) 'Saving your date',
      if (_photoBytes != null) 'Adding your photo',
      city.isEmpty ? 'Setting up vendor search' : 'Finding vendors in $city',
      'Building your dashboard',
    ];
    setState(() {
      _busy = true;
      _error = null;
      _building = true;
      _buildSteps = steps;
      _buildStep = 0;
      _buildStepAt = DateTime.now();
    });
    try {
      final data = <String, dynamic>{'createdAt': FieldValue.serverTimestamp()};
      if (hasDate) {
        data['weddingDate'] = Timestamp.fromDate(_pickedDate!);
        await _buildAdvance();
      }
      // v15: the city and the partner ride along when given. `names` is the
      // display pair the dashboard and vendors read ("Jordan & Sam").
      if (_metro.isNotEmpty) data['metroId'] = _metro;
      final String partner = _partner.text.trim();
      if (partner.isNotEmpty) {
        data['partnerName'] = partner;
        final String me = _firstWord(
            _name.text.trim().isNotEmpty ? _name.text : (u.displayName ?? ''));
        data['names'] =
            me.isEmpty ? _firstWord(partner) : '$me & ${_firstWord(partner)}';
      }
      if (_photoBytes != null) {
        final String ext = _photoType == 'image/png'
            ? 'png'
            : (_photoType == 'image/webp' ? 'webp' : 'jpg');
        final ref =
            FirebaseStorage.instance.ref('couples/${u.uid}/couple.$ext');
        await ref.putData(
            _photoBytes!, SettableMetadata(contentType: _photoType));
        data['photoUrl'] = await ref.getDownloadURL();
        await _buildAdvance();
      }
      // the couple doc is what vendor search and the dashboard read
      await _coupleRef(u.uid).set(data, SetOptions(merge: true));
      await _buildAdvance();
      // the last tick, then a beat to see the list complete
      await _buildAdvance();
      if (!mounted) return;
      await Future<void>.delayed(
          Duration(milliseconds: _reducedMotion(context) ? 200 : 520));
      if (!mounted) return;
      _goCouple();
    } catch (e) {
      if (!mounted) return;
      setState(() => _building = false);
      _fail('Could not save. Try again.${_why(e)}');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _resetPassword() async {
    if (_email.text.trim().isEmpty) {
      _fail('Enter your email above first.', focus: _emailFocus);
      return;
    }
    try {
      await FirebaseAuth.instance
          .sendPasswordResetEmail(email: _email.text.trim());
      setState(() {
        _error = null;
        _notice = 'Reset link sent - check your email.';
      });
    } on FirebaseAuthException catch (e) {
      _fail(_authMessage(e.code));
    } catch (e) {
      _fail('Couldn’t send the reset link.${_why(e)}');
    }
  }

  String _authMessage(String code) {
    switch (code) {
      case 'email-already-in-use':
        return 'That email already has an account - log in instead.';
      case 'invalid-email':
        return 'That email doesn’t look right.';
      case 'weak-password':
        return 'Password needs at least 6 characters.';
      case 'user-not-found':
      case 'wrong-password':
      case 'invalid-credential':
        return 'Email or password doesn’t match.';
      case 'too-many-requests':
        return 'Too many tries - wait a minute.';
      case 'network-request-failed':
        return 'No connection. Check your network and try again.';
      default:
        return 'Something went wrong ($code).';
    }
  }

  // v17: the real reason, so a rules denial can be read off the screen.
  String _why(Object e) {
    String t;
    if (e is FirebaseException) {
      t = e.code + (e.message == null ? '' : ': ${e.message}');
    } else {
      t = e.toString();
    }
    t = t.replaceAll('\n', ' ').trim();
    if (t.length > 160) t = '${t.substring(0, 160)}…';
    return t.isEmpty ? '' : ' ($t)';
  }

  String _prettyDate(DateTime d) {
    const months = [

      'March',
      'April',
      'May',
      'June',
      'July',
      'August',
      'September',
      'October',
      'November',
      'December'
    ];
    return '${months[d.month - 1]} ${d.day}, ${d.year}';
  }

  // v14: screens slide in from the right and leave the way they came.
  // Welcome - the thing underneath - only fades.
  Widget _viewSwitch(Widget child) {
    final bool reduced = _reducedMotion(context);
    return AnimatedSwitcher(
      duration: Duration(milliseconds: reduced ? 140 : 280),
      reverseDuration: Duration(milliseconds: reduced ? 140 : 200),
      switchInCurve: Curves.easeOutCubic,
      switchOutCurve: Curves.easeInCubic,
      layoutBuilder: (Widget? current, List<Widget> previous) => Stack(
        alignment: Alignment.topCenter,
        children: <Widget>[...previous, if (current != null) current],
      ),
      transitionBuilder: (Widget c, Animation<double> a) {
        final Key? k = c.key;
        final bool root = k is ValueKey<String> && k.value.endsWith(':root');
        final Widget faded = FadeTransition(opacity: a, child: c);
        if (reduced || root) return faded;
        return SlideTransition(
          position: Tween<Offset>(begin: const Offset(.06, 0), end: Offset.zero)
              .animate(a),
          child: faded,
        );
      },
      child: child,
    );
  }

          return Transform.translate(offset: Offset(dx, 0), child: ch);
        },
        child: child,
      );

  Widget _buildRow(int i) {
    final bool reduced = _reducedMotion(context);
    final bool done = i < _buildStep;
    final bool now = i == _buildStep;
    const Color green = Color(0xFF7FD3A9);
    final Widget mark;
    if (done) {
      mark = const Icon(Icons.check_circle_rounded,
          key: ValueKey<String>('done'), color: green, size: 18);
    } else if (now) {
      mark = Padding(
        key: const ValueKey<String>('now'),
        padding: const EdgeInsets.all(2),
        child: CircularProgressIndicator(strokeWidth: 2, color: gold),
      );
    } else {
      mark = Icon(Icons.circle_outlined,
          key: const ValueKey<String>('wait'),
          color: ivory.withOpacity(.35),
          size: 16);
    }
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 7),
      child: Row(children: <Widget>[
        SizedBox(
          width: 18,
          height: 18,
          child: AnimatedSwitcher(
            duration: Duration(milliseconds: reduced ? 0 : 220),
            switchInCurve: Curves.easeOutCubic,
            transitionBuilder: (Widget c, Animation<double> a) =>
                ScaleTransition(
                    scale: Tween<double>(begin: .6, end: 1).animate(a),
                    child: FadeTransition(opacity: a, child: c)),
            child: mark,
          ),
        ),
        const SizedBox(width: 12),
        Expanded(
          child: AnimatedDefaultTextStyle(
            duration: Duration(milliseconds: reduced ? 0 : 220),
            style: TextStyle(
                color: done
                    ? ivory.withOpacity(.85)
                    : (now ? ivory : ivory.withOpacity(.45)),
                fontSize: 14.5,
                fontWeight: now ? FontWeight.w600 : FontWeight.w500),
            child: Text(_buildSteps[i]),
          ),
        ),
      ]),
    );
  }

  // v14: four segments that fill with a ripple, a word for where it stands,
  // and the one thing that would move it. Shown only while there is a
  // password to talk about.
  Widget _strengthMeter() {
    final String p = _pass.text;
    final int score =
        p.isEmpty ? 0 : _passScore(p, name: _name.text, email: _email.text);
    const List<String> labels = <String>[
      'Too short',
      'Weak',
      'Fair',
      'Good',
      'Strong'
    ];
    const List<Color> colors = <Color>[
      Color(0xFFE08A8A),
      Color(0xFFE09A6A),
      Color(0xFFE6C270),
      gold,
      Color(0xFF7FD3A9),
    ];
    final Color c = colors[score];
    final bool reduced = _reducedMotion(context);
    return _grow(p.isEmpty
        ? null
        : Padding(
            padding: const EdgeInsets.only(top: 10, left: 4, right: 4),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: <Widget>[
                Row(children: <Widget>[
                  for (int i = 0; i < 4; i++)
                    Expanded(
                      child: Padding(
                        padding: EdgeInsets.only(right: i < 3 ? 4 : 0),
                        child: AnimatedContainer(
                          // each segment lands a beat after the last
                          duration: Duration(
                              milliseconds: reduced ? 0 : 220 + i * 70),
                          curve: Curves.easeOutCubic,
                          height: 4,
                          decoration: BoxDecoration(
                            color: i < score ? c : ivory.withOpacity(.18),
                            borderRadius: BorderRadius.circular(999),
                          ),
                        ),
                      ),
                    ),
                ]),
                const SizedBox(height: 6),
                Row(children: <Widget>[
                  AnimatedDefaultTextStyle(
                    duration: Duration(milliseconds: reduced ? 0 : 220),
                    curve: Curves.easeOut,
                    style: TextStyle(
                        color: c, fontSize: 12, fontWeight: FontWeight.w700),
                    child: Text(labels[score]),
                  ),
                  const SizedBox(width: 8),
                  Expanded(
                    child: Text(_passHint(p, score),
                        maxLines: 1,
                        overflow: TextOverflow.ellipsis,
                        style: TextStyle(
                            color: ivory.withOpacity(.65), fontSize: 11.5)),
                  ),
                ]),
              ],
            ),
          ));
  }

  Widget _field(TextEditingController c, String hint, TextInputType t,

      onChanged: (s) => setState(() {}),
      style: const TextStyle(
          color: ink, fontSize: 15, fontWeight: FontWeight.w500),
      decoration: _deco(hint, error: error),
    );
  }

        ],
        Text(busy ? 'One moment…' : label,
            style: TextStyle(
                fontSize: 15.5,
                fontWeight: outlined ? FontWeight.w600 : FontWeight.w700)),
      ],
    );
    return SizedBox(
      width: double.infinity,
      child: outlined
          ? OutlinedButton(
              onPressed: _h(onTap, tick),
              style: OutlinedButton.styleFrom(
                foregroundColor: fg,
                side: BorderSide(color: gold.withOpacity(.7), width: 1.2),
                padding: const EdgeInsets.symmetric(vertical: 16),
                shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(999)),
              ),
              child: child,
            )
          : FilledButton(
              onPressed: _h(onTap, tick),
              style: FilledButton.styleFrom(
                backgroundColor: bg,
                foregroundColor: fg,
                disabledBackgroundColor: gold.withOpacity(.35),
                disabledForegroundColor: plum.withOpacity(.55),
                padding: const EdgeInsets.symmetric(vertical: 16),
                shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(999)),
              ),
              child: child,
            ),
    );
  }
}

// ---------------- v14: MOTION ----------------
// Three rules, from the WWDC "Designing Fluid Interfaces" talk:
//  1. Answer the finger on pointer-DOWN, never only on release.
//  2. Critically damped by default. Overshoot only when something arrives
//     with momentum.
//  3. Reduced-motion users get a cross-fade - never a slide, never a shake.

bool _reducedMotion(BuildContext c) =>
    MediaQuery.maybeOf(c)?.disableAnimations ?? false;

// Haptics, in tiers, fired on the commit so the pulse and the visual change
// hit on the same frame. No-ops on web; no permission needed on phones.
enum _Tick { none, select, light, medium }

void _tick(_Tick t) {
  switch (t) {
    case _Tick.none:
      return;
    case _Tick.select:
      HapticFeedback.selectionClick();
      return;
    case _Tick.light:
      HapticFeedback.lightImpact();
      return;
    case _Tick.medium:
      HapticFeedback.mediumImpact();
      return;
  }
}

/// Wrap a button's onPressed. Null stays null, so disabled stays disabled.
VoidCallback? _h(VoidCallback? f, [_Tick t = _Tick.light]) => f == null
    ? null
    : () {
        _tick(t);
        f();
      };

/// A tappable that answers the instant the finger lands - 97% in 90ms on the
/// way down, a slower settle on the way up. Use it where the child paints its
/// own background; InkWell's ripple would land underneath and never show.
class _Press extends StatefulWidget {
  const _Press({
    super.key,
    required this.child,
    this.onTap,
    this.scale = .97,
    this.dim = .88,
    this.enabled = true,
    this.tick = _Tick.light,
  });
  final Widget child;
  final VoidCallback? onTap;
  final double scale;
  final double dim;
  final bool enabled;
  final _Tick tick;
  @override
  State<_Press> createState() => _PressState();
}

class _PressState extends State<_Press> {
  bool _down = false;
  void _set(bool v) {
    if (_down != v && mounted) setState(() => _down = v);
  }

  @override
  Widget build(BuildContext context) {
    final bool on = _down && widget.enabled && !_reducedMotion(context);
    final VoidCallback? tap = widget.enabled ? widget.onTap : null;
    return MouseRegion(
      cursor: widget.enabled && widget.onTap != null
          ? SystemMouseCursors.click
          : MouseCursor.defer,
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTapDown: widget.enabled ? (_) => _set(true) : null,
        onTapUp: (_) => _set(false),
        onTapCancel: () => _set(false),
        onTap: tap == null
            ? null
            : () {
                _tick(widget.tick);
                tap();
              },
        child: AnimatedScale(
          scale: on ? widget.scale : 1,
          duration: Duration(milliseconds: on ? 90 : 240),
          curve: on ? Curves.easeOut : Curves.easeOutCubic,
          child: AnimatedOpacity(
            opacity: on ? widget.dim : 1,
            duration: Duration(milliseconds: on ? 60 : 200),
            child: widget.child,
          ),
        ),
      ),
    );
  }
}
