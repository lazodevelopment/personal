Future stashWelcomeParams() async {
  try {
    final Map<String, String> q = Uri.base.queryParameters;
    final bool ours = (q['from'] ?? '').startsWith('welcome') ||
        (q['intent'] ?? '').isNotEmpty ||
        (q['names'] ?? '').isNotEmpty ||
        (q['date'] ?? '').isNotEmpty;
    if (!ours) return;
    const List<String> keep = <String>[
      'from',
      'intent',
      'names',
      'date',
      'metro',
      'design',
      'template',
      'source',
      'guests',
      'month',
      'utm_source',
      'utm_medium',
      'utm_campaign',
      'utm_content',
      'utm_term',
      'fbclid'
    ];
    final Map<String, String> out = <String, String>{};
    for (final String k in keep) {
      final String v = (q[k] ?? '').trim();
      if (v.isNotEmpty) out[k] = v.length > 200 ? v.substring(0, 200) : v;
    }
    if (out.isEmpty) return;
    final SharedPreferences prefs = await SharedPreferences.getInstance();
    await prefs.setString('lazo_pending_welcome', jsonEncode(out));
  } catch (_) {}
}
